// Preserve raw body for webhook signature verification. Must be registered
// BEFORE any route. Overrides the built-in application/json parser.
server.removeContentTypeParser('application/json');
server.addContentTypeParser('application/json', { parseAs: 'buffer' }, (req, body, done) => {
  (req as { rawBody?: Buffer }).rawBody = body as Buffer;
  try {
    const parsed = JSON.parse((body as Buffer).toString('utf8')) as unknown;
    done(null, parsed);
  } catch (err) {
    done(err as Error, undefined);
  }
});

server.setReplySerializer((payload) => {
  if (typeof payload === 'string') return payload;
  return JSON.stringify(payload, (_, value) => {
    if (typeof value === 'bigint') return Number(value);
    if (
      value &&
      typeof value === 'object' &&
      typeof (value as { toNumber?: unknown }).toNumber === 'function'
    ) {
      return (value as { toNumber: () => number }).toNumber();
    }
    return value;
  });
});

import type { FastifyInstance } from 'fastify';

function isDomainError(
  err: unknown,
): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return typeof e.code === 'string' && typeof e.httpStatus === 'number' && typeof e.message === 'string';
}

function isZodError(err: unknown): err is { issues: unknown[] } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return (err as { name?: string }).name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function tripRouter(server: FastifyInstance): Promise<void> {
  server.setErrorHandler((error, request, reply) => {
    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({
        error: {
          code: error.code,
          message: error.message,
          ...(error.details ? { details: error.details } : {}),
          requestId: request.id,
        },
      });
    }
    if (isZodError(error)) {
      return reply.status(422).send({
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Validation failed',
          details: { issues: error.issues },
          requestId: request.id,
        },
      });
    }
    const anyErr = error as { statusCode?: number; code?: string; message?: string };
    const statusCode = anyErr.statusCode ?? 500;
    const isProd = process.env.NODE_ENV === 'production';
    return reply.status(statusCode).send({
      error: {
        code: anyErr.code ?? 'INTERNAL_ERROR',
        message: isProd ? 'Something went wrong' : (anyErr.message ?? 'Unknown error'),
        requestId: request.id,
      },
    });
  });

  const auth = { preHandler: requireRole('DRIVER') };

  server.get('/driver/config', auth, c.getConfig);
  server.get('/driver/me', auth, c.getMe);
  server.get('/driver/trips', auth, c.getMyTrips);
  server.get('/driver/trips/history', auth, c.getHistory);
  server.get('/driver/trips/:id', auth, c.getTrip);
  server.post('/driver/trips/:id/start', auth, c.postStartTrip);
  server.post('/driver/trips/:id/end', auth, c.postEndTrip);
  server.get('/driver/trips/:id/manifest', auth, c.getManifest);
  server.post('/driver/trips/:id/gps', auth, c.postGps);
  server.post('/driver/trips/:id/scan', auth, c.postScan);
  server.post('/driver/trips/:id/delay', auth, c.postDelay);
  server.post('/driver/trips/:id/cancel', auth, c.postCancelTrip);
  server.post('/driver/uploads', auth, c.postDriverUploads);
  server.post('/driver/me/documents', auth, c.postDriverDocuments);
}

  const xff = request.headers['x-forwarded-for'];

import { z } from 'zod';
import {
  DELAY_REASON,
  DELAY_MIN_MINUTES,
  DELAY_MAX_MINUTES,
  DELAY_MAX_NOTES_LENGTH,
} from './trip.types.js';

export const tripIdParamSchema = z.object({ id: z.string().uuid() });

export const gpsBodySchema = z
  .object({
    lat: z.number().min(-90).max(90),
    lng: z.number().min(-180).max(180),
    speedKmh: z.number().min(0).max(300),
    accuracyM: z.number().min(0).max(1000),
    recordedAt: z.string().datetime(),
  })
  .strict();

export const delayBodySchema = z
  .object({
    reason: z.enum(DELAY_REASON),
    delayMinutes: z.number().int().min(DELAY_MIN_MINUTES).max(DELAY_MAX_MINUTES),
    notes: z.string().max(DELAY_MAX_NOTES_LENGTH).optional(),
  })
  .strict();

export const tripListQuerySchema = z.object({
  date: z
    .string()
    .regex(/^\d{4}-\d{2}-\d{2}$/, 'Expected YYYY-MM-DD')
    .optional(),
});

export const historyQuerySchema = z.object({
  from: z.string().regex(/^\d{4}-\d{2}-\d{2}$/).optional(),
  to: z.string().regex(/^\d{4}-\d{2}-\d{2}$/).optional(),
  limit: z.coerce.number().int().min(1).max(100).default(20),
  cursor: z.string().optional(),
});

export const scanBodySchema = z
  .object({
    qrPayload: z.string().min(1).max(800).optional(),
    scannedAt: z.string().datetime(),
    scannedLat: z.number().min(-90).max(90).optional(),
    scannedLng: z.number().min(-180).max(180).optional(),
    manualEntry: z.boolean().optional(),
    bookingReference: z.string().min(1).max(30).optional(),
    primaryPassengerName: z.string().min(1).max(100).optional(),
  })
  .strict()
  .refine(
    (d) => {
      if (d.manualEntry) return !!d.bookingReference && !!d.primaryPassengerName;
      return !!d.qrPayload;
    },
    { message: 'Provide qrPayload, OR manualEntry + bookingReference + primaryPassengerName' },
  );
export const uploadUrlBodySchema = z
  .object({
    filename: z.string().min(1).max(200),
    contentType: z.enum(['application/pdf', 'image/jpeg', 'image/png']),
    purpose: z.enum(['DRIVER_LICENSE', 'GOVERNMENT_ID', 'PROFILE_PHOTO']),
  })
  .strict();

export const attachDocsBodySchema = z
  .object({
    uploadIds: z.array(z.string().min(1).max(80)).min(1).max(3),
  })
  .strict();
export const cancelTripBodySchema = z
  .object({
    reason: z.string().min(5).max(500),
  })
  .strict();

// ---------- Start pre-checks (buscon/02 §7.2 + §7.3) ----------
// Note: license expiry check skipped — driver_profiles has no licenseExpiresAt column yet.

type StartContext = {
  trip: NonNullable<Awaited<ReturnType<typeof repo.findTripForStart>>>;
  driver: NonNullable<Awaited<ReturnType<typeof repo.findDriverContextForStart>>>;
  userId: string;
  now: Date;
};

async function runStartPreChecks(ctx: StartContext): Promise<void> {
  const { trip, driver, userId, now } = ctx;

  // 1. Driver not blocked/deactivated (checked first so TS narrowing works)
  if (driver.kycStatus === 'BLOCKED' || driver.kycStatus === 'DEACTIVATED') {
    throw tripErrors.driverBlocked();
  }
  // 2. Driver KYC must be VERIFIED or ACTIVE
  if (driver.kycStatus !== 'VERIFIED' && driver.kycStatus !== 'ACTIVE') {
    throw tripErrors.driverNotVerified();
  }
  // 3. (license expiry — skipped, no column)
  // 4. Bus approved
  if (trip.bus.approvalStatus !== 'APPROVED') {
    throw tripErrors.busNotApproved();
  }
  // 5. Bus has layout
  if (!trip.bus.layout) {
    throw tripErrors.busLayoutMissing();
  }
  // 6. Schedule is published (not cancelled)
  if (trip.schedule.status === 'CANCELLED') {
    throw tripErrors.scheduleCancelled();
  }
  if (trip.schedule.status !== 'PUBLISHED') {
    throw tripErrors.notActive();
  }
  // 7. Trip currently SCHEDULED (not already IN_PROGRESS/COMPLETED/CANCELLED)
  if (trip.status !== 'SCHEDULED') {
    if (trip.status === 'IN_PROGRESS') throw tripErrors.alreadyStarted();
    throw tripErrors.notActive();
  }
  // 8. Time window: within ±2h of scheduled departure
  const delta = Math.abs(now.getTime() - trip.schedule.departureTime.getTime());
  if (delta > TRIP_START_WINDOW_MS) {
    throw tripErrors.outsideWindow();
  }
  // 9. No other active trip for this driver
  const active = await repo.findActiveTripForDriver(userId);
  if (active && active.id !== trip.id) {
    throw tripErrors.activeTripExists(active.id);
  }
}
// ---------- Start trip (WF-D-02) ----------
// A5: SELECT ... FOR UPDATE on driver_profiles to prevent concurrent start race.
// Session metadata active_trip_id skipped — sessions has no metadata_json column.
// Double-start prevention comes from runStartPreChecks + FOR UPDATE, which is sufficient.

export async function startTrip(
  userId: string,
  tripId: string,
  actorIp?: string | undefined,
): Promise<StartTripResultDTO> {
  const trip = await repo.findTripForStart(tripId);
  if (!trip) throw tripErrors.notFound();

  const driver = await repo.findDriverContextForStart(userId);
  if (!driver) throw tripErrors.notAssigned();
  if (trip.driver.userId !== userId) throw tripErrors.notAssigned();

  const now = new Date();
  await runStartPreChecks({ trip, driver, userId, now });

  const startedAt = await prisma.$transaction(async (tx) => {
    // A5: row-lock driver_profiles so concurrent /start calls serialize
    const locked = await tx.$queryRaw<Array<{ id: string }>>`
      SELECT id FROM driver_profiles
      WHERE id = ${trip.driver.id}::uuid AND deleted_at IS NULL
      FOR UPDATE
    `;
    if (locked.length === 0) throw tripErrors.notAssigned();

    const updated = await tx.trip.updateMany({
      where: { id: tripId, status: 'SCHEDULED' },
      data: { status: 'IN_PROGRESS', startedAt: now },
    });
    if (updated.count === 0) throw tripErrors.alreadyStarted();

    await tx.driverProfile.update({
      where: { id: trip.driver.id },
      data: { onDuty: true },
    });

    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_started',
      ipAddress: actorIp ?? null,
      metadata: { scheduleId: trip.schedule.id, busId: trip.bus.id },
    });

    return now;
  });

  emitToTopic(`trip:live:${tripId}`, events.tripStarted({
    tripId,
    startedAt: startedAt.toISOString(),
  }));

  return {
    tripId,
    status: 'IN_PROGRESS',
    startedAt: startedAt.toISOString(),
  };
}
// ---------- End trip (WF-D-06) ----------
// Distance computed from all Postgres gps_samples (buscon/04 §18).
// Atomic: trips COMPLETED + driver.onDuty=false + audit_logs.

export async function endTrip(
  userId: string,
  tripId: string,
  actorIp?: string | undefined,
): Promise<EndTripResultDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (trip.status !== 'IN_PROGRESS') throw tripErrors.notInProgress();

  const gpsRows = await prisma.gpsSample.findMany({
    where: { tripId },
    select: { lat: true, lng: true, recordedAt: true },
    orderBy: { recordedAt: 'asc' },
  });
  const { computeTripDistanceKm } = await import('../../lib/eta.js');
  const samples = gpsRows.map((r) => ({
    lat: Number(r.lat),
    lng: Number(r.lng),
    recordedAt: r.recordedAt,
  }));
  const distanceKm = computeTripDistanceKm(samples);

  const counts = await repo
    .findTripsForDriver(userId, undefined)
    .then((all) => all.find((x) => x.id === tripId)?.counts ?? { boarded: 0, confirmed: 0 });

  const now = new Date();
  const startedAtMs = trip.startedAt?.getTime() ?? now.getTime();
  const durationMinutes = Math.round((now.getTime() - startedAtMs) / 60_000);

  await prisma.$transaction(async (tx) => {
    const updated = await tx.trip.updateMany({
      where: { id: tripId, status: 'IN_PROGRESS' },
      data: { status: 'COMPLETED', endedAt: now, distanceKm },
    });
    if (updated.count === 0) throw tripErrors.notInProgress();

    await tx.driverProfile.update({
      where: { id: trip.driverId },
      data: { onDuty: false },
    });

    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_ended',
      ipAddress: actorIp ?? null,
      metadata: { distanceKm, durationMinutes, boardedCount: counts.boarded },
    });
  });

  emitToTopic(`trip:live:${tripId}`, events.tripEnded({
    tripId,
    endedAt: now.toISOString(),
    distanceKm: Math.round(distanceKm * 100) / 100,
  }));
  clearThrottleState(tripId);

  return {
    tripId,
    status: 'COMPLETED',
    startedAt: trip.startedAt?.toISOString() ?? null,
    endedAt: now.toISOString(),
    durationMinutes,
    distanceKm: Math.round(distanceKm * 100) / 100,
    boardedCount: counts.boarded,
    confirmedCount: counts.confirmed,
  };
}

export async function getManifest(userId: string, tripId: string): Promise<ManifestDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();

  const rows = await repo.findManifestRowsForSchedule(trip.scheduleId);
  const stands = await repo.findBoardingStandsForSchedule(trip.scheduleId);

  type Agg = {
    bookingId: string;
    bookingReference: string;
    seatLabels: string[];
    passengerNames: string[];
    boardingStatus: 'CONFIRMED' | 'BOARDED';
    boardedAt: string | null;
    boardingStandId: string;
  };

  const bookingsById = new Map<string, Agg>();
  for (const row of rows) {
    let agg = bookingsById.get(row.bookingId);
    if (!agg) {
      agg = {
        bookingId: row.bookingId,
        bookingReference: row.booking.bookingReference,
        seatLabels: [],
        passengerNames: [],
        boardingStatus: row.booking.status === 'CHECKED_IN' ? 'BOARDED' : 'CONFIRMED',
        boardedAt: row.booking.checkedInAt?.toISOString() ?? null,
        boardingStandId: row.booking.boardingStand.id,
      };
      bookingsById.set(row.bookingId, agg);
    }
    agg.seatLabels.push(row.seat.seatLabel);
    agg.passengerNames.push(row.name);
  }

  const byStandMap = new Map<string, ManifestStandGroupDTO>();
  for (const s of stands) {
    byStandMap.set(s.stand.id, {
      standId: s.stand.id,
      standName: s.stand.name,
      sequence: s.sequence,
      passengers: [],
    });
  }

  let totalPassengers = 0;
  let totalBoarded = 0;
  for (const agg of bookingsById.values()) {
    agg.seatLabels.sort();
    agg.passengerNames.sort();
    totalPassengers += agg.passengerNames.length;
    if (agg.boardingStatus === 'BOARDED') totalBoarded += agg.passengerNames.length;

    const passenger: ManifestPassengerDTO = {
      bookingId: agg.bookingId,
      bookingReference: agg.bookingReference,
      seatLabels: agg.seatLabels,
      passengerNames: agg.passengerNames,
      boardingStatus: agg.boardingStatus,
      boardedAt: agg.boardedAt,
    };

    const group = byStandMap.get(agg.boardingStandId);
    if (group) group.passengers.push(passenger);
    else
      byStandMap.set(agg.boardingStandId, {
        standId: agg.boardingStandId,
        standName: 'Unknown Stand',
        sequence: 999,
        passengers: [passenger],
      });
  }

  return {
    tripId,
    scheduleId: trip.scheduleId,
    totalPassengers,
    totalBoarded,
    lastUpdatedAt: new Date().toISOString(),
    byStand: Array.from(byStandMap.values()).sort((a, b) => a.sequence - b.sequence),
  };
}

export async function listHistory(
  userId: string,
  opts: { from?: string | undefined; to?: string | undefined; limit: number; cursor?: string | undefined },
): Promise<{ data: DriverTripSummaryDTO[]; nextCursor: string | null }> {
  let cursorCreatedAt: Date | undefined;
  if (opts.cursor) {
    try {
      const decoded = JSON.parse(Buffer.from(opts.cursor, 'base64').toString('utf8')) as { createdAt: string };
      cursorCreatedAt = new Date(decoded.createdAt);
    } catch {
      throw tripErrors.notFound();
    }
  }

  const rows = await repo.findHistoryForDriver(userId, {
    ...(opts.from !== undefined ? { from: opts.from } : {}),
    ...(opts.to !== undefined ? { to: opts.to } : {}),
    limit: opts.limit,
    ...(cursorCreatedAt !== undefined ? { cursorCreatedAt } : {}),
  });

  const hasMore = rows.length > opts.limit;
  const data = hasMore ? rows.slice(0, opts.limit) : rows;
  const nextCursor = hasMore
    ? Buffer.from(JSON.stringify({ createdAt: data[data.length - 1]!.createdAt.toISOString() })).toString('base64')
    : null;

  const scheduleIds = data.map((t) => t.schedule.id);
  type CountRow = { scheduleId: string; status: string; cnt: number };
  const countRows =
    scheduleIds.length === 0
      ? []
      : await prisma.$queryRaw<CountRow[]>`
          SELECT bp.schedule_id AS "scheduleId", b.status, COUNT(*)::int AS cnt
          FROM booking_passengers bp
          JOIN bookings b ON b.id = bp.booking_id
          WHERE bp.schedule_id = ANY(${scheduleIds}::uuid[])
            AND bp.deleted_at IS NULL
            AND b.deleted_at IS NULL
            AND b.status IN ('CONFIRMED', 'CHECKED_IN')
          GROUP BY bp.schedule_id, b.status
        `;
  const countsMap = new Map<string, { boarded: number; confirmed: number }>();
  for (const id of scheduleIds) countsMap.set(id, { boarded: 0, confirmed: 0 });
  for (const r of countRows) {
    const cur = countsMap.get(r.scheduleId) ?? { boarded: 0, confirmed: 0 };
    if (r.status === 'CHECKED_IN') cur.boarded += r.cnt;
    cur.confirmed += r.cnt;
    countsMap.set(r.scheduleId, cur);
  }

  return {
    data: data.map((t) => {
      const counts = countsMap.get(t.schedule.id) ?? { boarded: 0, confirmed: 0 };
      return {
        tripId: t.id,
        scheduleId: t.schedule.id,
        bus: t.bus,
        route: {
          originCity: t.schedule.route.originCity.name,
          destinationCity: t.schedule.route.destCity.name,
        },
        departureTime: t.schedule.departureTime.toISOString(),
        arrivalTime: t.schedule.arrivalTime.toISOString(),
        status: t.status as TripStatus,
        boardedCount: counts.boarded,
        confirmedCount: counts.confirmed,
      };
    }),
    nextCursor,
  };
}

// ---------- Cancel trip (Phase 12 — cascade via cancelBooking) ----------

export async function cancelTrip(
  userId: string,
  tripId: string,
  body: { reason: string },
  actorIp?: string | undefined,
): Promise<import('./trip.types.js').CancelTripResultDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();

  if (trip.status === 'CANCELLED') {
    throw errors.invalidTransition('Trip is already cancelled');
  }
  if (trip.status === 'COMPLETED') {
    throw errors.invalidTransition('Cannot cancel a completed trip');
  }

  const bookings = await prisma.booking.findMany({
    where: {
      scheduleId: trip.scheduleId,
      deletedAt: null,
      status: { in: ['CONFIRMED', 'OPERATOR_CANCELLED', 'PENDING'] },
    },
    select: { id: true, userId: true, status: true },
  });

  const { cancelBooking, expirePendingBooking } = await import('../booking/booking.service.js');

  let cancelledBookings = 0;
  let refundedPaise = 0;

  for (const b of bookings) {
    try {
      if (b.status === 'CONFIRMED') {
        await prisma.booking.update({
          where: { id: b.id },
          data: { status: 'OPERATOR_CANCELLED' },
        });
        const result = await cancelBooking(b.userId, b.id);
        cancelledBookings++;
        refundedPaise += result.refund.amountPaise;
      } else if (b.status === 'OPERATOR_CANCELLED') {
        const result = await cancelBooking(b.userId, b.id);
        cancelledBookings++;
        refundedPaise += result.refund.amountPaise;
      } else if (b.status === 'PENDING') {
        await expirePendingBooking(b.id);
        cancelledBookings++;
      }
    } catch (err) {
      logger.warn({ err, bookingId: b.id, tripId }, 'cancelTrip: cascade failed for booking');
    }
  }

  const now = new Date();
  await prisma.$transaction(async (tx) => {
    await tx.trip.update({
      where: { id: tripId },
      data: { status: 'CANCELLED', cancelledAt: now },
    });
    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_cancelled',
      ipAddress: actorIp ?? null,
      metadata: {
        scheduleId: trip.scheduleId,
        reason: body.reason,
        cancelledBookings,
        refundedPaise,
      },
    });
  });

  emitToTopic('trip:live:' + tripId, events.tripCancelled({
    tripId,
    reason: body.reason,
  }));
  clearThrottleState(tripId);

  return {
    tripId,
    status: 'CANCELLED',
    cancelledBookings,
    refundedPaise,
  };
}

function istDayBounds(dateStr: string): { from: Date; to: Date } {
  return {
    from: new Date(`${dateStr}T00:00:00.000+05:30`),
    to: new Date(`${dateStr}T23:59:59.999+05:30`),
  };
}

function todayIst(): string {

export interface PassengerCounts { boarded: number; confirmed: number; }

async function getPassengerCountsForSchedules(
  scheduleIds: string[],
): Promise<Map<string, PassengerCounts>> {
  if (scheduleIds.length === 0) return new Map();
  const rows = await prisma.$queryRaw<Array<{ scheduleId: string; status: string; cnt: number }>>`
    SELECT bp.schedule_id AS "scheduleId", b.status, COUNT(*)::int AS cnt
    FROM booking_passengers bp
    JOIN bookings b ON b.id = bp.booking_id
    WHERE bp.schedule_id = ANY(${scheduleIds}::uuid[])
      AND bp.deleted_at IS NULL
      AND b.deleted_at IS NULL
      AND b.status IN ('CONFIRMED', 'CHECKED_IN')
    GROUP BY bp.schedule_id, b.status
  `;
  const out = new Map<string, PassengerCounts>();
  for (const id of scheduleIds) out.set(id, { boarded: 0, confirmed: 0 });
  for (const r of rows) {
    const cur = out.get(r.scheduleId) ?? { boarded: 0, confirmed: 0 };
    if (r.status === 'CHECKED_IN') cur.boarded += r.cnt;
    cur.confirmed += r.cnt;
    out.set(r.scheduleId, cur);
  }
  return out;
}

const GPS_LIST_MAX = 720;
const GPS_TTL_SEC = 6 * 3600;
const GPS_PG_EVERY_N = 6;

const GPS_FAIL_KEY_PREFIX = 'gps:fail:';
const GPS_FAIL_THRESHOLD = 3;

async function recordGpsFailure(
  tripId: string,
  driverId: string,
  failureType: events.GpsFailureType,
): Promise<void> {
  try {
    const key = GPS_FAIL_KEY_PREFIX + tripId;
    const count = await redis.incr(key);
    await redis.expire(key, GPS_TTL_SEC);
    if (count >= GPS_FAIL_THRESHOLD && count % GPS_FAIL_THRESHOLD === 0) {
      emitToTopic('trip:live:' + tripId, events.gpsAnomaly({
        tripId,
        driverId,
        failureCount: count,
        lastFailureType: failureType,
        detectedAt: new Date().toISOString(),
      }));
    }
  } catch {
    /* Redis down: skip emit, no throw */
  }
}

export interface GpsIngestResult {
  accepted: boolean;
  serverSpeedKmh: number | null;
}

async function readPreviousSample(tripId: string): Promise<NormalizedSample | null> {
  try {
    const raw = await redis.lindex(keys.gpsTrip(tripId), 0);
    if (raw) return JSON.parse(raw) as NormalizedSample;
  } catch { /* fall through to Postgres */ }

  const lastPg = await prisma.gpsSample.findFirst({
    where: { tripId },
    orderBy: { recordedAt: 'desc' },
  });
  if (!lastPg) return null;
  return {
    lat: Number(lastPg.lat),
    lng: Number(lastPg.lng),
    serverSpeedKmh: lastPg.speedKmh !== null ? Number(lastPg.speedKmh) : null,
    clientSpeedKmh: lastPg.clientReportedSpeedKmh !== null ? Number(lastPg.clientReportedSpeedKmh) : 0,
    accuracyM: lastPg.accuracyM !== null ? Number(lastPg.accuracyM) : 0,
    recordedAt: lastPg.recordedAt.toISOString(),
    receivedAt: lastPg.receivedAt.toISOString(),
    timestampDrift: false,
  };
}
export async function ingestGpsSample(
  userId: string,
  tripId: string,
  body: RawGpsSample,
  deviceHashes: { sessionDeviceHash: string | null; activeSessionDeviceHash: string | null },
): Promise<GpsIngestResult> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (trip.status !== 'IN_PROGRESS') throw tripErrors.notInProgress();

  const previous = await readPreviousSample(tripId);
  const receivedAt = new Date();

  let outcome;
  try {
    outcome = runProvenance({
      current: body,
      previous,
      receivedAt,
      sessionDeviceHash: deviceHashes.sessionDeviceHash,
      activeSessionDeviceHash: deviceHashes.activeSessionDeviceHash,
    });
  } catch (err) {
    const errCode = (err as { code?: string }).code;
    let failType: events.GpsFailureType | null = null;
    if (errCode === 'TELEPORT_DETECTED') failType = 'TELEPORT_DETECTED';
    else if (errCode === 'INVALID_COORDINATES' || errCode === 'INVALID_ACCURACY') failType = 'COORDINATE_INVALID';
    if (failType && trip.driver.id) void recordGpsFailure(tripId, trip.driver.id, failType).catch(() => {});
    throw err;
  }

  if (!outcome.accepted || !outcome.sample) {
    if (outcome.reason === 'speed_mismatch' && trip.driver.id) {
      void recordGpsFailure(tripId, trip.driver.id, 'SPEED_MISMATCH').catch(() => {});
    }
    return { accepted: false, serverSpeedKmh: null };
  }

  const redisKey = keys.gpsTrip(tripId);
  try {
    await redis.lpush(redisKey, JSON.stringify(outcome.sample));
    await redis.ltrim(redisKey, 0, GPS_LIST_MAX - 1);
    await redis.expire(redisKey, GPS_TTL_SEC);
  } catch { /* non-fatal: Redis down, Postgres below still persists */ }

  let counter = 0;
  try {
    const counterKey = `gps:count:${tripId}`;
    counter = await redis.incr(counterKey);
    await redis.expire(counterKey, GPS_TTL_SEC);
  } catch { counter = GPS_PG_EVERY_N; /* force write when Redis down */ }

  if (counter % GPS_PG_EVERY_N === 0) {
    await prisma.gpsSample.create({
      data: {
        tripId,
        lat: outcome.sample.lat,
        lng: outcome.sample.lng,
        speedKmh: outcome.sample.serverSpeedKmh,
        accuracyM: outcome.sample.accuracyM,
        clientReportedSpeedKmh: outcome.sample.clientSpeedKmh,
        receivedAt: new Date(outcome.sample.receivedAt),
        recordedAt: new Date(outcome.sample.recordedAt),
      },
    });
  }

  emitBusPositionThrottled(tripId, events.busPosition({
    tripId,
    scheduleId: trip.scheduleId,
    lat: outcome.sample.lat,
    lng: outcome.sample.lng,
    speedKmh: outcome.sample.serverSpeedKmh ?? 0,
    bearing: null,
    accuracyM: outcome.sample.accuracyM,
    recordedAt: outcome.sample.recordedAt,
    receivedAt: outcome.sample.receivedAt,
    nextStandId: null,
    etaMinutes: null,
  }));

  return { accepted: true, serverSpeedKmh: outcome.sample.serverSpeedKmh };
}

function normalizeName(s: string): string {
  return s.trim().toLowerCase().replace(/\s+/g, ' ');
}

function isTripScannable(status: string): boolean {
  return status === 'SCHEDULED' || status === 'IN_PROGRESS';
}

interface ResolvedTicket {
  ticketId: string;
  bookingId: string;
  boardingStandId: string;
  primaryName: string;
  seatLabels: string[];
  passengerCount: number;
  bookingStatus: string;
  ticketStatus: string;
  scheduleId: string;
}
async function resolveFromQr(qrPayload: string, tripScheduleId: string): Promise<ResolvedTicket> {
  let parsed: ReturnType<typeof parseWireFormat>;
  try {
    parsed = parseWireFormat(qrPayload);
  } catch {
    throw tripErrors.invalidSignature();
  }

  const sigOk = await verify(parsed.signature, parsed.canonicalBytes);
  if (!sigOk) throw tripErrors.invalidSignature();

  const payload = parsed.payload;
  if (payload.scheduleId !== tripScheduleId) {
    throw tripErrors.invalidTrip('Schedule mismatch');
  }

  const ticket = await repo.findTicketForScan(payload.ticketId);
  if (!ticket) throw tripErrors.invalidBooking('Ticket not found');
  // SCANNED allowed through so DUPLICATE path runs; misuse on another trip
  // is caught by the post-duplicate guard in scanTicket.
  if (ticket.status !== 'ACTIVE' && ticket.status !== 'ISSUED' && ticket.status !== 'SCANNED') {
    throw tripErrors.invalidBooking('Ticket not active');
  }
  if (ticket.booking.scheduleId !== tripScheduleId) {
    throw tripErrors.invalidTrip('Ticket is for a different schedule');
  }
  if (ticket.booking.status !== 'CONFIRMED' && ticket.booking.status !== 'CHECKED_IN') {
    throw tripErrors.invalidBooking('Booking is cancelled or not confirmed');
  }

  const passengers = ticket.booking.passengers;
  if (passengers.length === 0) throw tripErrors.invalidBooking('No passengers on booking');

  return {
    ticketId: ticket.id,
    bookingId: ticket.bookingId,
    boardingStandId: ticket.booking.boardingStandId,
    primaryName: passengers[0]!.name,
    seatLabels: passengers.map((p) => p.seat.seatLabel),
    passengerCount: passengers.length,
    bookingStatus: ticket.booking.status,
    ticketStatus: ticket.status,
    scheduleId: ticket.booking.scheduleId,
  };
}
async function resolveFromManual(
  bookingReference: string,
  primaryPassengerName: string,
  tripScheduleId: string,
): Promise<ResolvedTicket> {
  const booking = await repo.findBookingByReferenceAndSchedule(bookingReference, tripScheduleId);
  if (!booking) throw tripErrors.invalidBooking('Booking not found for this trip');
  if (!booking.ticket) throw tripErrors.invalidBooking('No ticket issued for this booking');
  if (booking.status !== 'CONFIRMED' && booking.status !== 'CHECKED_IN') {
    throw tripErrors.invalidBooking('Booking is cancelled or not confirmed');
  }
  const passengers = booking.passengers;
  if (passengers.length === 0) throw tripErrors.invalidBooking('No passengers on booking');
  if (normalizeName(passengers[0]!.name) !== normalizeName(primaryPassengerName)) {
    throw tripErrors.invalidBooking('Passenger name does not match');
  }
  return {
    ticketId: booking.ticket.id,
    bookingId: booking.id,
    boardingStandId: booking.boardingStandId,
    primaryName: passengers[0]!.name,
    seatLabels: passengers.map((p) => p.seat.seatLabel),
    passengerCount: passengers.length,
    bookingStatus: booking.status,
    ticketStatus: booking.ticket.status,
    scheduleId: booking.scheduleId,
  };
}

export async function scanTicket(
  userId: string,
  tripId: string,
  body: ScanRequestDTO,
  actorIp?: string | undefined,
): Promise<ScanResultDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (!isTripScannable(trip.status)) {
    if (trip.status === 'CANCELLED') throw tripErrors.invalidTrip('Trip cancelled');
    throw tripErrors.notActive();
  }

  const resolved = body.manualEntry
    ? await resolveFromManual(body.bookingReference ?? '', body.primaryPassengerName ?? '', trip.scheduleId)
    : await resolveFromQr(body.qrPayload ?? '', trip.scheduleId);

  const existing = await repo.findExistingScanForTicket(resolved.ticketId, tripId);
  if (!existing && resolved.ticketStatus === 'SCANNED') {
    throw tripErrors.invalidBooking('Ticket already used on another trip');
  }
  if (existing) {
    return {
      result: 'DUPLICATE',
      ticketId: resolved.ticketId,
      bookingId: resolved.bookingId,
      primaryPassengerName: resolved.primaryName,
      seatLabels: resolved.seatLabels,
      passengerCount: resolved.passengerCount,
      bookingStatus: 'CHECKED_IN',
      scannedAt: existing.scannedAt.toISOString(),
      originalScannedAt: existing.scannedAt.toISOString(),
    };
  }

  const now = new Date();
  await prisma.$transaction(async (tx) => {
    await tx.boardingScan.create({
      data: {
        tripId,
        ticketId: resolved.ticketId,
        driverId: trip.driverId,
        standId: resolved.boardingStandId,
        result: 'VALID',
        scannedAt: now,
        scannedLat: body.scannedLat ?? null,
        scannedLng: body.scannedLng ?? null,
      },
    });
    await tx.ticket.update({
      where: { id: resolved.ticketId },
      data: { status: 'SCANNED', scannedAt: now },
    });
    await tx.booking.update({
      where: { id: resolved.bookingId },
      data: { status: 'CHECKED_IN', checkedInAt: now },
    });
    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Ticket',
      entityId: resolved.ticketId,
      action: 'ticket_scanned',
      ipAddress: actorIp ?? null,
      metadata: {
        tripId,
        bookingId: resolved.bookingId,
        passengerCount: resolved.passengerCount,
        manualEntry: body.manualEntry === true,
      },
    });
  });

  emitToBooking(resolved.bookingId, events.boardingVerified({
    bookingId: resolved.bookingId,
    ticketId: resolved.ticketId,
    passengerCount: resolved.passengerCount,
    allSeatLabels: resolved.seatLabels,
    boardedAt: now.toISOString(),
  }));

  return {
    result: 'VALID',
    ticketId: resolved.ticketId,
    bookingId: resolved.bookingId,
    primaryPassengerName: resolved.primaryName,
    seatLabels: resolved.seatLabels,
    passengerCount: resolved.passengerCount,
    bookingStatus: 'CHECKED_IN',
    scannedAt: now.toISOString(),
  };
}

/**
 * Plugin-scoped error handler. Guarantees DomainErrors map to the right
 * HTTP status for identity routes, regardless of ordering issues in the
 * parent app's `setErrorHandler` registration.
 */
function isDomainError(
  err: unknown,
): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return (
    typeof e.code === 'string' &&
    typeof e.httpStatus === 'number' &&
    typeof e.message === 'string'
  );
}

function isZodError(err: unknown): err is { issues: unknown[]; message: string } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return e.name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function identityRouter(server: FastifyInstance): Promise<void> {
  // Scoped error handler — takes precedence for routes in this plugin
  server.setErrorHandler((error, request, reply) => {
    if (process.env.NODE_ENV === 'test') {
      console.log('[err-handler-ROUTER]', {
        url: request.url,
        errName: (error as { name?: string })?.name,
        errCtor: (error as { constructor?: { name?: string } })?.constructor?.name,
        code: (error as { code?: unknown })?.code,
        httpStatus: (error as { httpStatus?: unknown })?.httpStatus,
        hasIssues: Array.isArray((error as { issues?: unknown })?.issues),
        isDomain: isDomainError(error),
        isZod: isZodError(error),
      });
    }

    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({
        error: {
          code: error.code,
          message: error.message,
          ...(error.details ? { details: error.details } : {}),
          requestId: request.id,
        },
      });
    }

    if (isZodError(error)) {
      return reply.status(422).send({
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Validation failed',
          details: { issues: error.issues },
          requestId: request.id,
        },
      });
    }

    const anyErr = error as { statusCode?: number; code?: string; message?: string };
    const statusCode = anyErr.statusCode ?? 500;
    const isProd = process.env.NODE_ENV === 'production';
    return reply.status(statusCode).send({
      error: {
        code: anyErr.code ?? 'INTERNAL_ERROR',
        message: isProd ? 'Something went wrong' : (anyErr.message ?? 'Unknown error'),
        requestId: request.id,
      },
    });
  });

  // ── Routes ──
  server.post('/passenger/auth/otp/request', c.postPassengerOtpRequest);
  server.post('/passenger/auth/otp/verify', c.postPassengerOtpVerify);
  server.get('/passenger/me', { preHandler: requireRole('REGISTERED_PASSENGER') }, getPassengerMe);

  server.post('/driver/auth/otp/request', c.postDriverOtpRequest);
  server.post('/driver/auth/otp/verify', c.postDriverOtpVerify);

  server.post('/operator/auth/signup', c.postOperatorSignup);
  server.post('/operator/auth/login', c.postOperatorLogin);
  server.post('/operator/auth/otp/request', c.postOperatorOtpRequest);
  server.post('/operator/auth/otp/verify', c.postOperatorOtpVerify);
  server.post('/operator/auth/reset/request', c.postOperatorResetRequest);
  server.post('/operator/auth/reset/verify', c.postOperatorResetVerify);

  server.post('/admin/auth/login', c.postAdminLogin);
  server.post('/admin/auth/reset/request', c.postAdminResetRequest);
  server.post('/admin/auth/reset/verify', c.postAdminResetVerify);

  server.post('/:surface/auth/refresh', c.postRefresh);
  server.post('/:surface/auth/logout', c.postLogout);
}

export async function passengerOtpRequest(params: {
  phone?: string | undefined;
  email?: string | undefined;
  intent: 'LOGIN' | 'SIGNUP';
  ip?: string | undefined;
}): Promise<{ expiresInSeconds: number }> {
  const { identifier, channel } = resolveOtpIdentifier(params);

  if (params.intent === 'SIGNUP' && channel === 'EMAIL') {
    throw errors.emailSignupNotSupported();
  }

  if (params.intent === 'SIGNUP' && channel === 'PHONE') {
    const existing = await findUserByPhone(identifier);
    if (existing) throw errors.phoneAlreadyRegistered();
  }

  const result = await createOtpChallenge({ identifier, channel, purpose: 'LOGIN' });
  return { expiresInSeconds: result.expiresInSeconds };
}

export async function passengerOtpVerify(params: {
  phone?: string | undefined;
  email?: string | undefined;
  otp: string;
  ip?: string | undefined;
  userAgent?: string | undefined;
}): Promise<LoginResult> {
  const { identifier, channel } = resolveOtpIdentifier(params);

  await consumeOtpChallenge({ identifier, otp: params.otp, purpose: 'LOGIN' });

  let user =
    channel === 'PHONE' ? await findUserByPhone(identifier) : await findUserByEmail(identifier);

  if (!user) {
    if (channel === 'EMAIL') throw errors.invalidOtp();
    user = await createPassengerUser({ phone: identifier, ip: params.ip });
    emitIdentityEvent({ type: 'user_registered', userId: user.id });
  }

  const sessionId = newUuid();
  const tokens = await issueSessionTokens({
    userId: user.id,
    role: 'REGISTERED_PASSENGER',
    sessionId,
  });

  emitIdentityEvent({
    type: 'session_created',
    userId: user.id,
    sessionId,
    ip: params.ip,
    userAgent: params.userAgent,
    metadata: { role: 'REGISTERED_PASSENGER' },
  });

  return { ...tokens, user: toUserDTO(user) };
}

    sessionId,
    ip: params.ip,

export async function operatorSignup(params: {
  companyName: string;
  gstin: string;
  pan: string;
  email: string;
  phone: string;
  password: string;
  ip?: string | undefined;
}): Promise<OperatorSignupResult> {

  if (existingEmail) throw errors.emailAlreadyRegistered();

  const existingPhone = await findUserByPhone(params.phone);
  if (existingPhone) throw errors.phoneAlreadyRegistered();

  const existingGstin = await findOperatorByGstin(params.gstin);
  if (existingGstin) throw errors.gstinAlreadyRegistered();

  // 2. Hash password (Argon2id)
  const passwordHash = await hashPassword(params.password);

  // 3. Create User + Operator + OperatorUser + consent records in one tx
  const { user, operator } = await createOperatorSignup({
    companyName: params.companyName,
    gstin: params.gstin,
    pan: params.pan,
    email: params.email,
    phone: params.phone,
    passwordHash,
    ip: params.ip,
  });

  // 4. Audit log (best-effort, after tx commits)
  try {
    await writeAuditEvent({
      actorId: null,
      sessionId: null,
      entityType: 'Operator',
      entityId: operator.id,
      action: 'operator_signup',
      ipAddress: params.ip ?? null,
      metadata: {
        companyName: params.companyName,
        userId: user.id,
      },
    });
  } catch {
    /* non-fatal: entity creation already succeeded */
  }

  // 5. Identity event (structured audit log)
  emitIdentityEvent({
    type: 'operator_signup_submitted',
    userId: user.id,
    ip: params.ip,
    metadata: { role: 'OPERATOR_OWNER', operatorId: operator.id },
  });

  emitToTopic('admin:notifications', events.operatorApplied({
    operatorId: operator.id,
    companyName: params.companyName,
    submittedAt: new Date().toISOString(),
  }));

    operatorId: operator.id,
    userId: user.id,
    status: 'APPLIED',

  };

import crypto from 'node:crypto';
import jwt from 'jsonwebtoken';
import { env } from '../../config/env.js';
import { prisma } from '../../lib/prisma.js';
import { sha256 } from '../../lib/crypto.js';
import { errors } from '../../lib/errors.js';
import {
  ACCESS_TTL_SECONDS,
  REFRESH_TTL_SECONDS,
  type Role,
} from './identity.types.js';

export interface AccessTokenClaims {
  sub: string; // userId
  sid: string; // sessionId
  role: Role;
  iat: number;
  exp: number;
}

export interface IssuedTokens {
  accessToken: string;
  refreshToken: string;
  expiresAt: string;
}

/**
 * Issue access (JWT) + refresh (opaque) tokens for a new session.
 * Creates the session and refresh-token chain in one transaction.
 */
export async function issueSessionTokens(params: {
  userId: string;
  role: Role;
  sessionId: string;
  deviceFingerprintHash?: string;
  activeTripId?: string;
}): Promise<IssuedTokens> {
  const { userId, role, sessionId, deviceFingerprintHash } = params;

  const accessTtl = ACCESS_TTL_SECONDS[role];
  const refreshTtl = REFRESH_TTL_SECONDS[role];

  const accessToken = jwt.sign(
    { sub: userId, sid: sessionId, role, jti: crypto.randomUUID() },
    env.JWT_SECRET,
    { algorithm: 'HS256', expiresIn: accessTtl },
  );

  const refreshRaw = crypto.randomBytes(32).toString('base64');
  const refreshHash = sha256(refreshRaw);

  const expiresAt = new Date(Date.now() + refreshTtl * 1000);

  await prisma.$transaction([
    prisma.session.create({
      data: {
        id: sessionId,
        userId,
        role,
        refreshHash,
        deviceFingerprintHash: deviceFingerprintHash ?? null,
        expiresAt,
        status: 'ACTIVE',
      },
    }),
    prisma.refreshTokenChain.create({
      data: { sessionId, usedHashes: [] },
    }),
  ]);

  return {
    accessToken,
    refreshToken: refreshRaw,
    expiresAt: expiresAt.toISOString(),
  };
}

/**
 * Verify an access token. Returns claims or throws UNAUTHORIZED.
 * Callers should also check session status (via `assertSessionActive`).
 */
export function verifyAccessToken(token: string): AccessTokenClaims {
  try {
    const decoded = jwt.verify(token, env.JWT_SECRET, { algorithms: ['HS256'] });
    return decoded as AccessTokenClaims;
  } catch {
    throw errors.unauthorized('Invalid or expired token');
  }
}

/**
 * Rotate refresh token. Detects replay and revokes the entire chain if
 * an already-used token is presented.
 */
export async function rotateRefreshToken(
  oldRefreshRaw: string,
): Promise<IssuedTokens> {
  const oldHash = sha256(oldRefreshRaw);

  // Case 1: current hash matches — normal rotation
  const currentSession = await prisma.session.findFirst({
    where: { refreshHash: oldHash, status: 'ACTIVE' },
  });

  if (currentSession) {
    const role = currentSession.role as Role;
    const refreshTtl = REFRESH_TTL_SECONDS[role];

    const newRefreshRaw = crypto.randomBytes(32).toString('base64');
    const newRefreshHash = sha256(newRefreshRaw);

    const expiresAt = new Date(Date.now() + refreshTtl * 1000);

    await prisma.$transaction([
      prisma.session.update({
        where: { id: currentSession.id },
        data: { refreshHash: newRefreshHash, expiresAt },
      }),
      prisma.$executeRaw`
        UPDATE refresh_token_chains
        SET used_hashes = used_hashes || ${JSON.stringify([oldHash])}::jsonb,
            updated_at = now(),
            version = version + 1
        WHERE session_id = ${currentSession.id}::uuid
      `,
    ]);

    const newAccessToken = jwt.sign(
      { sub: currentSession.userId, sid: currentSession.id, role, jti: crypto.randomUUID() },
      env.JWT_SECRET,
      { algorithm: 'HS256', expiresIn: ACCESS_TTL_SECONDS[role] },
    );

    return {
      accessToken: newAccessToken,
      refreshToken: newRefreshRaw,
      expiresAt: expiresAt.toISOString(),
    };
  }

  // Case 2: hash is in used_hashes for an active session — REPLAY detected
  const replayRows = await prisma.$queryRaw<Array<{ id: string }>>`
    SELECT s.id
    FROM sessions s
    JOIN refresh_token_chains c ON c.session_id = s.id
    WHERE s.status = 'ACTIVE'
      AND c.used_hashes @> ${JSON.stringify([oldHash])}::jsonb
    LIMIT 1
  `;

  if (replayRows.length > 0 && replayRows[0]) {
    const replayedSessionId = replayRows[0].id;
    await prisma.session.update({
      where: { id: replayedSessionId },
      data: {
        status: 'REVOKED',
        revokedAt: new Date(),
        revokeReason: 'REFRESH_REPLAY',
      },
    });
    throw errors.refreshReplayed();
  }

  // Case 3: unknown token — invalid
  throw errors.unauthorized('Invalid refresh token');
}

import crypto from 'node:crypto';
import { env } from '../../config/env.js';
import { prisma } from '../../lib/prisma.js';
import { redis, keys } from '../../lib/redis.js';
import { sha256, timingSafeEqualString } from '../../lib/crypto.js';
import { errors } from '../../lib/errors.js';
import { smsAdapter } from '../../integrations/msg91.js';
import { emailAdapter } from '../../integrations/sendgrid.js';
import { isValidEmail, normalizeEmail } from '../../lib/email.js';
import { isValidPhone, normalizePhone } from '../../lib/phone.js';
import {
  OTP_MAX_ATTEMPTS,
  OTP_REQUEST_MAX,
  OTP_REQUEST_WINDOW_SECONDS,
  OTP_TTL_SECONDS,
} from './identity.types.js';

export type OtpPurpose = 'LOGIN' | 'RESET';

export interface GeneratedOtp {
  raw: string;
  hash: string;
  salt: string;
}

export function generateOtp(): GeneratedOtp {
  const raw = env.MOCK_OTP ? '123456' : String(crypto.randomInt(100000, 1000000));
  const salt = crypto.randomBytes(16).toString('base64');
  const hash = sha256(raw, salt);
  return { raw, hash, salt };
}

export function verifyOtp(raw: string, hash: string, salt: string): boolean {
  const computed = sha256(raw, salt);
  return timingSafeEqualString(computed, hash);
}

async function enforceOtpRequestRate(identifier: string): Promise<void> {
  const key = keys.rateLimit('otp-request', identifier);
  const count = await redis.incr(key);
  if (count === 1) await redis.expire(key, OTP_REQUEST_WINDOW_SECONDS);
  if (count > OTP_REQUEST_MAX) {
    throw errors.rateLimited(
      `Too many OTP requests. Try again in ${Math.ceil(OTP_REQUEST_WINDOW_SECONDS / 60)} minutes.`,
    );
  }
}

export async function enforceOtpVerifyRate(identifier: string): Promise<void> {
  const key = keys.rateLimit('otp-verify', identifier);
  const count = await redis.incr(key);
  if (count === 1) await redis.expire(key, OTP_REQUEST_WINDOW_SECONDS);
  if (count > OTP_MAX_ATTEMPTS) {
    throw errors.otpAttemptsExceeded();
  }
}

async function deliverOtp(
  identifier: string,
  channel: 'PHONE' | 'EMAIL',
  rawOtp: string,
  purpose: OtpPurpose,
): Promise<void> {
  const label = purpose === 'RESET' ? 'password reset' : 'login';
  if (channel === 'PHONE') {
    await smsAdapter.send({
      phone: identifier,
      body: `Your BusVerse ${label} OTP is ${rawOtp}. Valid for 10 minutes. Do not share.`,
      template: 'otp',
    });
  } else {
    await emailAdapter.send({
      to: identifier,
      subject: `BusVerse ${label} code`,
      body: `Your ${label} code is ${rawOtp}. Valid for 10 minutes. Do not share.`,
      template: 'otp',
    });
  }
}

export async function createOtpChallenge(params: {
  identifier: string;
  channel: 'PHONE' | 'EMAIL';
  purpose: OtpPurpose;
}): Promise<{ expiresInSeconds: number; rawOtpForDev?: string | undefined }> {
  const { identifier, channel, purpose } = params;

  await enforceOtpRequestRate(`${purpose}:${identifier}`);

  const { raw, hash, salt } = generateOtp();
  const expiresAt = new Date(Date.now() + OTP_TTL_SECONDS * 1000);

  await prisma.otpChallenge.create({
    data: {
      phoneOrEmail: identifier,
      otpHash: hash,
      otpSalt: salt,
      expiresAt,
      purpose,
    },
  });

  await deliverOtp(identifier, channel, raw, purpose);

  if (env.MOCK_OTP) {
    return { expiresInSeconds: OTP_TTL_SECONDS, rawOtpForDev: raw };
  }
  return { expiresInSeconds: OTP_TTL_SECONDS };
}

export async function consumeOtpChallenge(params: {
  identifier: string;
  otp: string;
  purpose: OtpPurpose;
}): Promise<{ challengeId: string }> {
  const { identifier, otp, purpose } = params;

  await enforceOtpVerifyRate(`${purpose}:${identifier}`);

  const challenge = await prisma.otpChallenge.findFirst({
    where: {
      phoneOrEmail: identifier,
      purpose,
      consumedAt: null,
      deletedAt: null,
      expiresAt: { gt: new Date() },
    },
    orderBy: { createdAt: 'desc' },
  });

  if (!challenge) throw errors.otpExpired();
  if (challenge.attempts >= OTP_MAX_ATTEMPTS) throw errors.otpAttemptsExceeded();

  if (!verifyOtp(otp, challenge.otpHash, challenge.otpSalt)) {
    await prisma.otpChallenge.update({
      where: { id: challenge.id },
      data: { attempts: { increment: 1 } },
    });
    throw errors.invalidOtp();
  }

  await prisma.otpChallenge.update({
    where: { id: challenge.id },
    data: { consumedAt: new Date() },
  });

  return { challengeId: challenge.id };
}

export function resolveOtpIdentifier(params: {
  phone?: string | undefined;
  email?: string | undefined;
}): { identifier: string; channel: 'PHONE' | 'EMAIL' } {
  if (params.phone && params.email) {
    throw errors.invalidRequest('Provide exactly one of phone or email');
  }
  if (params.phone) {
    const normalized = normalizePhone(params.phone);
    if (!isValidPhone(normalized)) throw errors.invalidPhoneFormat();
    return { identifier: normalized, channel: 'PHONE' };
  }
  if (params.email) {
    const normalized = normalizeEmail(params.email);
    if (!isValidEmail(normalized)) throw errors.invalidEmailFormat();
    return { identifier: normalized, channel: 'EMAIL' };
  }
  throw errors.invalidRequest('Provide either phone or email');
}

// GPS provenance checks (buscon/04 §11, §12). Pure functions — no DB, no IO.

import { errors } from './errors.js';

export interface RawGpsSample {
  lat: number;
  lng: number;
  speedKmh: number;
  accuracyM: number;
  recordedAt: string;
}

export interface NormalizedSample {
  lat: number;
  lng: number;
  serverSpeedKmh: number | null;
  clientSpeedKmh: number;
  accuracyM: number;
  recordedAt: string;
  receivedAt: string;
  timestampDrift: boolean;
}

export type ProvenanceFlag = 'timestamp_drift' | 'speed_mismatch';

export interface ProvenanceInput {
  current: RawGpsSample;
  previous: NormalizedSample | null;
  receivedAt: Date;
  sessionDeviceHash: string | null;
  activeSessionDeviceHash: string | null;
}

export interface ProvenanceOutcome {
  accepted: boolean;
  flags: ProvenanceFlag[];
  sample: NormalizedSample | null;
  reason?: 'speed_mismatch' | 'fingerprint_mismatch';
}

// Limits per buscon/04 §3.4, §11.2, §12.4
export const TIMESTAMP_WINDOW_MS = 5 * 60 * 1000;
export const MAX_ACCURACY_M = 100;
export const MAX_TELEPORT_SPEED_KMH = 200;
export const MAX_SPEED_MISMATCH_KMH = 30;
export const ZERO_SPEED_KMH = 3;
export const ZERO_SPEED_RADIUS_KM = 0.02;
// ---------- Haversine + speed ----------

const EARTH_RADIUS_KM = 6371;
const toRad = (d: number): number => (d * Math.PI) / 180;

export function haversineKm(
  a: { lat: number; lng: number },
  b: { lat: number; lng: number },
): number {
  const dLat = toRad(b.lat - a.lat);
  const dLng = toRad(b.lng - a.lng);
  const lat1 = toRad(a.lat);
  const lat2 = toRad(b.lat);
  const h =
    Math.sin(dLat / 2) ** 2 +
    Math.sin(dLng / 2) ** 2 * Math.cos(lat1) * Math.cos(lat2);
  return 2 * EARTH_RADIUS_KM * Math.asin(Math.sqrt(h));
}

export function computeSpeedKmh(
  prev: { lat: number; lng: number },
  curr: { lat: number; lng: number },
  deltaMs: number,
): number {
  if (deltaMs <= 0) return 0;
  const km = haversineKm(prev, curr);
  return km / (deltaMs / 3_600_000);
}
// ---------- Provenance checks (buscon/04 §11.2) ----------

export function checkTimestampWindow(
  recordedAt: string,
  receivedAt: Date,
): { effectiveRecordedAt: Date; drifted: boolean } {
  const rec = new Date(recordedAt);
  if (Number.isNaN(rec.getTime())) {
    return { effectiveRecordedAt: receivedAt, drifted: true };
  }
  const delta = Math.abs(receivedAt.getTime() - rec.getTime());
  if (delta > TIMESTAMP_WINDOW_MS) {
    return { effectiveRecordedAt: receivedAt, drifted: true };
  }
  return { effectiveRecordedAt: rec, drifted: false };
}

export function checkCoordinatePlausibility(lat: number, lng: number): void {
  if (!Number.isFinite(lat) || !Number.isFinite(lng)) throw errors.invalidCoordinates();
  if (lat < -90 || lat > 90) throw errors.invalidCoordinates();
  if (lng < -180 || lng > 180) throw errors.invalidCoordinates();
  if (lat === 0 && lng === 0) throw errors.invalidCoordinates();
}

export function checkAccuracy(accuracyM: number): void {
  if (!Number.isFinite(accuracyM) || accuracyM < 0) throw errors.invalidAccuracy();
  if (accuracyM > MAX_ACCURACY_M) throw errors.invalidAccuracy();
}
export function checkTeleport(
  prev: NormalizedSample | null,
  curr: { lat: number; lng: number; recordedAt: Date },
): void {
  if (!prev) return;
  const prevDate = new Date(prev.recordedAt);
  const deltaMs = curr.recordedAt.getTime() - prevDate.getTime();
  if (deltaMs <= 0) return;
  const impliedSpeed = computeSpeedKmh(prev, curr, deltaMs);
  if (impliedSpeed > MAX_TELEPORT_SPEED_KMH) throw errors.teleportDetected(impliedSpeed);
}

export function checkSpeedConsistency(
  serverSpeedKmh: number | null,
  clientSpeedKmh: number,
): boolean {
  if (serverSpeedKmh === null) return true;
  return Math.abs(serverSpeedKmh - clientSpeedKmh) <= MAX_SPEED_MISMATCH_KMH;
}

export function checkDeviceFingerprint(
  sessionDeviceHash: string | null,
  activeSessionDeviceHash: string | null,
): boolean {
  if (!sessionDeviceHash || !activeSessionDeviceHash) return true;
  return sessionDeviceHash === activeSessionDeviceHash;
}
// ---------- Orchestrator ----------

export function runProvenance(input: ProvenanceInput): ProvenanceOutcome {
  const flags: ProvenanceFlag[] = [];

  checkCoordinatePlausibility(input.current.lat, input.current.lng);
  checkAccuracy(input.current.accuracyM);

  const { effectiveRecordedAt, drifted } = checkTimestampWindow(
    input.current.recordedAt,
    input.receivedAt,
  );
  if (drifted) flags.push('timestamp_drift');

  checkTeleport(input.previous, {
    lat: input.current.lat,
    lng: input.current.lng,
    recordedAt: effectiveRecordedAt,
  });

  let serverSpeedKmh: number | null = null;
  if (input.previous) {
    const prevDate = new Date(input.previous.recordedAt);
    const deltaMs = effectiveRecordedAt.getTime() - prevDate.getTime();
    if (deltaMs > 0) {
      serverSpeedKmh = computeSpeedKmh(input.previous, input.current, deltaMs);
    }
  }

  if (!checkSpeedConsistency(serverSpeedKmh, input.current.speedKmh)) {
    flags.push('speed_mismatch');
    return { accepted: false, flags, sample: null, reason: 'speed_mismatch' };
  }

  if (!checkDeviceFingerprint(input.sessionDeviceHash, input.activeSessionDeviceHash)) {
    return { accepted: false, flags, sample: null, reason: 'fingerprint_mismatch' };
  }

  return {
    accepted: true,
    flags,
    sample: {
      lat: input.current.lat,
      lng: input.current.lng,
      serverSpeedKmh,
      clientSpeedKmh: input.current.speedKmh,
      accuracyM: input.current.accuracyM,
      recordedAt: effectiveRecordedAt.toISOString(),
      receivedAt: input.receivedAt.toISOString(),
      timestampDrift: drifted,
    },
  };
}

export interface QRPayloadV2 {
  v: 1;
  ticketId: string;
  bookingId: string;
  scheduleId: string;
  primarySeatId: string;
  primarySeatLabel: string;
  primaryPassengerName: string;
  passengerCount: number;
  allSeatLabels: string[];
  boardingStandId: string;
  issuedAt: string;
  nonce: string;
}

const CANONICAL_KEYS: (keyof QRPayloadV2)[] = [
  'allSeatLabels', 'boardingStandId', 'bookingId', 'issuedAt', 'nonce',
  'passengerCount', 'primaryPassengerName', 'primarySeatId', 'primarySeatLabel',
  'scheduleId', 'ticketId', 'v',
];

export const MAX_PAYLOAD_BYTES = 500;
export const MAX_SIGNATURE_BYTES = 64;

export function canonicalizeToBytes(payload: QRPayloadV2): Uint8Array {
  const ordered: Record<string, unknown> = {};
  for (const k of CANONICAL_KEYS) ordered[k] = payload[k];
  return new TextEncoder().encode(JSON.stringify(ordered));
}
export function buildWireFormat(payload: QRPayloadV2, signature: Uint8Array): string {
  const bytes = canonicalizeToBytes(payload);
  return `v1:${Buffer.from(bytes).toString('base64')}:${Buffer.from(signature).toString('base64')}`;
}

export function parseWireFormat(wire: string): {
  payload: QRPayloadV2;
  signature: Uint8Array;
  canonicalBytes: Uint8Array;
} {
  const parts = wire.split(':');
  if (parts.length !== 3) throw new Error('INVALID_FORMAT');
  if (parts[0] !== 'v1') throw new Error('UNSUPPORTED_WIRE_VERSION');
  const canonicalBytes = new Uint8Array(Buffer.from(parts[1]!, 'base64'));
  const signature = new Uint8Array(Buffer.from(parts[2]!, 'base64'));
  if (signature.length !== 64) throw new Error('INVALID_SIGNATURE_LENGTH');
  let payload: QRPayloadV2;
  try {
    payload = JSON.parse(Buffer.from(canonicalBytes).toString('utf-8')) as QRPayloadV2;
  } catch {
    throw new Error('INVALID_PAYLOAD');
  }
  if (payload.v !== 1) throw new Error('UNSUPPORTED_PAYLOAD_VERSION');
  return { payload, signature, canonicalBytes };
}

let cached: Keypair | null = null;

const isPlaceholder = (v: string) => /^REPLACE/i.test(v);
const decodeB64 = (b64: string) => new Uint8Array(Buffer.from(b64, 'base64'));

async function getKeypair(): Promise<Keypair> {
  if (cached) return cached;
  const skB64 = env.QR_SIGNING_PRIVATE_KEY;
  const pkB64 = env.QR_SIGNING_PUBLIC_KEY;
  if (skB64 && pkB64 && !isPlaceholder(skB64) && !isPlaceholder(pkB64)) {
    const sk = decodeB64(skB64);
    const pk = decodeB64(pkB64);
    if (sk.length !== 32) throw new Error('QR_SIGNING_PRIVATE_KEY must be 32 raw bytes');
    if (pk.length !== 32) throw new Error('QR_SIGNING_PUBLIC_KEY must be 32 raw bytes');
    cached = { sk, pk, ephemeral: false };
    return cached;
  }
  if (isProd) throw new Error('QR_SIGNING_PRIVATE_KEY and QR_SIGNING_PUBLIC_KEY required in production');
  if (isTest) throw new Error('QR_SIGNING_*_KEY missing in .env.test — Session 0d must run first');
  console.warn('[ed25519] QR_SIGNING_*_KEY missing — using ephemeral dev key. Tickets will NOT verify across restarts.');
  const sk = ed.utils.randomSecretKey();
  const pk = await ed.getPublicKeyAsync(sk);
  cached = { sk, pk, ephemeral: true };
  return cached;
}

  const { sk } = await getKeypair();

const E164_REGEX = /^\+[1-9]\d{1,14}$/;
const INDIA_MOBILE_REGEX = /^\+91[6-9]\d{9}$/;

export function isValidPhone(phone: string): boolean {
  return E164_REGEX.test(phone);
}

export function isIndianMobile(phone: string): boolean {
  return INDIA_MOBILE_REGEX.test(phone);
}

/**
 * Normalize to E.164. Accepts:
 *  - "+919876543210"
 *  - "919876543210" (12 digits starting with 91)
 *  - "9876543210" (10 digits, Indian mobile)
 */
export function normalizePhone(phone: string): string {
  const trimmed = phone.trim().replace(/[\s\-()]/g, '');
  if (trimmed.startsWith('+')) return trimmed;
  if (trimmed.length === 12 && trimmed.startsWith('91')) return `+${trimmed}`;
  if (trimmed.length === 10) return `+91${trimmed}`;
  return trimmed;
}

/**
 * Deterministic SHA-256 hash with fixed app-level salt.
 * Used for searchable lookup of encrypted phone column.
 */
export function computePhoneHash(phone: string): string {
  return sha256(normalizePhone(phone), env.PHONE_HASH_SALT);
}

/**
 * Mask all but last 4 chars: "••••••3210"
 */
export function maskPhone(phone: string): string {
  if (phone.length < 4) return '••••';
  return `${'•'.repeat(phone.length - 4)}${phone.slice(-4)}`;
}

/**
 * Generate a device fingerprint hash from an arbitrary client-provided
 * fingerprint string. Used for single-device binding on driver sessions.
 */
export function hashDeviceFingerprint(fingerprint: string): string {
  // Fixed app-level salt — deterministic for lookup
  return crypto.createHash('sha256').update(`dfp:${fingerprint}`).digest('hex');
}

/**
 * Standard response envelope.
 * Success: { data: T, meta?: { requestId?, durationMs?, ... } }
 * Error:   { error: { code, message, details?, requestId? } }
 */

export interface SuccessEnvelope<T> {
  data: T;
  meta?: {
    requestId?: string;
    durationMs?: number;
    [k: string]: unknown;
  };
}

export interface ErrorEnvelope {
  error: {
    code: string;
    message: string;
    details?: Record<string, unknown>;
    requestId?: string;
  };
}

export function success<T>(data: T, meta?: SuccessEnvelope<T>['meta']): SuccessEnvelope<T> {
  return meta ? { data, meta } : { data };
}

export function failure(
  code: string,
  message: string,
  details?: Record<string, unknown>,
  requestId?: string,
): ErrorEnvelope {
  return {
    error: {
      code,
      message,
      ...(details ? { details } : {}),
      ...(requestId ? { requestId } : {}),
    },
  };
}

import type { FastifyInstance } from 'fastify';
import { verifyAccessToken } from '../modules/identity/jwt.service.js';
import { prisma } from '../lib/prisma.js';
import { redis, keys } from '../lib/redis.js';

export async function authMiddleware(server: FastifyInstance): Promise<void> {
  server.addHook('onRequest', async (request) => {
    const header = request.headers.authorization;
    if (!header || !header.startsWith('Bearer ')) {
      request.user = null;
      return;
    }

    const token = header.slice('Bearer '.length).trim();
    let claims;
    try {
      claims = verifyAccessToken(token);
    } catch {
      request.user = null;
      return;
    }

    const revoked = await redis.exists(keys.revokedSession(claims.sid));
    if (revoked === 1) {
      request.user = null;
      return;
    }

    const session = await prisma.session.findUnique({ where: { id: claims.sid } });
    if (!session || session.status !== 'ACTIVE' || session.expiresAt < new Date()) {
      request.user = null;
      return;
    }

    request.user = {
      userId: claims.sub,
      role: claims.role,
      sessionId: claims.sid,
    };
  });
}

export interface WsClaims {
  userId: string;
  sessionId: string;
  role: string;
  exp: number;
}

export type WsAuthReason = 'INVALID_TOKEN' | 'SESSION_REVOKED' | 'SESSION_INACTIVE';

export class WsAuthError extends Error {
  constructor(public readonly reason: WsAuthReason) {
    super(reason);
    this.name = 'WsAuthError';
  }
}

export function isWsAuthError(err: unknown): err is { reason: WsAuthReason } {
  return !!err && typeof err === 'object' && typeof (err as { reason?: unknown }).reason === 'string';
}

export async function verifyHandshake(token: string): Promise<WsClaims> {
  let claims;
  try {
    claims = verifyAccessToken(token);
  } catch {
    throw new WsAuthError('INVALID_TOKEN');
  }

  const revoked = await redis.exists(keys.revokedSession(claims.sid));
  if (revoked === 1) throw new WsAuthError('SESSION_REVOKED');

  const session = await prisma.session.findUnique({ where: { id: claims.sid } });
  if (!session || session.status !== 'ACTIVE' || session.expiresAt < new Date()) {
    throw new WsAuthError('SESSION_INACTIVE');
  }

  return {
    userId: claims.sub,
    sessionId: claims.sid,
    role: claims.role,
    exp: claims.exp,
  };
}

import type { Socket } from 'socket.io';
import { prisma } from '../lib/prisma.js';
import { logger } from '../lib/logger.js';
import type { WsClaims } from './auth.js';

export interface SubscribeResult {
  allowed: boolean;
  reason?: string;
}

const UUID = '[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}';
const SEAT_AVAILABILITY_RE = new RegExp(`^seat:availability:(${UUID})$`);
const TRIP_LIVE_RE = new RegExp(`^trip:live:(${UUID})$`);
const BOOKING_RE = new RegExp(`^booking:(${UUID})$`);
const ADMIN_NOTIFICATIONS = 'admin:notifications';

async function canSeatAvailability(scheduleId: string): Promise<boolean> {
  const row = await prisma.schedule.findFirst({
    where: { id: scheduleId, status: 'PUBLISHED', deletedAt: null },
    select: { id: true },
  });
  return !!row;
}

async function canTripLive(tripId: string, claims: WsClaims): Promise<boolean> {
  const trip = await prisma.trip.findUnique({
    where: { id: tripId },
    select: { scheduleId: true },
  });
  if (!trip) return false;
  if (claims.role === 'SUPER_ADMIN') return true;
  const booking = await prisma.booking.findFirst({
    where: {
      userId: claims.userId,
      scheduleId: trip.scheduleId,
      status: { in: ['CONFIRMED', 'CHECKED_IN'] },
      deletedAt: null,
    },
    select: { id: true },
  });
  return !!booking;
}

async function canBooking(bookingId: string, claims: WsClaims): Promise<boolean> {
  if (claims.role === 'SUPER_ADMIN') return true;
  const b = await prisma.booking.findFirst({
    where: { id: bookingId, userId: claims.userId, deletedAt: null },
    select: { id: true },
  });
  return !!b;
}

export async function subscribeToTopic(
  socket: Socket,
  topic: string,
  claims: WsClaims,
): Promise<SubscribeResult> {
  if (topic === ADMIN_NOTIFICATIONS) {
    if (claims.role !== 'SUPER_ADMIN') return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  let m = topic.match(SEAT_AVAILABILITY_RE);
  if (m) {
    if (!(await canSeatAvailability(m[1]!))) return { allowed: false, reason: 'SCHEDULE_NOT_FOUND' };
    socket.join(topic);
    return { allowed: true };
  }

  m = topic.match(TRIP_LIVE_RE);
  if (m) {
    if (!(await canTripLive(m[1]!, claims))) return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  m = topic.match(BOOKING_RE);
  if (m) {
    if (!(await canBooking(m[1]!, claims))) return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  logger.warn({ topic, userId: claims.userId }, 'unknown topic pattern');
  return { allowed: false, reason: 'UNKNOWN_TOPIC' };
}

import { ApiResponse, ApiError, ApiErrorResponse, Surface } from './types';
import { CONFIG } from '../config';

type Method = 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';

interface Options {
  method?: Method;
  body?: unknown;
  headers?: Record<string, string>;
  skipAuth?: boolean;
}

let isRefreshing = false;
let subscribers: Array<(token: string) => void> = [];

class HttpClient {
  private baseUrl: string;
  private surface: Surface | null = null;

  constructor() {
    this.baseUrl = typeof window !== 'undefined' ? CONFIG.API_BASE : 'http://localhost:4310/v1';
    if (typeof window !== 'undefined') {
      this.surface = (localStorage.getItem('buscon_surface') as Surface | null);
    }
  }

  public setSurface(surface: Surface | null): void {
    this.surface = surface;
    if (typeof window === 'undefined') return;
    if (surface) localStorage.setItem('buscon_surface', surface);
    else localStorage.removeItem('buscon_surface');
  }

  private getAccess(): string | null {
    return typeof window !== 'undefined' ? localStorage.getItem('buscon_access_token') : null;
  }

  private getRefresh(): string | null {
    return typeof window !== 'undefined' ? localStorage.getItem('buscon_refresh_token') : null;
  }

  private saveTokens(access: string, refresh: string): void {
    if (typeof window === 'undefined') return;
    localStorage.setItem('buscon_access_token', access);
    localStorage.setItem('buscon_refresh_token', refresh);
  }

  private clearAuth(): void {
    if (typeof window === 'undefined') return;
    localStorage.removeItem('buscon_access_token');
    localStorage.removeItem('buscon_refresh_token');
    localStorage.removeItem('buscon_user');
    localStorage.removeItem('buscon_surface');
    this.surface = null;
    window.dispatchEvent(new CustomEvent('buscon:auth:logout'));
  }

  private async refreshTokens(): Promise<string> {
    const refreshToken = this.getRefresh();
    if (!refreshToken || !this.surface) {
      this.clearAuth();
      throw new ApiError(401, { code: 'SESSION_REVOKED', message: 'Session expired.' });
    }

    const res = await fetch(`${this.baseUrl}/${this.surface}/auth/refresh`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ refreshToken }),
    });

    if (!res.ok) {
      this.clearAuth();
      throw new ApiError(res.status, { code: 'SESSION_REVOKED', message: 'Session expired.' });
    }

    const payload = (await res.json()) as ApiResponse<{ accessToken: string; refreshToken: string }>;
    this.saveTokens(payload.data.accessToken, payload.data.refreshToken);
    return payload.data.accessToken;
  }

  public async request<T>(endpoint: string, opts: Options = {}): Promise<ApiResponse<T>> {
    const { method = 'GET', body, headers = {}, skipAuth = false } = opts;
    const url = `${this.baseUrl}${endpoint.startsWith('/') ? endpoint : `/${endpoint}`}`;
    const reqHeaders: Record<string, string> = { 'Content-Type': 'application/json', ...headers };

    if (!skipAuth) {
      const token = this.getAccess();
      if (token) reqHeaders.Authorization = `Bearer ${token}`;
    }

    const init: RequestInit = { method, headers: reqHeaders };
    if (body !== undefined && method !== 'GET') init.body = JSON.stringify(body);

    const response = await fetch(url, init);

    if (response.status === 401 && !skipAuth) {
      if (!isRefreshing) {
        isRefreshing = true;
        try {
          const fresh = await this.refreshTokens();
          subscribers.forEach((cb) => cb(fresh));
          subscribers = [];
        } catch (err) {
          subscribers = [];
          throw err;
        } finally {
          isRefreshing = false;
        }
      }

      return new Promise((resolve, reject) => {
        subscribers.push(async (token) => {
          try {
            const retryHeaders = { ...reqHeaders, Authorization: `Bearer ${token}` };
            const replayed = await fetch(url, { ...init, headers: retryHeaders });
            resolve(await this.parse<T>(replayed));
          } catch (err) { reject(err); }
        });
      });
    }

    return this.parse<T>(response);
  }

  private async parse<T>(response: Response): Promise<ApiResponse<T>> {
    const ct = response.headers.get('content-type') || '';
    const isJson = ct.includes('application/json');

    if (!response.ok) {
      if (isJson) {
        const errJson = (await response.json()) as ApiErrorResponse;
        throw new ApiError(response.status, errJson.error || { code: 'INTERNAL_ERROR', message: response.statusText });
      }
      throw new ApiError(response.status, { code: 'NETWORK_ERROR', message: response.statusText || 'Network error' });
    }

    if (response.status === 204) return { data: {} as T };
    return (await response.json()) as ApiResponse<T>;
  }

  public get<T>(endpoint: string) { return this.request<T>(endpoint, { method: 'GET' }); }
  public post<T>(endpoint: string, body?: unknown, opts: Omit<Options, 'method' | 'body'> = {}) {
    return this.request<T>(endpoint, { method: 'POST', body, ...opts });
  }
  public put<T>(endpoint: string, body?: unknown) {
    return this.request<T>(endpoint, { method: 'PUT', body });
  }
  public patch<T>(endpoint: string, body?: unknown) {
    return this.request<T>(endpoint, { method: 'PATCH', body });
  }
  public delete<T>(endpoint: string) { return this.request<T>(endpoint, { method: 'DELETE' }); }
}

'use client';

import React, { createContext, useContext, useEffect, useState, useCallback } from 'react';
import { api } from '../api/client';
import { UserSession, Surface, ApiError } from '../api/types';
import { DEMO } from '../config';

function decodeJwtPayload(token: string): any {
  try {
    const parts = token.split('.');
    if (parts.length !== 3) return null;
    const base64Url = parts[1];
    const base64 = base64Url.replace(/-/g, '+').replace(/_/g, '/');
    const padded = base64 + '='.repeat((4 - (base64.length % 4)) % 4);
    return JSON.parse(atob(padded));
  } catch {
    return null;
  }
}

interface AuthCtx {
  user: UserSession | null;
  surface: Surface | null;
  isAdmin: boolean;
  isAuthenticated: boolean;
  isLoading: boolean;
  loginOperator: (email: string, password: string) => Promise<void>;
  loginAdmin: (email: string, password: string) => Promise<void>;
  switchDemoRole: () => Promise<void>;
  logout: () => Promise<void>;
}

const Ctx = createContext<AuthCtx | undefined>(undefined);

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<UserSession | null>(null);
  const [surface, setSurface] = useState<Surface | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    try {
      const token = localStorage.getItem('buscon_access_token');
      const userJson = localStorage.getItem('buscon_user');
      const savedSurface = localStorage.getItem('buscon_surface') as Surface | null;
      if (token && userJson && savedSurface) {
        const storedUser = JSON.parse(userJson);
        // Re-decode role in case it was missing in a previous session
        const payload = decodeJwtPayload(token);
        const role = payload?.role || storedUser.role || (savedSurface === 'admin' ? 'SUPER_ADMIN' : 'OPERATOR_OWNER');
        setUser({ ...storedUser, role });
        setSurface(savedSurface);
        api.setSurface(savedSurface);
      }
    } catch { /* ignore */ } finally { setIsLoading(false); }

    const onLogout = () => { setUser(null); setSurface(null); };
    window.addEventListener('buscon:auth:logout', onLogout);
    return () => window.removeEventListener('buscon:auth:logout', onLogout);
  }, []);

  const doLogin = useCallback(async (surfaceParam: Surface, email: string, password: string) => {
    api.setSurface(surfaceParam);
    const res = await api.post<{
      accessToken: string; refreshToken: string; expiresAt: string; user: UserSession;
    }>(`/${surfaceParam}/auth/login`, { email, password }, { skipAuth: true });

    const { accessToken, refreshToken, user: u } = res.data;
    
    // Decode JWT to extract role reliably
    const payload = decodeJwtPayload(accessToken);
    const role = payload?.role || payload?.userRole || payload?.user?.role || u.role || 
      (surfaceParam === 'admin' ? 'SUPER_ADMIN' : 'OPERATOR_OWNER');
    
    const userWithRole: UserSession = { ...u, role };

    localStorage.setItem('buscon_access_token', accessToken);
    localStorage.setItem('buscon_refresh_token', refreshToken);
    localStorage.setItem('buscon_user', JSON.stringify(userWithRole));
    setUser(userWithRole);
    setSurface(surfaceParam);
  }, []);

  const loginOperator = useCallback((email: string, password: string) => doLogin('operator', email, password), [doLogin]);
  const loginAdmin = useCallback((email: string, password: string) => doLogin('admin', email, password), [doLogin]);

  // Compute isAdmin from either surface or role (defensive — covers all cases)
  const isAdmin = surface === 'admin' || user?.role === 'SUPER_ADMIN';

  const switchDemoRole = useCallback(async () => {
    if (isAdmin) {
      if (!DEMO.operatorEmail || !DEMO.operatorPassword) throw new ApiError(400, { code: 'DEMO_NOT_CONFIGURED', message: 'Demo credentials missing in .env.local' });
      await doLogin('operator', DEMO.operatorEmail, DEMO.operatorPassword);
    } else {
      if (!DEMO.adminEmail || !DEMO.adminPassword) throw new ApiError(400, { code: 'DEMO_NOT_CONFIGURED', message: 'Demo credentials missing in .env.local' });
      await doLogin('admin', DEMO.adminEmail, DEMO.adminPassword);
    }
  }, [isAdmin, doLogin]);

  const logout = useCallback(async () => {
    try { await api.post(`/${surface}/auth/logout`, {}); } catch { /* ignore */ }
    localStorage.removeItem('buscon_access_token');
    localStorage.removeItem('buscon_refresh_token');
    localStorage.removeItem('buscon_user');
    localStorage.removeItem('buscon_surface');
    api.setSurface(null);
    setUser(null);
    setSurface(null);
  }, [surface]);

  return (
    <Ctx.Provider value={{
      user, surface, isAdmin, isLoading,
      isAuthenticated: !!user,
      loginOperator, loginAdmin, switchDemoRole, logout,
    }}>
      {children}
    </Ctx.Provider>
  );
}

export function useAuth(): AuthCtx {
  const ctx = useContext(Ctx);
  if (!ctx) throw new Error('useAuth must be used within AuthProvider');
  return ctx;
}

import { io, Socket } from 'socket.io-client';
import { CONFIG } from '../config';

type SocketEventListener = (payload: any) => void;

class WebSocketManager {
  private socket: Socket | null = null;
  private subscribedTopics: Set<string> = new Set();
  private listeners: Map<string, Set<SocketEventListener>> = new Map();
  private authTimeoutTimer: NodeJS.Timeout | null = null;

  public connect(accessToken: string) {
    if (this.socket && this.socket.connected) {
      return;
    }

    if (this.socket) {
      this.socket.disconnect();
    }

    // Connect to backend Fastify Socket.io server
    this.socket = io(CONFIG.WS_URL, {
      path: '/v1/ws',
      transports: ['websocket', 'polling'],
      reconnectionDelay: 1000,
      reconnectionDelayMax: 16000,
      reconnectionAttempts: 5,
      autoConnect: true,
    });

    // RULE 8.11: Handshake timing — server terminates after 5s if auth is delayed
    this.socket.on('connect', () => {
      this.socket?.emit('message', {
        action: 'auth',
        token: accessToken,
      });

      // Clear safeguard timer once connect initiates
      if (this.authTimeoutTimer) clearTimeout(this.authTimeoutTimer);
    });

    this.socket.on('auth:ok', () => {
      // Re-subscribe to all active topics on reconnection
      if (this.subscribedTopics.size > 0) {
        this.socket?.emit('message', {
          action: 'subscribe',
          topics: Array.from(this.subscribedTopics),
        });
      }
    });

    // Server sends auth:expiring at (exp - 60s)
    this.socket.on('auth:expiring', async () => {
      const refreshToken = localStorage.getItem('busverse_refresh_token');
      if (refreshToken && this.socket) {
        this.socket.emit('message', {
          action: 'auth:refresh',
          refreshToken,
        });
      }
    });

    this.socket.on('auth:refreshed', (data: { accessToken: string; refreshToken: string }) => {
      localStorage.setItem('busverse_access_token', data.accessToken);
      localStorage.setItem('busverse_refresh_token', data.refreshToken);
    });

    // General message router for incoming events
    this.socket.on('message', (msg: { event?: string; topic?: string; data?: any }) => {
      if (msg.event) {
        const topicListeners = this.listeners.get(msg.event);
        if (topicListeners) {
          topicListeners.forEach((listener) => listener(msg.data));
        }
      }
    });

    // Close code management
    this.socket.on('disconnect', (reason) => {
      if (reason === 'io server disconnect') {
        // Closed by server (e.g., Code 4001, 4003 or 1008)
        const freshToken = localStorage.getItem('busverse_access_token');
        if (freshToken) {
          setTimeout(() => this.connect(freshToken), 1500);
        }
      }
    });
  }

  public subscribe(topics: string | string[]) {
    const topicArray = Array.isArray(topics) ? topics : [topics];
    topicArray.forEach((t) => this.subscribedTopics.add(t));

    if (this.socket?.connected) {
      this.socket.emit('message', {
        action: 'subscribe',
        topics: topicArray,
      });
    }
  }

  public unsubscribe(topics: string | string[]) {
    const topicArray = Array.isArray(topics) ? topics : [topics];
    topicArray.forEach((t) => this.subscribedTopics.delete(t));

    if (this.socket?.connected) {
      this.socket.emit('message', {
        action: 'unsubscribe',
        topics: topicArray,
      });
    }
  }

  public on(event: string, callback: SocketEventListener) {
    if (!this.listeners.has(event)) {
      this.listeners.set(event, new Set());
    }
    this.listeners.get(event)!.add(callback);

    // Also register on underlying socket instance
    this.socket?.on(event, callback);

    return () => {
      this.listeners.get(event)?.delete(callback);
      this.socket?.off(event, callback);
    };
  }

  public disconnect() {
    if (this.authTimeoutTimer) clearTimeout(this.authTimeoutTimer);
    this.subscribedTopics.clear();
    this.listeners.clear();
    this.socket?.disconnect();
    this.socket = null;
  }
}

'use client';

import { useEffect } from 'react';
import { wsManager } from './socket';

/**
 * useWsEvent hook
 * Subscribes to a WebSocket topic and attaches a listener with automatic cleanup.
 */
export function useWsEvent<T = any>(
  topic: string | null,
  event: string,
  handler: (payload: T) => void
) {
  useEffect(() => {
    if (!topic) return;

    wsManager.subscribe(topic);
    const unbind = wsManager.on(event, handler);

    return () => {
      unbind();
      wsManager.unsubscribe(topic);
    };
  }, [topic, event, handler]);
}
