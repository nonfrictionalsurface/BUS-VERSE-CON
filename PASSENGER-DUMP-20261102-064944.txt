import React, { useEffect, useRef } from 'react';
import { View, Animated, Easing, StyleSheet, Text } from 'react-native';
import { useTheme } from '../../theme/ThemeContext';

export function TransitLoader({ message = 'CONNECTING TRANSIT' }: { message?: string }) {
  const { tokens } = useTheme();
  const bounce1 = useRef(new Animated.Value(0)).current;
  const bounce2 = useRef(new Animated.Value(0)).current;
  const bounce3 = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    const createBounce = (val: Animated.Value, delay: number) => {
      return Animated.loop(
        Animated.sequence([
          Animated.delay(delay),
          Animated.timing(val, { toValue: -10, duration: 300, easing: Easing.out(Easing.ease), useNativeDriver: true }),
          Animated.timing(val, { toValue: 0, duration: 300, easing: Easing.in(Easing.ease), useNativeDriver: true }),
          Animated.delay(600 - delay),
        ])
      );
    };

    Animated.parallel([
      createBounce(bounce1, 0),
      createBounce(bounce2, 150),
      createBounce(bounce3, 300)
    ]).start();
  }, []);

  return (
    <View style={styles.container}>
      <View style={styles.dotsRow}>
        <Animated.View style={[styles.dot, { backgroundColor: tokens.brand.primary, transform: [{ translateY: bounce1 }] }]} />
        <Animated.View style={[styles.dot, { backgroundColor: tokens.telemetry.cyan, transform: [{ translateY: bounce2 }] }]} />
        <Animated.View style={[styles.dot, { backgroundColor: tokens.signal.emerald, transform: [{ translateY: bounce3 }] }]} />
      </View>
      <Text style={[styles.text, { color: tokens.text.muted }]}>{message}</Text>
    </View>
  );
}

  visible,
  selectedDate,
  onClose,
  onSelectDate,
}: DatePickerModalProps) {
  const { tokens } = useTheme();

  // Generate the next 14 calendar dates
  const nextDates = Array.from({ length: 14 }).map((_, i) => {
    const d = new Date();
    d.setDate(d.getDate() + i);

import React from 'react';
import Svg, { Defs, LinearGradient as SvgLinearGradient, Stop, Rect } from 'react-native-svg';

interface Props {
  from: string;
  to: string;
  width?: number | string;
  height?: number;
  radius?: number;
}

export function GradientBar({ from, to, width = '100%', height = 4, radius = 2 }: Props) {
  const w = typeof width === 'number' ? width : 400;
  return (
    <Svg width={width as any} height={height} viewBox={`0 0 ${w} ${height}`} preserveAspectRatio="none">
      <Defs>
        <SvgLinearGradient id="gb" x1="0%" y1="0%" x2="100%" y2="0%">
          <Stop offset="0%" stopColor={from} />
          <Stop offset="100%" stopColor={to} />
        </SvgLinearGradient>
      </Defs>
      <Rect x="0" y="0" width={w} height={height} rx={radius} fill="url(#gb)" />
    </Svg>
  );
}

export function GradientCard({ from, to, height = 3 }: { from: string; to: string; height?: number }) {
  return <GradientBar from={from} to={to} height={height} radius={0} />;
}

import React from 'react';

interface SeatDeckProps {
  seats: SeatItem[];

export const SeatDeck: React.FC<SeatDeckProps> = ({ seats, selectedSeatId, onSelectSeat }) => {
  const { tokens } = useTheme();

  const rows: Record<string, SeatItem[]> = {};
  seats.forEach((seat) => {
    if (!rows[seat.seatRow]) rows[seat.seatRow] = [];
    rows[seat.seatRow].push(seat);
  });

  return (
    <View style={[styles.fuselage, { backgroundColor: tokens.surfaceCard, borderColor: tokens.borderStrong }]}>
      <View style={[styles.windshield, { borderColor: tokens.telemetryCyan, backgroundColor: tokens.canvasSubtle }]} />
      {Object.keys(rows).sort().map((rowLetter) => {
        const rowSeats = rows[rowLetter].sort((a, b) => a.seatColumn - b.seatColumn);
        const leftPair = rowSeats.slice(0, 2);
        const rightPair = rowSeats.slice(2);

        return (
          <View key={rowLetter} style={styles.rowContainer}>
            <View style={styles.pair}>
              {leftPair.map((s) => (
                <SeatCell
                  key={s.id}

export function resolveCityIds(
  cities: ApiCity[],
  originName: string,
  destName: string,
): { originCityId: string | null; destinationCityId: string | null } {
  const norm = (s: string) => s.trim().toLowerCase().replace(/[^a-z0-9]/g, '');

  const find = (name: string) => {
    if (!name.trim()) return null;
    const target = norm(name);
    
    let match = cities.find((c) => norm(c.name) === target);
    if (!match) match = cities.find((c) => norm(c.name).startsWith(target));
    if (!match) match = cities.find((c) => norm(c.name).includes(target));

    return match?.id ?? null;
  };

  return {
    originCityId: find(originName),
    destinationCityId: find(destName),
  };
}

import AsyncStorage from '@react-native-async-storage/async-storage';
import { CONFIG } from '../config';

export interface ApiResponse<T> {
  data: T;
  meta?: { requestId?: string; nextCursor?: string | null };
}

export class ApiError extends Error {
  constructor(public status: number, public code: string, message: string) {
    super(message);
    this.name = 'ApiError';
  }
}

class HttpClient {
  private async getAccessToken() {
    return AsyncStorage.getItem('busverse_access_token');
  }

  public async request<T>(endpoint: string, options: RequestInit & { idempotencyKey?: string } = {}): Promise<ApiResponse<T>> {
    const token = await this.getAccessToken();
    const headers: Record<string, string> = {
      'Content-Type': 'application/json',
      'X-Request-Id': `req_${Date.now()}`,
      ...(options.headers as Record<string, string>),
    };

    if (token) headers['Authorization'] = `Bearer ${token}`;
    if (options.idempotencyKey) headers['Idempotency-Key'] = options.idempotencyKey;

    const url = `${CONFIG.API_BASE}${endpoint.startsWith('/') ? endpoint : `/${endpoint}`}`;
    const res = await fetch(url, { ...options, headers });

    if (res.status === 401) {
      const refreshToken = await AsyncStorage.getItem('busverse_refresh_token');
      if (refreshToken) {
        const refreshRes = await fetch(`${CONFIG.API_BASE}/passenger/auth/refresh`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ refreshToken }),
        });
        if (refreshRes.ok) {
          const fresh = await refreshRes.json();
          await AsyncStorage.setItem('busverse_access_token', fresh.data.accessToken);
          await AsyncStorage.setItem('busverse_refresh_token', fresh.data.refreshToken);
          headers['Authorization'] = `Bearer ${fresh.data.accessToken}`;
          const retried = await fetch(url, { ...options, headers });
          return retried.json();
        }
      }
      await AsyncStorage.multiRemove(['busverse_access_token', 'busverse_refresh_token', 'busverse_user']);
      throw new ApiError(401, 'UNAUTHORIZED', 'Session expired');
    }

    if (!res.ok) {
      const err = await res.json().catch(() => ({}));
      throw new ApiError(res.status, err.error?.code || 'ERROR', err.error?.message || 'Request failed');
    }

    if (res.status === 204) return { data: {} as T };
    return res.json();
  }

  get<T>(endpoint: string) {
    return this.request<T>(endpoint, { method: 'GET' });
  }

  post<T>(endpoint: string, body?: unknown, idempotencyKey?: string) {
    return this.request<T>(endpoint, {
      method: 'POST',
      body: body ? JSON.stringify(body) : undefined,
      idempotencyKey,
    });
  }

  delete<T>(endpoint: string) {
    return this.request<T>(endpoint, { method: 'DELETE' });
  }
}

export const api = new HttpClient();

import React, { createContext, useContext, useEffect, useState } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { api } from '../api/client';
import { wsManager } from '../ws/socket';
import { UserSession } from '../../types';

interface AuthContextValue {
  user: UserSession | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  loginWithTokens: (tokens: { accessToken: string; refreshToken: string; user: UserSession }) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | undefined>(undefined);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [user, setUser] = useState<UserSession | null>(null);
  const [isLoading, setIsLoading] = useState(true);

  useEffect(() => {
    (async () => {
      const token = await AsyncStorage.getItem('busverse_access_token');
      const cached = await AsyncStorage.getItem('busverse_user');
      if (token && cached) {
        setUser(JSON.parse(cached));
        wsManager.connect(token);
        api.get<{ user: UserSession }>('/passenger/me')
          .then((res) => {
            setUser(res.data.user);
            AsyncStorage.setItem('busverse_user', JSON.stringify(res.data.user));
          })
          .catch(() => {});
      }
      setIsLoading(false);
    })();
  }, []);

  const loginWithTokens = async (tokens: { accessToken: string; refreshToken: string; user: UserSession }) => {
    await AsyncStorage.setItem('busverse_access_token', tokens.accessToken);
    await AsyncStorage.setItem('busverse_refresh_token', tokens.refreshToken);
    await AsyncStorage.setItem('busverse_user', JSON.stringify(tokens.user));
    setUser(tokens.user);
    wsManager.connect(tokens.accessToken);
  };

  const logout = async () => {
    try { await api.post('/passenger/auth/logout'); } catch {}
    await AsyncStorage.multiRemove(['busverse_access_token', 'busverse_refresh_token', 'busverse_user']);
    setUser(null);
    wsManager.disconnect();
  };

  return (
    <AuthContext.Provider value={{ user, isAuthenticated: !!user, isLoading, loginWithTokens, logout }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error('useAuth must be used within AuthProvider');
  return ctx;
};

  return 'v1:' + b64 + ':' + fakeSignature;

import { io, Socket } from 'socket.io-client';
import { CONFIG } from '../config';

type Listener = (data: any) => void;

class WebSocketManager {
  private socket: Socket | null = null;
  private topics: Set<string> = new Set();
  private listeners: Map<string, Set<Listener>> = new Map();

  connect(token: string) {
    if (this.socket?.connected) return;
    this.socket?.disconnect();

    this.socket = io(CONFIG.WS_URL, {
      path: '/v1/ws',
      transports: ['websocket'],
      reconnectionAttempts: 5,
      reconnectionDelay: 1000,
    });

    this.socket.on('connect', () => {
      this.socket?.emit('message', { action: 'auth', token });
    });

    this.socket.on('auth:ok', () => {
      if (this.topics.size > 0) {
        this.socket?.emit('message', { action: 'subscribe', topics: Array.from(this.topics) });
      }
    });

    this.socket.on('message', (msg: { event?: string; data?: any }) => {
      if (msg.event) {
        this.listeners.get(msg.event)?.forEach((fn) => fn(msg.data));
      }
    });
  }

  subscribe(topic: string) {
    this.topics.add(topic);
    if (this.socket?.connected) {
      this.socket.emit('message', { action: 'subscribe', topics: [topic] });
    }
  }

  unsubscribe(topic: string) {
    this.topics.delete(topic);
    if (this.socket?.connected) {
      this.socket.emit('message', { action: 'unsubscribe', topics: [topic] });
    }
  }

  on(event: string, callback: Listener) {
    if (!this.listeners.has(event)) this.listeners.set(event, new Set());
    this.listeners.get(event)!.add(callback);
    return () => {
      this.listeners.get(event)?.delete(callback);
    };
  }

  disconnect() {
    this.topics.clear();
    this.listeners.clear();
    this.socket?.disconnect();
    this.socket = null;
  }
}

export const wsManager = new WebSocketManager();

  const { tokens } = useTheme();

  const [selectedSeat, setSelectedSeat] = useState<SeatItem | null>(null);
  const [activeLock, setActiveLock] = useState<{ lockId: string } | null>(null);
  const [loading, setLoading] = useState(true);

  const fetchSeats = () => {
    api.get<ScheduleAvailability>(`/passenger/schedules/${scheduleId}/availability`)
      .then((res) => setAvailability(res.data))
      .catch(console.error)
      .finally(() => setLoading(false));
  };

    fetchSeats();

    wsManager.subscribe(topic);

    const u1 = wsManager.on('seat:locked', fetchSeats);
    const u2 = wsManager.on('seat:released', fetchSeats);
    const u3 = wsManager.on('seat:booked', fetchSeats);

    return () => {
      wsManager.unsubscribe(topic);
      u1(); u2(); u3();
    };

    if (selectedSeat?.id === seat.id && activeLock) {

      setActiveLock(null);
      fetchSeats();
      return;
    }

    try {
      const res = await api.post<{ lockId: string }>('/passenger/seat-locks', { scheduleId, seatId: seat.id });
      setSelectedSeat(seat);
      setActiveLock(res.data);
    } catch {
      fetchSeats();
    }
  };

  if (loading || !availability) {
    return <ActivityIndicator size="large" color={tokens.brandPrimary} style={{ flex: 1 }} />;
  }

  return (

import React, { createContext, useContext, useState, useEffect } from 'react';
import { useColorScheme } from 'react-native';
import { brandTokens, ThemeTokens } from './tokens';

interface ThemeContextValue {
  theme: 'light' | 'dark';
  tokens: ThemeTokens;
  toggleTheme: () => void;
  setTheme: (t: 'light' | 'dark') => void;
}

const ThemeContext = createContext<ThemeContextValue | undefined>(undefined);

export const ThemeProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const systemScheme = useColorScheme();
  const [theme, setThemeState] = useState<'light' | 'dark'>('dark');

  useEffect(() => {
    if (systemScheme) {
      setThemeState(systemScheme === 'light' ? 'light' : 'dark');
    }
  }, [systemScheme]);

  const toggleTheme = () => setThemeState((curr) => (curr === 'dark' ? 'light' : 'dark'));
  const setTheme = (t: 'light' | 'dark') => setThemeState(t);

  return (
    <ThemeContext.Provider value={{ theme, tokens: brandTokens[theme], toggleTheme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
};

export const useTheme = () => {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme must be used within ThemeProvider');
  return ctx;
};
