const STATUSBAR_OFFSET = Platform.OS === 'android' ? (RNStatusBar.currentHeight || 28) + 4 : 44;

function MainAppShell() {
  const { theme, tokens, toggleTheme } = useTheme();

  const [activeTab, setActiveTab] = useState<'SEARCH' | 'TRACK' | 'TICKETS'>('SEARCH');
  const [selectedSchedule, setSelectedSchedule] = useState<DepartureSchedule | null>(null);
  const [selectedSeats, setSelectedSeats] = useState<string[]>([]);
  const [totalFare, setTotalFare] = useState(0);
  const [bookingData, setBookingData] = useState<any>(null);
  const [bookingStep, setBookingStep] = useState<'NONE' | 'SEATS' | 'MANIFEST' | 'PAYMENT'>('NONE');
  const [profileMenuOpen, setProfileMenuOpen] = useState(false);

  // Flow navigation handlers
  const handleSelectSchedule = (schedule: DepartureSchedule) => {
    setSelectedSchedule(schedule);
    setBookingStep('SEATS');
  };

  const handleSeatConfirm = (seats: string[], fare: number) => {
    setSelectedSeats(seats);
    setTotalFare(fare);
    setBookingStep('MANIFEST');
  };

  const handleProceedToPayment = (data: any) => {
    setBookingData(data);
    setBookingStep('PAYMENT');
  };

  const handlePaymentSuccess = () => {
    setBookingStep('NONE');
    setSelectedSchedule(null);
    setActiveTab('TICKETS');
  };

  return (
    <View style={[styles.rootContainer, { backgroundColor: tokens.canvas.bg, paddingTop: STATUSBAR_OFFSET }]}>
      <StatusBar style={theme === 'dark' ? 'light' : 'dark'} />

      {/* Header */}
      <View style={[styles.header, { backgroundColor: tokens.surface.card, borderColor: tokens.border.hairline }]}>
        <BusVerseLogo size={32} />

        <View style={styles.headerRight}>
          {/* Theme Toggle Pill */}
          <TouchableOpacity
            style={[styles.themePill, { borderColor: tokens.border.subtle, backgroundColor: tokens.surface.card }]}
            onPress={toggleTheme}
            activeOpacity={0.8}
          >
            <Ionicons
              name={theme === 'dark' ? 'sunny' : 'moon-outline'}
              size={13}
              color={theme === 'dark' ? tokens.warning.amber : tokens.brand.primary}
              style={{ marginRight: 4 }}
            />
            <Text style={[styles.themePillText, { color: tokens.text.primary }]}>
              {theme === 'dark' ? 'Light Mode' : 'Dark Mode'}
            </Text>
          </TouchableOpacity>

          {/* Profile Menu Pill */}
          <TouchableOpacity
            style={[styles.profilePill, { backgroundColor: tokens.brand.subtle, borderColor: tokens.brand.border }]}
            onPress={() => setProfileMenuOpen(true)}
            activeOpacity={0.8}
          >
            <Ionicons name="person" size={11} color={tokens.brand.primary} style={{ marginRight: 4 }} />
            <Text style={[styles.profilePillText, { color: tokens.brand.primary }]}>+917235041123</Text>
          </TouchableOpacity>
        </View>
      </View>

      {/* Main Viewport Stage */}
      <View style={styles.bodyStage}>
        {bookingStep === 'SEATS' && selectedSchedule && (
          <SeatSelectionScreen
            schedule={selectedSchedule}
            onBack={() => setBookingStep('NONE')}
            onConfirmReservation={handleSeatConfirm}
          />
        )}

        {bookingStep === 'MANIFEST' && selectedSchedule && (
          <PassengerManifestScreen
            schedule={selectedSchedule}
            selectedSeats={selectedSeats}
            totalFare={totalFare}
            onBack={() => setBookingStep('SEATS')}
            onProceedToPayment={handleProceedToPayment}
          />
        )}

        {bookingStep === 'PAYMENT' && (
          <PaymentScreen
            bookingData={bookingData}
            onPaymentSuccess={handlePaymentSuccess}
          />
        )}

        {bookingStep === 'NONE' && (
          <>
            {activeTab === 'SEARCH' && <SearchScreen onSelectSchedule={handleSelectSchedule} />}
            {activeTab === 'TRACK' && <LiveTrackingScreen onBack={() => setActiveTab('TICKETS')} />}
            {activeTab === 'TICKETS' && (
              <MyTicketsScreen onTrackLive={() => setActiveTab('TRACK')} />
            )}
          </>
        )}
      </View>

      {/* Profile & Navigation Menu Modal */}
      <ProfileMenuModal
        visible={profileMenuOpen}
        onClose={() => setProfileMenuOpen(false)}
        onNavigate={(tab) => {
          setBookingStep('NONE');
          setSelectedSchedule(null);
          setActiveTab(tab);
        }}
      />
    </View>
  );
}

export default function App() {
  return (
    <ThemeProvider>
      <MainAppShell />
    </ThemeProvider>
  );
}
