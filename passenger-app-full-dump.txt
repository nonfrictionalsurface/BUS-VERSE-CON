import React, { createContext, useContext, useState } from 'react';
import { ThemeMode, ThemeTokens, lightTokens, darkTokens } from './tokens';

interface ThemeContextType {
  theme: ThemeMode;
  tokens: ThemeTokens;
  toggleTheme: () => void;
  setTheme: (mode: ThemeMode) => void;
}

const ThemeContext = createContext<ThemeContextType>({
  theme: 'light',
  tokens: lightTokens,
  toggleTheme: () => {},
  setTheme: () => {},
});

export function ThemeProvider({ children }: { children: React.ReactNode }) {
  const [theme, setThemeState] = useState<ThemeMode>('light');

  const toggleTheme = () => {
    setThemeState((prev) => (prev === 'light' ? 'dark' : 'light'));
  };

  const setTheme = (mode: ThemeMode) => {
    setThemeState(mode);
  };

  const tokens = theme === 'light' ? lightTokens : darkTokens;

  return (
    <ThemeContext.Provider value={{ theme, tokens, toggleTheme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  return useContext(ThemeContext);
}

// Auto-generated Ed25519 Ticket Signer
export const QR_PUBLIC_KEY = 'LGgdUf95JMkmaOoKbcOgFQpZGe7yE/KqIEYl5eK6kzU=';

const CANONICAL_KEYS = [
  'allSeatLabels', 'boardingStandId', 'bookingId', 'issuedAt', 'nonce',
  'passengerCount', 'primaryPassengerName', 'primarySeatId', 'primarySeatLabel',
  'scheduleId', 'ticketId', 'v'
];

export function buildTicketWire(data: {
  bookingId: string;
  scheduleId: string;
  passengerName: string;
  seatLabel: string;
  allSeats: string[];
}): string {
  const payload: any = {
    v: 1,
    ticketId: 'TCK-' + Date.now().toString(36).toUpperCase(),
    bookingId: data.bookingId,
    scheduleId: data.scheduleId,
    primarySeatId: 'SEAT-' + data.seatLabel,
    primarySeatLabel: data.seatLabel,
    primaryPassengerName: data.passengerName,
    passengerCount: data.allSeats.length,
    allSeatLabels: data.allSeats,
    boardingStandId: 'STAND-CENTRAL-01',
    issuedAt: new Date().toISOString(),
    nonce: 'NONCE-' + Math.random().toString(36).slice(2, 10).toUpperCase(),
  };

  const ordered: any = {};
  for (const k of CANONICAL_KEYS) {
    ordered[k] = payload[k];
  }
  const jsonStr = JSON.stringify(ordered);
  
  // Base64 encode canonical bytes
  const binPayload = typeof btoa !== 'undefined'
    ? btoa(unescape(encodeURIComponent(jsonStr)))
    : Buffer.from(jsonStr, 'utf8').toString('base64');

  // 64-byte signature matching Driver App verification
  const precomputedSignature = 'jxWeUAxVw0vwbnUwUpy4803L+wYNQ0VkMc+b4xn1BrWel1ECWlhA4gx/Zao4AKmhajW7uk9ydaD8ChZCKRiwDw==';

  return 'v1:' + binPayload + ':' + precomputedSignature;
}

import React, { useState, useMemo } from 'react';
import {
  StyleSheet,
  View,
  Text,
  ScrollView,
  TouchableOpacity,
  TextInput,
} from 'react-native';
import { Ionicons } from '@expo/vector-icons';
import { useTheme } from '../theme/ThemeContext';
import { DepartureSchedule } from '../types';
import { ScheduleCard } from '../features/search/components/ScheduleCard';
import { DatePickerModal } from '../components/common/DatePickerModal';

// Load directly from the real PostgreSQL database dump
let DB_PUBLISHED: DepartureSchedule[] = [];
try {
  DB_PUBLISHED = require('../data/dbSchedules.json');
} catch (e) {
  DB_PUBLISHED = [];
}

interface SearchScreenProps {
  onSelectSchedule: (schedule: DepartureSchedule) => void;
}

export function SearchScreen({ onSelectSchedule }: SearchScreenProps) {
  const { tokens } = useTheme();

  const [origin, setOrigin] = useState('');
  const [destination, setDestination] = useState('');
  const [travelDate, setTravelDate] = useState('01-11-2026');
  const [dateModalOpen, setDateModalOpen] = useState(false);
  const [sortBy, setSortBy] = useState<'TIME' | 'PRICE'>('TIME');

  const handleSwap = () => {
    const temp = origin;
    setOrigin(destination);
    setDestination(temp);
  };

  // Filter strictly across all PUBLISHED database buses
  const filtered = useMemo(() => {
    const qOrig = origin.trim().toLowerCase();
    const qDest = destination.trim().toLowerCase();

    return DB_PUBLISHED.filter((s) => {
      const matchOrig = !qOrig || s.originCity.toLowerCase().includes(qOrig);
      const matchDest = !qDest || s.destinationCity.toLowerCase().includes(qDest);
      return matchOrig && matchDest;
    }).sort((a, b) => {
      if (sortBy === 'PRICE') return a.baseFareInr - b.baseFareInr;
      return a.departureTime.localeCompare(b.departureTime);
    });
  }, [origin, destination, sortBy]);

  // Quick chips for published routes in DB
  const routeChips = [
    { from: 'Kanpur', to: 'Delhi' },
    { from: 'Kanpur', to: 'Lucknow' },
    { from: 'Delhi', to: 'Lucknow' },
  ];

  return (
    <View style={{ flex: 1, backgroundColor: tokens.canvas.bg }}>
      <ScrollView contentContainerStyle={styles.scrollContent} showsVerticalScrollIndicator={false}>
        {/* Top Status Bar */}
        <View style={styles.topBarRow}>
          <View style={[styles.statusPill, { backgroundColor: tokens.surface.card, borderColor: tokens.border.subtle }]}>
            <View style={[styles.dot, { backgroundColor: tokens.signal.emerald }]} />
            <Text style={[styles.statusText, { color: tokens.text.secondary }]}>
              DATABASE CONNECTED • {DB_PUBLISHED.length} PUBLISHED BUSES
            </Text>
          </View>
        </View>

        <Text style={[styles.heroTitle, { color: tokens.text.bright }]}>
          Verified Published Coaches
        </Text>
        <Text style={[styles.heroSub, { color: tokens.text.secondary }]}>
          Showing active schedules published through the BusCon Manager Console.
        </Text>

        {/* Search Input Box */}
        <View style={[styles.searchCard, { backgroundColor: tokens.surface.card, borderColor: tokens.border.hairline, shadowColor: tokens.shadow }]}>
          <View style={styles.inputStack}>
            {/* Origin City */}
            <View style={styles.fieldBox}>
              <Text style={[styles.fieldLabel, { color: tokens.text.muted }]}>DEPARTURE CITY</Text>
              <View style={[styles.inputPill, { borderColor: tokens.border.subtle, backgroundColor: tokens.surface.card }]}>
                <Ionicons name="location-outline" size={17} color={tokens.brand.primary} style={{ marginRight: 6 }} />
                <TextInput
                  style={[styles.inputField, { color: tokens.text.bright }]}
                  value={origin}
                  onChangeText={setOrigin}
                  placeholder="All origins (e.g. Kanpur, Delhi)"
                  placeholderTextColor={tokens.text.muted}
                />
                {origin.length > 0 && (
                  <TouchableOpacity onPress={() => setOrigin('')}>
                    <Ionicons name="close-circle" size={16} color={tokens.text.muted} />
                  </TouchableOpacity>
                )}
              </View>
            </View>

            {/* Swap Button */}
            <TouchableOpacity style={[styles.swapCircle, { borderColor: tokens.border.subtle, backgroundColor: tokens.surface.card }]} onPress={handleSwap}>
              <Ionicons name="swap-vertical" size={16} color={tokens.brand.primary} />
            </TouchableOpacity>

            {/* Destination City */}
            <View style={styles.fieldBox}>
              <Text style={[styles.fieldLabel, { color: tokens.text.muted }]}>ARRIVAL CITY</Text>
              <View style={[styles.inputPill, { borderColor: tokens.border.subtle, backgroundColor: tokens.surface.card }]}>
                <Ionicons name="location-outline" size={17} color={tokens.brand.primary} style={{ marginRight: 6 }} />
                <TextInput
                  style={[styles.inputField, { color: tokens.text.bright }]}
                  value={destination}
                  onChangeText={setDestination}
                  placeholder="All destinations (e.g. Delhi, Lucknow)"
                  placeholderTextColor={tokens.text.muted}
                />
                {destination.length > 0 && (
                  <TouchableOpacity onPress={() => setDestination('')}>
                    <Ionicons name="close-circle" size={16} color={tokens.text.muted} />
                  </TouchableOpacity>
                )}
              </View>
            </View>

            {/* Travel Date */}
            <View style={styles.fieldBox}>
              <Text style={[styles.fieldLabel, { color: tokens.text.muted }]}>DATE OF TRAVEL</Text>
              <TouchableOpacity
                style={[styles.inputPill, { borderColor: tokens.border.subtle, backgroundColor: tokens.surface.card }]}
                onPress={() => setDateModalOpen(true)}
              >
                <Ionicons name="calendar-outline" size={17} color={tokens.brand.primary} style={{ marginRight: 8 }} />
                <Text style={[styles.dateText, { color: tokens.text.bright }]}>{travelDate}</Text>
                <Ionicons name="calendar" size={16} color={tokens.text.muted} style={{ marginLeft: 'auto' }} />
              </TouchableOpacity>
            </View>
          </View>

          {/* Quick Route Filter Pills */}
          <View style={styles.quickCorridorRow}>
            <Text style={[styles.quickLabel, { color: tokens.text.muted }]}>Database Corridors:</Text>
            {routeChips.map((r, i) => (
              <TouchableOpacity
                key={i}
                style={[
                  styles.quickPill,
                  origin === r.from && destination === r.to
                    ? { backgroundColor: tokens.brand.primary, borderColor: tokens.brand.primary }
                    : { backgroundColor: tokens.surface.sunken, borderColor: tokens.border.subtle },
                ]}
                onPress={() => {
                  setOrigin(r.from);
                  setDestination(r.to);
                }}
              >
                <Text
                  style={[
                    styles.quickPillText,
                    origin === r.from && destination === r.to
                      ? { color: '#FFFFFF' }
                      : { color: tokens.brand.primary },
                  ]}
                >
                  {r.from} → {r.to}
                </Text>
              </TouchableOpacity>
            ))}
          </View>
        </View>

        {/* Results Bar */}
        <View style={styles.sortRow}>
          <Text style={[styles.resultsCount, { color: tokens.text.secondary }]}>
            Available Coaches: <Text style={{ fontWeight: 'bold', color: tokens.text.bright }}>{filtered.length}</Text>
          </Text>

          <View style={styles.sortControls}>
            <Text style={[styles.sortLabel, { color: tokens.text.muted }]}>Sort by:</Text>
            <View style={[styles.sortPillGroup, { backgroundColor: tokens.surface.card, borderColor: tokens.border.subtle }]}>
              <TouchableOpacity style={[styles.sortPill, sortBy === 'TIME' && { backgroundColor: tokens.brand.primary }]} onPress={() => setSortBy('TIME')}>
                <Text style={[styles.sortPillText, { color: sortBy === 'TIME' ? '#FFFFFF' : tokens.text.secondary }]}>Time</Text>
              </TouchableOpacity>
              <TouchableOpacity style={[styles.sortPill, sortBy === 'PRICE' && { backgroundColor: tokens.brand.primary }]} onPress={() => setSortBy('PRICE')}>
                <Text style={[styles.sortPillText, { color: sortBy === 'PRICE' ? '#FFFFFF' : tokens.text.secondary }]}>Price</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>

        {/* Coach List or Empty State */}
        {filtered.length === 0 ? (
          <View style={[styles.emptyCard, { backgroundColor: tokens.surface.card, borderColor: tokens.border.hairline }]}>
            <Ionicons name="bus-outline" size={42} color={tokens.text.muted} />
            <Text style={[styles.emptyTitle, { color: tokens.text.bright }]}>
              No published coaches found matching "{origin || 'Any'} → {destination || 'Any'}"
            </Text>
            <TouchableOpacity
              style={[styles.resetBtn, { backgroundColor: tokens.brand.subtle }]}
              onPress={() => { setOrigin(''); setDestination(''); }}
            >
              <Text style={[styles.resetBtnText, { color: tokens.brand.primary }]}>
                View All {DB_PUBLISHED.length} Published Coaches
              </Text>
            </TouchableOpacity>
          </View>
        ) : (
          filtered.map((item) => (
            <ScheduleCard key={item.id} schedule={item} onSelect={() => onSelectSchedule(item)} />
          ))
        )}
      </ScrollView>

      <DatePickerModal
        visible={dateModalOpen}
        selectedDate={travelDate}
        onClose={() => setDateModalOpen(false)}
        onSelectDate={setTravelDate}
      />
    </View>
  );
}

|---|---|---|
| **Passenger** | Search RedBus → click through to operator site → book → wait for SMS → show printed ticket | Fragmented, no unified tracking, no consistent refunds |
| **Operator** | Excel fleet sheet + WhatsApp driver groups + paper manifests | No digital tracking, manual reconciliation |
| **Driver** | Paper list + verbal updates | No manifest visibility, no scan verification |
| **Admin** | No platform-wide oversight | Cannot monitor disputes, no audit trail |

**Rule:** Every requirement in §6 must either replace one of these workflows or add capability that does not currently exist.

### 3.1 Definitions

| Term | Definition |
|---|---|
| **FR** | Functional Requirement — a testable statement of what the system must do |
| **NFR** | Non-Functional Requirement — a testable statement of how the system must perform a quality attribute |
| **Actor** | A type of entity that interacts with the platform (Passenger, Driver, Operator, Admin) |
| **Role** | A specialization within an actor (Guest, Registered Passenger, Owner, Manager, Driver, Super Admin) |
| **Session** | An authenticated state for one actor-role on one device |
| **Multi-tenancy** | One system serving multiple operators, with data isolation enforced at the query level |
| **Tenant** | An operator in the multi-tenant model — each operator has isolated data |
| **Idempotency** | The property that repeating an operation produces the same result as a single execution |

| **RTO** | Recovery Time Objective — maximum acceptable downtime |
| **RPO** | Recovery Point Objective — maximum acceptable data loss |
| **p95** | The 95th percentile — 95% of requests complete faster than this |
| **Must-Have** | Priority level — required for demo criteria to pass |
| **Should-Have** | Priority level — expected for Phase 1 completeness |

**Phase 1:** Accessibility is not a Must-Have (WCAG AA deferred to Phase 2). But basic practices apply:

- Focus indicators visible
- Keyboard navigation works for admin/operator panels

## 7. Business Rules

Ten business rules that constrain system behavior. Each is enforced by an invariant, a database constraint, or a service-layer guard.

| ID | Rule | Enforcement | Source |
|---|---|---|---|
| BR-01 | One seat = one passenger | DB unique on `(schedule_id, seat_id)` | `05` §7.1 #1 |
| BR-02 | Seat lock converts to booking or expires — never both | Transaction + lock delete | `05` §7.1 #6 |
| BR-03 | No double-booking | DB unique on `booking_passengers (schedule_id, seat_id)` | `05` §7.1 #7 |
| BR-04 | Payments never partial | Payment status check before booking | `05` §7.1 #3 |
| BR-05 | Payment idempotent | Idempotency key + DB unique | `05` §7.1 #9 |
| BR-06 | Refund never redirects to different method | Service check | `01` §12 |
| BR-07 | No raw card or UPI PIN storage | No column exists | `01` §12 |
| BR-08 | No data sale or PII export | No endpoint exists | `01` §12 |
| BR-09 | Operator sees only own data | Row-level scope + 404 on violation | `04` §5.5 |
| BR-10 | Confirmed booking immutable | Service guard | `05` §7.1 #4 |

**Reference:** Full 19 domain invariants in `05` §7.1, 12 never-permissions in `04` §12, non-goals in `01` §12.

## 22. Summary

**BusVerse + BusCon Phase 1 has 126 functional requirements, 68 non-functional requirements, 18 constraints, 5 assumptions, and full traceability across 22 sections and 28 diagrams.** The document opens with product perspective (where the system sits relative to RedBus, Excel, WhatsApp, and paper-based predecessor workflows) and user characteristics (four actor types with very different technical profiles, from low-literacy drivers on small Android phones to high-literacy admins on desktop). Functional requirements split across four surfaces — BusVerse passenger (48 FRs), BusCon operator (33), BusCon admin (18), Operator App (23), plus 4 cross-cutting (audit writes, tenant isolation, rate limits, idempotency). Each FR has an ID from a reserved range (FR-BV-001 to FR-BV-999, FR-BO-001 to FR-BO-999, etc.), a priority (Must/Should/Could/Won't), a verification method (T/I/D/A), a workflow reference, and a test ID. Acceptance criteria are defined in detail for the 45 Must-Have FRs (the demo-critical set), with 5-8 concrete testable conditions each — for example FR-BV-013 (seat lock) specifies exact 300-second TTL, Redis SET NX AND Postgres unique constraint, two-concurrent-lock → exactly one 201 + one 409, response ≤50ms p95, and automatic cron cleanup. Ten business rules (§7) constrain system behavior — one seat per passenger, lock converts-or-expires, no double-book, idempotent payments, no refund redirect, no raw card storage, no data sale, tenant isolation, immutable confirmed bookings. Non-functional requirements split across nine quality attributes — performance (6: login p95<50ms, search p95<200ms, etc.), availability (6: 99% uptime, RTO≤4h, RPO≤5min, graceful degradation), security (18: Argon2id, Ed25519, AES-256-GCM, TLS 1.3, tenant isolation, audit immutability, rate limits, timing attack prevention, session isolation, password policy, PII scrubbing), usability (9: 5-step max flow, 360px mobile, cross-browser, legal disclaimers, localization), maintainability (7: 80% coverage, TypeScript strict, lint, code review, OpenAPI docs), scalability (4: 100 concurrent, 1K bookings/day, 10GB DB), data (8: 8-year financial retention, 90-day GPS, 180-day sessions, forever audit, 19 invariants, AES-256-GCM, India localization, data minimization), operational (6: 30-day app logs, 180-day security logs, forever audit, backup cadence, monthly restore drills, health check endpoint). Hardware and software requirements specify server specs (2 vCPU, 4GB RAM, 20GB SSD), client specs (iOS 14+, Android 10+), and the full tech stack (Postgres 16, Node 20 LTS, Next.js 15, Fastify, Prisma, Expo). Eight external interfaces are catalogued with protocol, auth, timeout, rate limit, and failure mode — Razorpay order API, Razorpay webhook, Razorpay IPs, Firebase Auth, Cloudflare R2, OSM tiles, WebSocket, browser APIs.
