const POLICY_VERSION = 'v1.0';
const CONSENT_PURPOSES = ['terms_of_service', 'privacy_policy'] as const;

export async function captureSignupConsent(params: {
  userId: string;
  ipAddress?: string | null;
}): Promise<void> {
  await prisma.consentRecord.createMany({
    data: CONSENT_PURPOSES.map((purpose) => ({

    userId: params.userId,

  });
}

function extractRefreshToken(req: FastifyRequest): string {
  const body = req.body as { refreshToken?: string } | undefined;
  if (body && typeof body.refreshToken === 'string') return body.refreshToken;

  const header = req.headers.authorization;
  if (header && header.startsWith('Bearer ')) return header.slice(7).trim();

  throw errors.unauthorized('Missing refresh token');
}

export async function postLogout(req: FastifyRequest, reply: FastifyReply) {
  surfaceParamSchema.parse(req.params);
  if (!req.user) throw errors.unauthorized();

  const header = req.headers.authorization;
  let ttlSeconds = 300;
  if (header && header.startsWith('Bearer ')) {
    try {
      const claims = verifyAccessToken(header.slice(7));
      ttlSeconds = Math.max(1, claims.exp - Math.floor(Date.now() / 1000));
    } catch {
      /* ignore */
    }
  }

  await service.logout(req.user.sessionId, req.user.userId);
  await redis.set(keys.revokedSession(req.user.sessionId), '1', 'EX', ttlSeconds);

  return reply.status(204).send();
}

/**
 * Plugin-scoped error handler. Guarantees DomainErrors map to the right
 * HTTP status for identity routes, regardless of ordering issues in the
 * parent app's `setErrorHandler` registration.
 */
function isDomainError(
  err: unknown,
): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return (
    typeof e.code === 'string' &&
    typeof e.httpStatus === 'number' &&
    typeof e.message === 'string'
  );
}

function isZodError(err: unknown): err is { issues: unknown[]; message: string } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return e.name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function identityRouter(server: FastifyInstance): Promise<void> {
  // Scoped error handler — takes precedence for routes in this plugin
  server.setErrorHandler((error, request, reply) => {
    if (process.env.NODE_ENV === 'test') {
      console.log('[err-handler-ROUTER]', {
        url: request.url,
        errName: (error as { name?: string })?.name,
        errCtor: (error as { constructor?: { name?: string } })?.constructor?.name,
        code: (error as { code?: unknown })?.code,
        httpStatus: (error as { httpStatus?: unknown })?.httpStatus,
        hasIssues: Array.isArray((error as { issues?: unknown })?.issues),
        isDomain: isDomainError(error),
        isZod: isZodError(error),
      });
    }

    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({
        error: {
          code: error.code,
          message: error.message,
          ...(error.details ? { details: error.details } : {}),
          requestId: request.id,
        },
      });
    }

    if (isZodError(error)) {
      return reply.status(422).send({
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Validation failed',
          details: { issues: error.issues },
          requestId: request.id,
        },
      });
    }

    const anyErr = error as { statusCode?: number; code?: string; message?: string };
    const statusCode = anyErr.statusCode ?? 500;
    const isProd = process.env.NODE_ENV === 'production';
    return reply.status(statusCode).send({
      error: {
        code: anyErr.code ?? 'INTERNAL_ERROR',
        message: isProd ? 'Something went wrong' : (anyErr.message ?? 'Unknown error'),
        requestId: request.id,
      },
    });
  });

  // ── Routes ──
  server.post('/passenger/auth/otp/request', c.postPassengerOtpRequest);
  server.post('/passenger/auth/otp/verify', c.postPassengerOtpVerify);
  server.get('/passenger/me', { preHandler: requireRole('REGISTERED_PASSENGER') }, getPassengerMe);

  server.post('/driver/auth/otp/request', c.postDriverOtpRequest);
  server.post('/driver/auth/otp/verify', c.postDriverOtpVerify);

  server.post('/operator/auth/signup', c.postOperatorSignup);
  server.post('/operator/auth/login', c.postOperatorLogin);
  server.post('/operator/auth/otp/request', c.postOperatorOtpRequest);
  server.post('/operator/auth/otp/verify', c.postOperatorOtpVerify);
  server.post('/operator/auth/reset/request', c.postOperatorResetRequest);
  server.post('/operator/auth/reset/verify', c.postOperatorResetVerify);

  server.post('/admin/auth/login', c.postAdminLogin);
  server.post('/admin/auth/reset/request', c.postAdminResetRequest);
  server.post('/admin/auth/reset/verify', c.postAdminResetVerify);

  server.post('/:surface/auth/refresh', c.postRefresh);
  server.post('/:surface/auth/logout', c.postLogout);
}

export async function passengerOtpRequest(params: {
  phone?: string | undefined;
  email?: string | undefined;
  intent: 'LOGIN' | 'SIGNUP';
  ip?: string | undefined;
}): Promise<{ expiresInSeconds: number }> {
  const { identifier, channel } = resolveOtpIdentifier(params);

  if (params.intent === 'SIGNUP' && channel === 'EMAIL') {
    throw errors.emailSignupNotSupported();
  }

  if (params.intent === 'SIGNUP' && channel === 'PHONE') {
    const existing = await findUserByPhone(identifier);
    if (existing) throw errors.phoneAlreadyRegistered();
  }

  const result = await createOtpChallenge({ identifier, channel, purpose: 'LOGIN' });
  return { expiresInSeconds: result.expiresInSeconds };
}

export async function passengerOtpVerify(params: {
  phone?: string | undefined;
  email?: string | undefined;
  otp: string;
  ip?: string | undefined;
  userAgent?: string | undefined;
}): Promise<LoginResult> {
  const { identifier, channel } = resolveOtpIdentifier(params);

  await consumeOtpChallenge({ identifier, otp: params.otp, purpose: 'LOGIN' });

  let user =
    channel === 'PHONE' ? await findUserByPhone(identifier) : await findUserByEmail(identifier);

  if (!user) {
    if (channel === 'EMAIL') throw errors.invalidOtp();
    user = await createPassengerUser({ phone: identifier, ip: params.ip });
    emitIdentityEvent({ type: 'user_registered', userId: user.id });
  }

  const sessionId = newUuid();
  const tokens = await issueSessionTokens({
    userId: user.id,
    role: 'REGISTERED_PASSENGER',
    sessionId,
  });

  emitIdentityEvent({
    type: 'session_created',
    userId: user.id,
    sessionId,
    ip: params.ip,
    userAgent: params.userAgent,
    metadata: { role: 'REGISTERED_PASSENGER' },
  });

  return { ...tokens, user: toUserDTO(user) };
}

export async function driverOtpVerify(params: {
  phone: string;
  otp: string;
  deviceFingerprint: string;
  ip?: string | undefined;
  userAgent?: string | undefined;
}): Promise<LoginResult> {
  const { identifier } = resolveOtpIdentifier({ phone: params.phone });

  await consumeOtpChallenge({ identifier, otp: params.otp, purpose: 'LOGIN' });

  const user = await findUserByPhone(identifier);
  if (!user) throw errors.invalidCredentials();

  const driver = await findDriverProfileByUser(user.id);
  if (!driver) throw errors.forbidden('Not a registered driver');
  if (driver.kycStatus !== 'VERIFIED') throw errors.forbidden('Driver KYC not verified');

  const sessionId = newUuid();
  const deviceFingerprintHash = hashDeviceFingerprint(params.deviceFingerprint);

  const tokens = await issueSessionTokens({
    userId: user.id,
    role: 'DRIVER',
    sessionId,
    deviceFingerprintHash,
  });

  if (!env.DEMO_MULTI_DEVICE_DRIVER) {
    const revoked = await revokeOtherDriverSessions(user.id, sessionId);
    if (revoked > 0) {
      emitIdentityEvent({
        type: 'single_device_revoked',
        userId: user.id,
        sessionId,
        metadata: { revokedCount: revoked },
      });
    }
  }

  emitIdentityEvent({
    type: 'session_created',
    userId: user.id,
    sessionId,
    ip: params.ip,
    userAgent: params.userAgent,
    metadata: { role: 'DRIVER' },
  });

  return { ...tokens, user: toUserDTO(user) };
}

  }

    type: 'operator_signup_submitted',
    userId: user.id,
    ip: params.ip,

    companyName: params.companyName,

import crypto from 'node:crypto';
import jwt from 'jsonwebtoken';
import { env } from '../../config/env.js';
import { prisma } from '../../lib/prisma.js';
import { sha256 } from '../../lib/crypto.js';
import { errors } from '../../lib/errors.js';
import {
  ACCESS_TTL_SECONDS,
  REFRESH_TTL_SECONDS,
  type Role,
} from './identity.types.js';

export interface AccessTokenClaims {
  sub: string; // userId
  sid: string; // sessionId
  role: Role;
  iat: number;
  exp: number;
}

export interface IssuedTokens {
  accessToken: string;
  refreshToken: string;
  expiresAt: string;
}

/**
 * Issue access (JWT) + refresh (opaque) tokens for a new session.
 * Creates the session and refresh-token chain in one transaction.
 */
export async function issueSessionTokens(params: {
  userId: string;
  role: Role;
  sessionId: string;
  deviceFingerprintHash?: string;
  activeTripId?: string;
}): Promise<IssuedTokens> {
  const { userId, role, sessionId, deviceFingerprintHash } = params;

  const accessTtl = ACCESS_TTL_SECONDS[role];
  const refreshTtl = REFRESH_TTL_SECONDS[role];

  const accessToken = jwt.sign(
    { sub: userId, sid: sessionId, role, jti: crypto.randomUUID() },
    env.JWT_SECRET,
    { algorithm: 'HS256', expiresIn: accessTtl },
  );

  const refreshRaw = crypto.randomBytes(32).toString('base64');
  const refreshHash = sha256(refreshRaw);

  const expiresAt = new Date(Date.now() + refreshTtl * 1000);

  await prisma.$transaction([
    prisma.session.create({
      data: {
        id: sessionId,
        userId,
        role,
        refreshHash,
        deviceFingerprintHash: deviceFingerprintHash ?? null,
        expiresAt,
        status: 'ACTIVE',
      },
    }),
    prisma.refreshTokenChain.create({
      data: { sessionId, usedHashes: [] },
    }),
  ]);

  return {
    accessToken,
    refreshToken: refreshRaw,
    expiresAt: expiresAt.toISOString(),
  };
}

/**
 * Verify an access token. Returns claims or throws UNAUTHORIZED.
 * Callers should also check session status (via `assertSessionActive`).
 */
export function verifyAccessToken(token: string): AccessTokenClaims {
  try {
    const decoded = jwt.verify(token, env.JWT_SECRET, { algorithms: ['HS256'] });
    return decoded as AccessTokenClaims;
  } catch {
    throw errors.unauthorized('Invalid or expired token');
  }
}

/**
 * Rotate refresh token. Detects replay and revokes the entire chain if
 * an already-used token is presented.
 */
export async function rotateRefreshToken(
  oldRefreshRaw: string,
): Promise<IssuedTokens> {
  const oldHash = sha256(oldRefreshRaw);

  // Case 1: current hash matches — normal rotation
  const currentSession = await prisma.session.findFirst({
    where: { refreshHash: oldHash, status: 'ACTIVE' },
  });

  if (currentSession) {
    const role = currentSession.role as Role;
    const refreshTtl = REFRESH_TTL_SECONDS[role];

    const newRefreshRaw = crypto.randomBytes(32).toString('base64');
    const newRefreshHash = sha256(newRefreshRaw);

    const expiresAt = new Date(Date.now() + refreshTtl * 1000);

    await prisma.$transaction([
      prisma.session.update({
        where: { id: currentSession.id },
        data: { refreshHash: newRefreshHash, expiresAt },
      }),
      prisma.$executeRaw`
        UPDATE refresh_token_chains
        SET used_hashes = used_hashes || ${JSON.stringify([oldHash])}::jsonb,
            updated_at = now(),
            version = version + 1
        WHERE session_id = ${currentSession.id}::uuid
      `,
    ]);

    const newAccessToken = jwt.sign(
      { sub: currentSession.userId, sid: currentSession.id, role, jti: crypto.randomUUID() },
      env.JWT_SECRET,
      { algorithm: 'HS256', expiresIn: ACCESS_TTL_SECONDS[role] },
    );

    return {
      accessToken: newAccessToken,
      refreshToken: newRefreshRaw,
      expiresAt: expiresAt.toISOString(),
    };
  }

  // Case 2: hash is in used_hashes for an active session — REPLAY detected
  const replayRows = await prisma.$queryRaw<Array<{ id: string }>>`
    SELECT s.id
    FROM sessions s
    JOIN refresh_token_chains c ON c.session_id = s.id
    WHERE s.status = 'ACTIVE'
      AND c.used_hashes @> ${JSON.stringify([oldHash])}::jsonb
    LIMIT 1
  `;

  if (replayRows.length > 0 && replayRows[0]) {
    const replayedSessionId = replayRows[0].id;
    await prisma.session.update({
      where: { id: replayedSessionId },
      data: {
        status: 'REVOKED',
        revokedAt: new Date(),
        revokeReason: 'REFRESH_REPLAY',
      },
    });
    throw errors.refreshReplayed();
  }

  // Case 3: unknown token — invalid
  throw errors.unauthorized('Invalid refresh token');
}

/**
 * Revoke a session by id. Idempotent.
 */
export async function revokeSession(
  sessionId: string,
  reason = 'USER_LOGOUT',
): Promise<void> {
  await prisma.session.updateMany({
    where: { id: sessionId, status: 'ACTIVE' },
    data: { status: 'REVOKED', revokedAt: new Date(), revokeReason: reason },
  });
}

/**
 * Revoke all sessions of a given actor. Used on password reset and
 * on cascade revocation (Phase 2b calls this).
 */
export async function revokeAllSessionsForUser(
  userId: string,
  reason = 'PASSWORD_CHANGED',
): Promise<number> {
  const result = await prisma.session.updateMany({
    where: { userId, status: 'ACTIVE' },
    data: { status: 'REVOKED', revokedAt: new Date(), revokeReason: reason },
  });
  return result.count;
}

/**
 * Revoke all DRIVER sessions for a driver userId EXCEPT the given session.
 * Used for single-device binding (per `04` §7.3).
 */
export async function revokeOtherDriverSessions(

import crypto from 'node:crypto';
import { env } from '../../config/env.js';
import { prisma } from '../../lib/prisma.js';
import { redis, keys } from '../../lib/redis.js';
import { sha256, timingSafeEqualString } from '../../lib/crypto.js';

import { smsAdapter } from '../../integrations/msg91.js';
import { emailAdapter } from '../../integrations/sendgrid.js';
import { isValidEmail, normalizeEmail } from '../../lib/email.js';
import { isValidPhone, normalizePhone } from '../../lib/phone.js';
import {
  OTP_MAX_ATTEMPTS,
  OTP_REQUEST_MAX,
  OTP_REQUEST_WINDOW_SECONDS,
  OTP_TTL_SECONDS,
} from './identity.types.js';

export type OtpPurpose = 'LOGIN' | 'RESET';

export interface GeneratedOtp {
  raw: string;
  hash: string;
  salt: string;
}

export function generateOtp(): GeneratedOtp {
  const raw = env.MOCK_OTP ? '123456' : String(crypto.randomInt(100000, 1000000));
  const salt = crypto.randomBytes(16).toString('base64');
  const hash = sha256(raw, salt);
  return { raw, hash, salt };
}

export function verifyOtp(raw: string, hash: string, salt: string): boolean {
  const computed = sha256(raw, salt);
  return timingSafeEqualString(computed, hash);
}

async function enforceOtpRequestRate(identifier: string): Promise<void> {
  const key = keys.rateLimit('otp-request', identifier);
  const count = await redis.incr(key);
  if (count === 1) await redis.expire(key, OTP_REQUEST_WINDOW_SECONDS);
  if (count > OTP_REQUEST_MAX) {
    throw errors.rateLimited(
      `Too many OTP requests. Try again in ${Math.ceil(OTP_REQUEST_WINDOW_SECONDS / 60)} minutes.`,
    );
  }
}

export async function enforceOtpVerifyRate(identifier: string): Promise<void> {
  const key = keys.rateLimit('otp-verify', identifier);
  const count = await redis.incr(key);
  if (count === 1) await redis.expire(key, OTP_REQUEST_WINDOW_SECONDS);
  if (count > OTP_MAX_ATTEMPTS) {
    throw errors.otpAttemptsExceeded();
  }
}

async function deliverOtp(
  identifier: string,
  channel: 'PHONE' | 'EMAIL',
  rawOtp: string,
  purpose: OtpPurpose,
): Promise<void> {
  const label = purpose === 'RESET' ? 'password reset' : 'login';
  if (channel === 'PHONE') {
    await smsAdapter.send({
      phone: identifier,
      body: `Your BusVerse ${label} OTP is ${rawOtp}. Valid for 10 minutes. Do not share.`,
      template: 'otp',
    });
  } else {
    await emailAdapter.send({
      to: identifier,
      subject: `BusVerse ${label} code`,
      body: `Your ${label} code is ${rawOtp}. Valid for 10 minutes. Do not share.`,
      template: 'otp',
    });
  }
}

export async function createOtpChallenge(params: {
  identifier: string;
  channel: 'PHONE' | 'EMAIL';
  purpose: OtpPurpose;
}): Promise<{ expiresInSeconds: number; rawOtpForDev?: string | undefined }> {
  const { identifier, channel, purpose } = params;

  await enforceOtpRequestRate(`${purpose}:${identifier}`);

  const { raw, hash, salt } = generateOtp();
  const expiresAt = new Date(Date.now() + OTP_TTL_SECONDS * 1000);

  await prisma.otpChallenge.create({
    data: {
      phoneOrEmail: identifier,
      otpHash: hash,
      otpSalt: salt,
      expiresAt,
      purpose,
    },
  });

  await deliverOtp(identifier, channel, raw, purpose);

  if (env.MOCK_OTP) {
    return { expiresInSeconds: OTP_TTL_SECONDS, rawOtpForDev: raw };
  }
  return { expiresInSeconds: OTP_TTL_SECONDS };
}

export async function consumeOtpChallenge(params: {
  identifier: string;
  otp: string;
  purpose: OtpPurpose;
}): Promise<{ challengeId: string }> {
  const { identifier, otp, purpose } = params;

  await enforceOtpVerifyRate(`${purpose}:${identifier}`);

  const challenge = await prisma.otpChallenge.findFirst({
    where: {
      phoneOrEmail: identifier,
      purpose,
      consumedAt: null,
      deletedAt: null,
      expiresAt: { gt: new Date() },
    },
    orderBy: { createdAt: 'desc' },
  });

  if (!challenge) throw errors.otpExpired();
  if (challenge.attempts >= OTP_MAX_ATTEMPTS) throw errors.otpAttemptsExceeded();

  if (!verifyOtp(otp, challenge.otpHash, challenge.otpSalt)) {
    await prisma.otpChallenge.update({
      where: { id: challenge.id },
      data: { attempts: { increment: 1 } },
    });
    throw errors.invalidOtp();
  }

  await prisma.otpChallenge.update({
    where: { id: challenge.id },
    data: { consumedAt: new Date() },
  });

  return { challengeId: challenge.id };
}

export function resolveOtpIdentifier(params: {
  phone?: string | undefined;
  email?: string | undefined;
}): { identifier: string; channel: 'PHONE' | 'EMAIL' } {
  if (params.phone && params.email) {
    throw errors.invalidRequest('Provide exactly one of phone or email');
  }
  if (params.phone) {
    const normalized = normalizePhone(params.phone);
    if (!isValidPhone(normalized)) throw errors.invalidPhoneFormat();
    return { identifier: normalized, channel: 'PHONE' };
  }
  if (params.email) {
    const normalized = normalizeEmail(params.email);
    if (!isValidEmail(normalized)) throw errors.invalidEmailFormat();
    return { identifier: normalized, channel: 'EMAIL' };
  }
  throw errors.invalidRequest('Provide either phone or email');
}

export async function completeEmailReset(
  rawToken: string,
  newPassword: string,
): Promise<{ reset: true; sessionsRevoked: number }> {
  const tokenHash = sha256(rawToken);

  const row = await prisma.passwordResetToken.findUnique({ where: { tokenHash } });
  if (!row || row.usedAt || row.deletedAt) throw errors.resetTokenInvalid();
  if (row.expiresAt < new Date()) throw errors.resetTokenExpired();

  const passwordHash = await hashPassword(newPassword);

  const [tokenUpdate, operatorUpdate, adminUpdate] = await prisma.$transaction([
    prisma.passwordResetToken.updateMany({
      where: { id: row.id, usedAt: null },
      data: { usedAt: new Date() },
    }),
    prisma.operatorUser.updateMany({
      where: { userId: row.userId, deletedAt: null },
      data: { passwordHash },
    }),
    prisma.adminUser.updateMany({
      where: { userId: row.userId, deletedAt: null },
      data: { passwordHash },
    }),
  ]);

  if (tokenUpdate.count === 0) throw errors.resetTokenInvalid();
  if (operatorUpdate.count === 0 && adminUpdate.count === 0) {
    throw errors.resetTokenInvalid();
  }

  const sessionsRevoked = await revokeAllSessionsForUser(row.userId, 'PASSWORD_RESET');

  emitIdentityEvent({
    type: 'password_reset_completed',
    userId: row.userId,
    metadata: { channel: 'EMAIL', sessionsRevoked },
  });

  return { reset: true, sessionsRevoked };
}

function isDomainError(err: unknown): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return typeof e.code === 'string' && typeof e.httpStatus === 'number' && typeof e.message === 'string';
}
function isZodError(err: unknown): err is { issues: unknown[]; message: string } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return e.name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function networkRouter(server: FastifyInstance): Promise<void> {
  server.setErrorHandler((error, request, reply) => {
    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({ error: { code: error.code, message: error.message, ...(error.details ? { details: error.details } : {}), requestId: request.id } });

interface RouteStandInput {
  standId: string; sequence: number; distanceFromPrevKm: number; durationFromPrevMin: number;
}
async function validateRouteStands(params: { originCityId: string; destinationCityId: string; stands: RouteStandInput[] }): Promise<void> {
  const { originCityId, destinationCityId, stands } = params;
  if (originCityId === destinationCityId) throw networkErrors.routeSameCity();
  if (stands.length < 2) throw networkErrors.routeStandsInsufficient();
  const sorted = [...stands].sort((a, b) => a.sequence - b.sequence);
  for (let i = 0; i < sorted.length; i++) {
    if (sorted[i]!.sequence !== i + 1) throw networkErrors.routeStandSequenceInvalid(`expected ${i + 1}, got ${sorted[i]!.sequence}`);
  }
  const seen = new Set<string>();
  for (const s of stands) {
    if (seen.has(s.standId)) throw networkErrors.routeStandDuplicate();
    seen.add(s.standId);
  }
  const first = await repo.findStandById(sorted[0]!.standId);
  const last = await repo.findStandById(sorted[sorted.length - 1]!.standId);
  if (!first) throw networkErrors.standNotFound();
  if (!last) throw networkErrors.standNotFound();
  if (first.cityId !== originCityId) throw networkErrors.routeStandCityMismatch('first stand must be in origin city');
  if (last.cityId !== destinationCityId) throw networkErrors.routeStandCityMismatch('last stand must be in destination city');
  for (const s of stands) {
    const st = await repo.findStandById(s.standId);
    if (!st) throw networkErrors.standNotFound();
  }
}

  const opUser = await requireOperator(p.userId);

      distanceFromPrevKm: p.distanceFromPrevKm, durationFromPrevMin: p.durationFromPrevMin,

    repo.bookedSeatCountsForSchedules(scheduleIds),
    repo.activeSeatLockCountsForSchedules(scheduleIds),

export async function getRevenue(p: { userId: string; from: string; to: string; groupBy: 'day' | 'week' | 'month'; }): Promise<RevenueDTO> {
  const opUser = await requireOperator(p.userId);
  const fromDate = istDayBoundsUtc(p.from).from;
  const toDate = istDayBoundsUtc(p.to).to;
  const payments = await repo.sumPaymentsForRevenue({ operatorId: opUser.operatorId, from: fromDate, to: toDate });
  const refunds = await repo.sumRefundsForRevenue({ operatorId: opUser.operatorId, from: fromDate, to: toDate });
  function istDayKey(d: Date): string {
    return new Date(d.getTime() + 5.5 * 3600 * 1000).toISOString().slice(0, 10);
  }
  const bucketMap = new Map<string, { revenuePaise: number; refundsPaise: number; bookings: Set<string> }>();
  for (const r of payments) {
    if (!r.succeededAt) continue;
    const k = istDayKey(r.succeededAt);
    if (!bucketMap.has(k)) bucketMap.set(k, { revenuePaise: 0, refundsPaise: 0, bookings: new Set() });
    const b = bucketMap.get(k)!;
    b.revenuePaise += Number(r.amountPaise);
    b.bookings.add(r.bookingId);
  }
  for (const r of refunds) {
    const k = istDayKey(r.createdAt);
    if (!bucketMap.has(k)) bucketMap.set(k, { revenuePaise: 0, refundsPaise: 0, bookings: new Set() });
    bucketMap.get(k)!.refundsPaise += Number(r.amountPaise);
  }
  const buckets = [...bucketMap.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([date, v]) => ({
    date, revenuePaise: v.revenuePaise, refundsPaise: v.refundsPaise, bookingCount: v.bookings.size,
  }));
  const totalRevenuePaise = buckets.reduce((s, b) => s + b.revenuePaise, 0);
  const totalRefundsPaise = buckets.reduce((s, b) => s + b.refundsPaise, 0);
  return { from: p.from, to: p.to, groupBy: p.groupBy, totalRevenuePaise, totalRefundsPaise, netRevenuePaise: totalRevenuePaise - totalRefundsPaise, buckets };
}
export async function getRevenueDaily(p: { userId: string }): Promise<RevenueDailyDTO> {
  const opUser = await requireOperator(p.userId);
  const days: Array<{ date: string; revenuePaise: number; bookingCount: number }> = [];
  for (let i = 6; i >= 0; i--) {
    const d = new Date(Date.now() - i * 24 * 3600 * 1000 + 5.5 * 3600 * 1000);
    const dayStr = d.toISOString().slice(0, 10);
    const { from, to } = istDayBoundsUtc(dayStr);
    const payments = await repo.sumPaymentsForRevenue({ operatorId: opUser.operatorId, from, to });
    const revenuePaise = payments.reduce((s, p) => s + Number(p.amountPaise), 0);
    const bookingCount = new Set(payments.map((p) => p.bookingId)).size;
    days.push({ date: dayStr, revenuePaise, bookingCount });
  }
  return { days, timezone: 'Asia/Kolkata' };
}

export async function assignTrip(p: {
  userId: string;
  scheduleId: string;
  driverId: string;
}): Promise<OperatorTripDTO> {
  const opUser = await requireOperator(p.userId);

  // 1. Schedule ownership + status + future departure
  const schedule = await repo.findScheduleForTripAssignment(p.scheduleId, opUser.operatorId);
  if (!schedule) throw networkErrors.scheduleNotFound();
  if (schedule.status !== 'PUBLISHED') {
    throw errors.invalidSchedule();
  }
  if (schedule.departureTime.getTime() <= Date.now()) {
    throw errors.invalidSchedule();
  }

  // 2. Driver ownership + KYC + status
  const driver = await repo.findDriverForTripAssignment(p.driverId, opUser.operatorId);
  if (!driver) throw networkErrors.driverNotFound();
  if (driver.kycStatus === 'BLOCKED' || driver.kycStatus === 'DEACTIVATED') {
    throw errors.driverBlocked();
  }
  if (driver.kycStatus !== 'VERIFIED' && driver.kycStatus !== 'ACTIVE') {
    throw networkErrors.driverKycNotVerified();
  }

  // 3. No duplicate active trip for this schedule
  const existingTrip = await repo.findActiveTripForSchedule(p.scheduleId);
  if (existingTrip) {
    throw errors.invalidTransition('A trip is already assigned to this schedule');
  }

  // 4. No overlapping driver trip
  const driverConflict = await repo.findOverlappingDriverTrip({
    driverId: p.driverId,
    departureTime: schedule.departureTime,
    arrivalTime: schedule.arrivalTime,
  });
  if (driverConflict) {
    throw errors.activeTripExists(driverConflict.id);
  }

  // 5. Create
  try {
    const trip = await repo.createOperatorTrip({
      scheduleId: p.scheduleId,
      busId: schedule.busId,
      driverId: p.driverId,
    });
    return toOperatorTripDTO(trip);
  } catch (err) {
    rethrowPrismaError(err);
  }
}

export interface SeatInput {
  seatRow: string;
  seatColumn: number;
  seatType: SeatType;
  isFemaleOnly: boolean;
}

export interface PutSeatLayoutInput {
  rowsCount: number;
  columnsCount: number;
  seats: SeatInput[];
}

/**
 * Replace the seat layout of a bus. Fails if the bus has any bookings
 * (to preserve booking_passengers FK integrity).
 */
export async function replaceSeatLayout(params: {
  busId: string;
  busCapacity: number;
  input: PutSeatLayoutInput;
}) {
  const { busId, busCapacity, input } = params;

  // 1. Cannot change layout if the bus has any bookings
  const hasBookings = await (await import('./network.repository.js')).busLayoutHasBookings(busId);
  if (hasBookings) throw networkErrors.layoutHasBookings();

  // 2. Validate seat count vs capacity
  if (input.seats.length !== busCapacity) {
    throw networkErrors.capacityMismatch(input.seats.length, busCapacity);
  }

  // 3. Validate each seat within grid bounds
  const maxRowIdx = input.rowsCount; // 1..rowsCount → A..[A+rowsCount-1]
  const maxRowChar = String.fromCharCode('A'.charCodeAt(0) + maxRowIdx - 1);
  for (const s of input.seats) {
    if (s.seatRow < 'A' || s.seatRow > maxRowChar) {
      throw networkErrors.invalidSeatCoordinates(`Row ${s.seatRow} outside grid (A..${maxRowChar})`);
    }
    if (s.seatColumn < 1 || s.seatColumn > input.columnsCount) {
      throw networkErrors.invalidSeatCoordinates(
        `Column ${s.seatColumn} outside grid (1..${input.columnsCount})`,
      );
    }
  }

  // 4. Validate no duplicate (row, column)
  const occupied = new Set<string>();
  for (const s of input.seats) {
    const key = `${s.seatRow}${s.seatColumn}`;
    if (occupied.has(key)) throw networkErrors.duplicateSeatCoordinates(key);
    occupied.add(key);
  }

  // 5. Compute aisle_cells = all grid cells minus seats
  const aisleCells: Array<{ row: string; column: number }> = [];
  for (let r = 0; r < input.rowsCount; r++) {
    const rowChar = String.fromCharCode('A'.charCodeAt(0) + r);
    for (let c = 1; c <= input.columnsCount; c++) {
      if (!occupied.has(`${rowChar}${c}`)) {
        aisleCells.push({ row: rowChar, column: c });
      }
    }
  }

  // 6. Upsert layout + replace seats in a single transaction
  return prisma.$transaction(async (tx) => {
    let layout = await tx.seatLayout.findFirst({ where: { busId, deletedAt: null } });

    if (layout) {
      // Delete existing seats (safe — no bookings by step 1)
      await tx.seat.deleteMany({ where: { layoutId: layout.id } });
      layout = await tx.seatLayout.update({
        where: { id: layout.id },
        data: {
          rowsCount: input.rowsCount,
          columnsCount: input.columnsCount,
          aisleMapJson: { _v: 1, aisle_cells: aisleCells },
        },
      });
    } else {
      layout = await tx.seatLayout.create({
        data: {
          id: newUuid(),
          busId,
          rowsCount: input.rowsCount,
          columnsCount: input.columnsCount,
          aisleMapJson: { _v: 1, aisle_cells: aisleCells },
        },
      });
    }

    // Create seats
    const createdSeats = await Promise.all(
      input.seats.map((s) =>
        tx.seat.create({
          data: {
            id: newUuid(),
            layoutId: layout!.id,
            seatRow: s.seatRow,
            seatColumn: s.seatColumn,
            seatLabel: `${s.seatRow}${s.seatColumn}`,
            seatType: s.seatType,
            isFemaleOnly: s.isFemaleOnly,
          },
        }),
      ),
    );

    return { layout, seats: createdSeats };
  });
}

const STALE_MS = 30_000;

async function readRedisSamples(tripId: string): Promise<NormalizedSample[]> {
  try {
    const raws = await redis.lrange(keys.gpsTrip(tripId), 0, -1);
    return raws.map((r) => JSON.parse(r) as NormalizedSample);
  } catch {
    return [];
  }

export async function getLiveTrip(userId: string, tripId: string): Promise<LiveTripDTO> {
  const trip = await repo.findTripForPassenger(userId, tripId);
  if (!trip) throw errors.notFound('Trip not found or not accessible');

  const samples = await readRedisSamples(tripId);
  const occupancy = await repo.findOccupancyForSchedule(trip.scheduleId);

  if (samples.length === 0) {
    return {
      tripId,
      status: trip.status as TripStatus,
      currentPosition: null,
      speedKmh: null,
      lastUpdatedAt: null,
      isStale: true,
      eta: [],
      occupancy,
      nextStandId: null,
    };
  }

  const latest = samples[0]!;
  const lastUpdatedAt = latest.receivedAt;
  const isStale = Date.now() - new Date(lastUpdatedAt).getTime() > STALE_MS;

  const routeStands = trip.schedule.route.stands.map((s) => ({
    standId: s.stand.id,
    standName: s.stand.name,
    sequence: s.sequence,
    distanceFromPrevKm: Number(s.distanceFromPrevKm),
  }));

  const ordered = samples
    .slice()
    .reverse()
    .map((s) => ({ lat: s.lat, lng: s.lng, recordedAt: s.recordedAt }));

  const etaResult = computeEta({ samples: ordered, routeStands, now: new Date() });

  const currentPosition: LiveTripPositionDTO = {
    lat: latest.lat,
    lng: latest.lng,
    recordedAt: latest.recordedAt,
  };
  const eta: LiveTripEtaDTO[] = etaResult.eta;

  return {
    tripId,
    status: trip.status as TripStatus,
    currentPosition,
    speedKmh: latest.serverSpeedKmh,
    lastUpdatedAt,
    isStale,
    eta,
    occupancy,
    nextStandId: etaResult.nextStandId,
  };
}

function isDomainError(err: unknown): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return typeof e.code === 'string' && typeof e.httpStatus === 'number' && typeof e.message === 'string';
}

function isZodError(err: unknown): err is { issues: unknown[] } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return (err as { name?: string }).name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function bookingRouter(server: FastifyInstance): Promise<void> {
  server.setErrorHandler((error, request, reply) => {
    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({ error: { code: error.code, message: error.message, ...(error.details ? { details: error.details } : {}), requestId: request.id } });
    }
    if (isZodError(error)) {
      return reply.status(422).send({ error: { code: 'VALIDATION_ERROR', message: 'Validation failed', details: { issues: error.issues }, requestId: request.id } });
    }
    const anyErr = error as { statusCode?: number; code?: string; message?: string };
    const statusCode = anyErr.statusCode ?? 500;
    const isProd = process.env.NODE_ENV === 'production';
    return reply.status(statusCode).send({ error: { code: anyErr.code ?? 'INTERNAL_ERROR', message: isProd ? 'Something went wrong' : (anyErr.message ?? 'Unknown error'), requestId: request.id } });
  });

  baseFare: bigint; taxPaise: bigint; feePaise: bigint; total: bigint;

  const baseFare = scheduleFarePaise * BigInt(count);

export async function createBooking(
  userId: string,
  payload: {
    lockId: string;
    boardingStandId: string;
    droppingStandId: string;
    passengers: Array<{ seatId: string; name: string; age?: number | undefined; gender?: string | undefined }>;
  },
): Promise<CreateBookingResultDTO> {
  // 1. Representative lock — must exist, belong to this user, still active
  const lock = await prisma.seatLock.findFirst({
    where: { id: payload.lockId, userId, expiresAt: { gt: new Date() }, deletedAt: null },
    select: { scheduleId: true },
  });
  if (!lock) throw bookingErrors.lockNotFound();

  // 2. Schedule guard — PUBLISHED, departure in future
  const schedule = await prisma.schedule.findFirst({
    where: { id: lock.scheduleId, deletedAt: null },
    select: { id: true, status: true, departureTime: true, fareAmountPaise: true, fareCurrency: true },
  });
  if (!schedule || schedule.status !== 'PUBLISHED') throw bookingErrors.invalidSchedule();
  if (schedule.departureTime.getTime() <= Date.now()) throw bookingErrors.scheduleDeparted();

  // 3. Stands exist
  const [boarding, dropping] = await Promise.all([
    prisma.stand.findFirst({ where: { id: payload.boardingStandId, deletedAt: null } }),
    prisma.stand.findFirst({ where: { id: payload.droppingStandId, deletedAt: null } }),
  ]);
  if (!boarding || !dropping) throw bookingErrors.invalidStand();

  // 4. Fare calculation
  const { baseFare, taxPaise, feePaise, total } = calcFare(schedule.fareAmountPaise, payload.passengers.length);
  const seatIds = payload.passengers.map((p) => p.seatId);
  const extendedExpiresAt = new Date(Date.now() + FIFTEEN_MIN_MS);

  // 5. Transaction: insert booking + passengers, extend locks (G1 Postgres side)
  try {
    const result = await prisma.$transaction(async (tx) => {
      const bookingReference = await nextBookingReference(tx);
      const booking = await tx.booking.create({
        data: {
          bookingReference,
          userId,
          scheduleId: schedule.id,
          boardingStandId: payload.boardingStandId,
          droppingStandId: payload.droppingStandId,
          totalFareAmount: total,
          totalFareCurrency: schedule.fareCurrency,
          fareBreakdownJson: {
            _v: 1,
            base_fare_paise: Number(baseFare),
            tax_paise: Number(taxPaise),
            convenience_fee_paise: Number(feePaise),
            total_paise: Number(total),
          },
          status: 'PENDING',
        },
        select: { id: true, bookingReference: true },
      });
      await tx.bookingPassenger.createMany({
        data: payload.passengers.map((p) => ({
          bookingId: booking.id,
          seatId: p.seatId,
          scheduleId: schedule.id,
          name: p.name,
          age: p.age ?? null,
          gender: p.gender ?? null,
        })),
      });
      const extended = await tx.seatLock.updateMany({
        where: { scheduleId: schedule.id, seatId: { in: seatIds }, userId, expiresAt: { gt: new Date() }, deletedAt: null },
        data: { expiresAt: extendedExpiresAt },
      });
      if (extended.count !== seatIds.length) throw bookingErrors.lockExpired();
      return { bookingId: booking.id, bookingReference: booking.bookingReference };
    });

    // 6. G1 Redis side — best-effort, Postgres already correct
    for (const seatId of seatIds) {
      await redis.expire(keys.seatLock(schedule.id, seatId), BOOKING_LOCK_TTL_SECONDS).catch(() => {});
    }

    return {
      bookingId: result.bookingId,
      bookingReference: result.bookingReference,
      status: 'PENDING' as BookingStatus,
      expiresAt: extendedExpiresAt.toISOString(),
      totalFarePaise: Number(total),
    };
  } catch (err) {
    if (isPrismaUniqueViolation(err)) throw bookingErrors.seatAlreadyBooked(seatIds[0]!);
    throw err;
  }
}
export async function listMyBookings(
  userId: string,
  params: { limit: number; cursor?: string | undefined; status?: string | undefined },
): Promise<{ data: BookingSummaryDTO[]; nextCursor: string | null }> {
  let cursorCreatedAt: Date | undefined;
  if (params.cursor) {
    try {
      const decoded = JSON.parse(Buffer.from(params.cursor, 'base64').toString('utf8')) as { createdAt: string };
      cursorCreatedAt = new Date(decoded.createdAt);
    } catch { throw bookingErrors.notFound(); }
  }

  const rows = await prisma.booking.findMany({
    where: {
      userId,
      deletedAt: null,
      ...(params.status ? { status: params.status } : {}),
      ...(cursorCreatedAt ? { createdAt: { lt: cursorCreatedAt } } : {}),
    },
    take: params.limit + 1,
    orderBy: [{ createdAt: 'desc' }, { id: 'desc' }],
    include: {
      passengers: { where: { deletedAt: null }, include: { seat: { select: { seatLabel: true } } } },
      schedule: { include: { route: { include: { originCity: true, destCity: true } } } },
    },
  });

  const hasMore = rows.length > params.limit;
  const data = hasMore ? rows.slice(0, params.limit) : rows;
  const nextCursor = hasMore
    ? Buffer.from(JSON.stringify({ createdAt: data[data.length - 1]!.createdAt.toISOString() })).toString('base64')
    : null;

  return {
    data: data.map((b) => ({
      bookingId: b.id,
      bookingReference: b.bookingReference,
      status: b.status as BookingStatus,
      scheduleId: b.scheduleId,
      departureTime: b.schedule.departureTime.toISOString(),

// Canonical refund tier calculator per shared/02 §10 + 07 §4.5
function calcRefundTier(params: {
  departureAt: Date;
  farePaise: bigint;
  taxPaise: bigint;
  feePaise: bigint;
  operatorCancelled: boolean;
}): { tier: RefundTier; fare: bigint; tax: bigint; fee: bigint; total: bigint; deadline: Date | null } {
  const now = Date.now();
  const t = params.departureAt.getTime() - now;

  if (params.operatorCancelled) {
    return { tier: 'TIER_FULL', fare: params.farePaise, tax: params.taxPaise, fee: params.feePaise,
      total: params.farePaise + params.taxPaise + params.feePaise, deadline: null };
  }
  if (t <= 0) {
    return { tier: 'TIER_NOT_ELIGIBLE', fare: 0n, tax: 0n, fee: 0n, total: 0n, deadline: null };
  }

  const H = 3_600_000;
  let farePct: bigint, feeKept: boolean, tier: RefundTier, deadlineMs: number | null;
  if (t >= 24 * H)      { farePct = 100n; feeKept = true; tier = 'TIER_FULL'; deadlineMs = params.departureAt.getTime() - 24 * H; }
  else if (t >= 12 * H) { farePct = 75n;  feeKept = true; tier = 'TIER_75';   deadlineMs = params.departureAt.getTime() - 12 * H; }
  else if (t >= 2 * H)  { farePct = 50n;  feeKept = true; tier = 'TIER_50';   deadlineMs = params.departureAt.getTime() - 2 * H; }
  else                  { farePct = 0n;   feeKept = true; tier = 'TIER_ZERO'; deadlineMs = params.departureAt.getTime(); }

  const fareRefund = (params.farePaise * farePct) / 100n;
  const taxRefund = (params.taxPaise * farePct) / 100n;
  const feeRefund = feeKept ? params.feePaise : 0n;
  return { tier, fare: fareRefund, tax: taxRefund, fee: feeRefund,
    total: fareRefund + taxRefund + feeRefund, deadline: deadlineMs !== null ? new Date(deadlineMs) : null };
}

export async function cancelBooking(
  userId: string,
  bookingId: string,
): Promise<{ bookingId: string; status: 'CANCELLED'; refund: { refundId: string; tier: RefundTier; amountPaise: number; status: string } }> {
  const b = await prisma.booking.findFirst({
    where: { id: bookingId, userId, deletedAt: null },
    include: { schedule: true, payments: { where: { deletedAt: null, status: 'SUCCEEDED' }, take: 1, orderBy: { createdAt: 'desc' } } },
  });
  if (!b) throw bookingErrors.notFound();
  if (b.status !== 'CONFIRMED' && b.status !== 'OPERATOR_CANCELLED') throw bookingErrors.bookingNotConfirmed();

  const payment = b.payments[0];
  if (!payment) throw bookingErrors.paymentNotFound();

  const fb = b.fareBreakdownJson as { base_fare_paise?: number; tax_paise?: number; convenience_fee_paise?: number };
  const result = calcRefundTier({
    departureAt: b.schedule.departureTime,
    farePaise: BigInt(fb.base_fare_paise ?? 0),
    taxPaise: BigInt(fb.tax_paise ?? 0),
    feePaise: BigInt(fb.convenience_fee_paise ?? 0),
    operatorCancelled: b.status === 'OPERATOR_CANCELLED',
  });

  const reason = b.status === 'OPERATOR_CANCELLED' ? 'OPERATOR_CANCELLED' : 'PASSENGER_CANCELLED';

  const refundId = await prisma.$transaction(async (tx) => {
    const refund = await tx.refund.create({
      data: {
        paymentId: payment.id,
        reason,
        refundTier: result.tier,
        amountPaise: result.total,
        status: 'INITIATED',
      },
      select: { id: true },
    });
    await tx.booking.update({
      where: { id: bookingId },
      data: { status: 'CANCELLED', cancelledAt: new Date() },
    });
    await tx.ticket.updateMany({
      where: { bookingId },
      data: { status: 'CANCELLED' },
    });
    await writeAuditEvent({
      tx, actorId: userId, sessionId: null,
      entityType: 'Booking', entityId: bookingId,
      action: 'booking_cancelled',
      metadata: { tier: result.tier, refundPaise: Number(result.total) },
    });
    return refund.id;
  });

  emitToBooking(bookingId, events.refundRequested({
    refundId,
    bookingId,
    amountPaise: Number(result.total),
    reason,
  }));
  emitToBooking(bookingId, events.bookingCancelled({
    bookingId,
    reason,
    refundId,
  }));

  // Phase 5: fire gateway refund async; webhook confirms state transition
  if (result.total > 0n && payment.gatewayPaymentId) {
    try {
      const gatewayRefund = await razorpayAdapter.createRefund({
        paymentId: payment.gatewayPaymentId,
        amountPaise: result.total,
        notes: { bookingId, refundTier: result.tier },
      });
      await prisma.refund.update({
        where: { id: refundId },
        data: { gatewayRefundId: gatewayRefund.refundId },
      });
    } catch (err) {
      logger.error(
        { err, refundId, bookingId, tier: result.tier },
        'razorpay refund failed (refund row stays INITIATED)',
      );
    }
  }

  return {
    bookingId,
    status: 'CANCELLED',
    refund: { refundId, tier: result.tier, amountPaise: Number(result.total), status: 'INITIATED' },
  };
}
/**
 * Handles a single PENDING booking timeout. Phase 8 cron calls this in a loop;
 * Phase 4 tests call it directly for TV-11.
 * Returns true if the booking transitioned PENDING → EXPIRED, false if not PENDING (idempotent).
 * Per shared/03 §6.2: also soft-deletes booking_passengers (G4 fix) so the seat is re-lockable.
 */
export async function expirePendingBooking(bookingId: string): Promise<boolean> {
  const b = await prisma.booking.findFirst({
    where: { id: bookingId, status: 'PENDING', deletedAt: null },
    include: { passengers: { where: { deletedAt: null }, select: { seatId: true } } },
  });
  if (!b) return false;

  const seatIds = b.passengers.map((p) => p.seatId);

  const didExpire = await prisma.$transaction(async (tx) => {
    const updated = await tx.booking.updateMany({
      where: { id: bookingId, status: 'PENDING' },
      data: { status: 'EXPIRED', expiredAt: new Date() },
    });
    if (updated.count === 0) return false;

    await tx.payment.updateMany({
      where: { bookingId, status: { in: ['INITIATED', 'PENDING'] } },
      data: { status: 'EXPIRED', expiredAt: new Date() },
    });
    await tx.seatLock.deleteMany({
      where: { scheduleId: b.scheduleId, seatId: { in: seatIds } },
    });
    // G4 — soft-delete passengers so pre-check releases the seat
    await tx.bookingPassenger.updateMany({
      where: { bookingId, deletedAt: null },
      data: { deletedAt: new Date() },
    });
    await writeAuditEvent({
      tx, actorId: null, sessionId: null,
      entityType: 'Booking', entityId: bookingId,
      action: 'booking_expired',
    });
    return true;
  });

  if (didExpire) {
    // G6 — Redis DEL after commit, best-effort
    for (const seatId of seatIds) {
      await redis.del(keys.seatLock(b.scheduleId, seatId)).catch(() => {});
      emitToTopic(`seat:availability:${b.scheduleId}`, events.seatReleased({
        seatId,
        reason: 'expired',
      }));
    }
  }
  if (didExpire) {
    emitToBooking(bookingId, events.bookingExpired({ bookingId }));
  }
  return didExpire;
}

export async function nextInvoiceNumber(tx: { $queryRawUnsafe: <T>(q: string) => Promise<T> }): Promise<string> {
  const now = new Date();
  const month = now.getMonth() + 1;
  const fullYear = now.getFullYear();
  const fyStart = month >= 4 ? fullYear : fullYear - 1;
  const fyEnd = fyStart + 1;
  const fyEndShort = fyEnd % 100;
  const seqName = `invoice_number_seq_${fyStart}_${String(fyEndShort).padStart(2, '0')}`;
  const rows = await tx.$queryRawUnsafe<Array<{ nextval: bigint }>>(`SELECT nextval('${seqName}'::regclass)`);
  const n = Number(rows[0]!.nextval);
  return `INV/${fyStart}-${String(fyEndShort).padStart(2, '0')}/${String(n).padStart(5, '0')}`;
}

export async function acquireSeatLock(

  params: { scheduleId: string; seatId: string },
): Promise<SeatLockDTO> {
  // Layer 0 — input validation
  const schedule = await repo.findScheduleForLock(params.scheduleId);
  if (!schedule || schedule.status !== 'PUBLISHED') throw bookingErrors.invalidSchedule();
  if (schedule.departureTime.getTime() <= Date.now()) throw bookingErrors.scheduleDeparted();

  const seat = await repo.findSeatForLock(params.seatId);
  if (!seat) throw bookingErrors.invalidSeat();
  if (seat.layout.busId !== schedule.busId) throw bookingErrors.invalidSeat();

  // G11 — idempotent same-user re-acquire
  const existing = await repo.findExistingActiveLock({
    userId,
    scheduleId: params.scheduleId,
    seatId: params.seatId,
  });
  if (existing) return mapLock(existing, seat.seatLabel, params.scheduleId, params.seatId);

  // Layer 1 — pre-check for active booking (G7)
  if (await repo.isSeatBooked(params.scheduleId, params.seatId)) {
    throw bookingErrors.seatAlreadyBooked(params.seatId);
  }

  // Layer 2 — Redis SET NX (degrades gracefully if Redis is down, §13.3)
  const redisKey = keys.seatLock(params.scheduleId, params.seatId);
  let redisAcquired = false;
  let redisError = false;
  try {
    const res = await redis.set(redisKey, userId, 'EX', LOCK_TTL_SECONDS, 'NX');
    redisAcquired = res === 'OK';
  } catch {
    redisError = true;
  }
  if (!redisAcquired && !redisError) throw bookingErrors.seatLocked(params.seatId);

  // Layer 3 — Postgres transaction
  let result: repo.AcquireResult;
  try {
    result = await repo.tryAcquireLock({
      scheduleId: params.scheduleId,
      seatId: params.seatId,
      userId,
    });
  } catch (err) {
    if (redisAcquired) await redis.del(redisKey).catch(() => {});
    throw mapDbConnectionError(err);
  }
  if (result.kind === 'booked') {
    if (redisAcquired) await redis.del(redisKey).catch(() => {});
    throw bookingErrors.seatAlreadyBooked(params.seatId);
  }
  if (result.kind === 'locked') {
    if (redisAcquired) await redis.del(redisKey).catch(() => {});
    throw bookingErrors.seatLocked(params.seatId);
  }

  // Best-effort repopulate Redis if it was down
  if (redisError) {
    await redis.set(redisKey, userId, 'EX', LOCK_TTL_SECONDS).catch(() => {});
  }

  const dto = mapLock(result.lock, seat.seatLabel, params.scheduleId, params.seatId);
  emitToTopic(`seat:availability:${params.scheduleId}`, events.seatLocked({
    seatId: params.seatId,
    lockedUntil: dto.expiresAt,
    lockedByMe: false,
  }));
  return dto;
}

import { Prisma } from '@prisma/client';
import { randomBytes } from 'node:crypto';
import { sign } from '../../lib/ed25519.js';
import { canonicalizeToBytes, buildWireFormat, MAX_PAYLOAD_BYTES, type QRPayloadV2 } from '../../lib/qr.js';
import { newUuid } from '../../lib/ids.js';

export interface IssueTicketInput {
  bookingId: string;
  scheduleId: string;
  boardingStandId: string;
  passengers: Array<{ seatId: string; seatLabel: string; name: string }>;
}

export async function issueTicket(
  tx: Prisma.TransactionClient,
  input: IssueTicketInput,
): Promise<{ ticketId: string; qrPayload: string; qrSignature: string; wireFormat: string }> {
  if (input.passengers.length === 0) throw new Error('Cannot issue ticket without passengers');
  if (input.passengers.length > 8) throw new Error('Too many passengers for one ticket');

  const sorted = [...input.passengers].sort((a, b) => a.seatLabel.localeCompare(b.seatLabel));
  const primary = sorted[0]!;

  const payload: QRPayloadV2 = {
    v: 1,
    ticketId: newUuid(),
    bookingId: input.bookingId,
    scheduleId: input.scheduleId,
    primarySeatId: primary.seatId,
    primarySeatLabel: primary.seatLabel,
    primaryPassengerName: primary.name,
    passengerCount: sorted.length,
    allSeatLabels: sorted.map((p) => p.seatLabel),
    boardingStandId: input.boardingStandId,
    issuedAt: new Date().toISOString(),
    nonce: randomBytes(16).toString('base64'),
  };

  const canonicalBytes = canonicalizeToBytes(payload);
  if (canonicalBytes.length > MAX_PAYLOAD_BYTES) {
    throw new Error(`QR payload ${canonicalBytes.length}B exceeds ${MAX_PAYLOAD_BYTES}B limit`);
  }

  const signature = await sign(canonicalBytes);
  const qrPayload = Buffer.from(canonicalBytes).toString('base64');
  const qrSignature = Buffer.from(signature).toString('base64');
  const wireFormat = buildWireFormat(payload, signature);

  const ticket = await tx.ticket.create({
    data: {
      id: payload.ticketId,
      bookingId: input.bookingId,
      qrPayload,
      qrSignature,
      status: 'ISSUED',
    },
    select: { id: true },
  });

  return { ticketId: ticket.id, qrPayload, qrSignature, wireFormat };
}

  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;

}

export interface PaymentDTO {
  paymentId: string;
  bookingId: string;
  status: PaymentStatus;
  amountPaise: number;
  currency: string;
  checkoutUrl: string | null;
  expiresAt: string;
}

export async function initiatePayment(
  userId: string,
  body: { bookingId: string; method?: string | undefined },
  idempotencyKey: string,
): Promise<PaymentDTO> {
  const bodyHash = hashBody(body);

  const cached = await getCachedResponse(idempotencyKey);
  if (cached) {
    if (cached.bodyHash !== bodyHash) throw paymentErrors.idempotencyReused();
    return cached.payload as PaymentDTO;
  }
  const lockAcquired = await acquireIdempotencyLock(idempotencyKey);
  if (!lockAcquired) {
    await new Promise((r) => setTimeout(r, 100));
    const retry = await getCachedResponse(idempotencyKey);
    if (retry && retry.bodyHash === bodyHash) return retry.payload as PaymentDTO;
    throw paymentErrors.idempotencyReused();
  }
  try {
    const booking = await repo.findBookingForPayment(body.bookingId, userId);
    if (!booking) throw paymentErrors.notFound();
    if (['CONFIRMED', 'CHECKED_IN', 'COMPLETED', 'REFUNDED'].includes(booking.status)) throw paymentErrors.alreadySucceeded();
    if (booking.status !== 'PENDING') throw paymentErrors.notAllowed(booking.status);

    const expiresAt = new Date(booking.createdAt.getTime() + PAYMENT_WINDOW_MS);
    if (expiresAt.getTime() < Date.now()) throw paymentErrors.notAllowed(booking.status);

    const order = await razorpayAdapter.createOrder({
      amountPaise: booking.totalFareAmount,
      receipt: booking.id,
      notes: { bookingId: booking.id },
    });

    const row = await repo.createPaymentRow({
      bookingId: booking.id,
      idempotencyKey,
      method: body.method ?? null,
      amountPaise: booking.totalFareAmount,
      currency: booking.totalFareCurrency,
      gatewayOrderId: order.orderId,
    });

    const result: PaymentDTO = {
      paymentId: row.id,
      bookingId: booking.id,
      status: row.status as PaymentStatus,
      amountPaise: Number(row.amountPaise),
      currency: row.currency,
      checkoutUrl: order.checkoutUrl,
      expiresAt: expiresAt.toISOString(),
    };
    await setCachedResponse(idempotencyKey, bodyHash, 201, result);
    return result;
  } finally {
    await releaseIdempotencyLock(idempotencyKey);
  }
}

async function recordGpsFailure(

  failureType: events.GpsFailureType,

        tripId,
        driverId,
        failureCount: count,
        lastFailureType: failureType,
        detectedAt: new Date().toISOString(),

    }

    /* Redis down: skip emit, no throw */
  }

export interface GpsIngestResult {
  accepted: boolean;
  serverSpeedKmh: number | null;

async function readPreviousSample(tripId: string): Promise<NormalizedSample | null> {
  try {
    const raw = await redis.lindex(keys.gpsTrip(tripId), 0);
    if (raw) return JSON.parse(raw) as NormalizedSample;
  } catch { /* fall through to Postgres */ }

  const lastPg = await prisma.gpsSample.findFirst({
    where: { tripId },
    orderBy: { recordedAt: 'desc' },
  });
  if (!lastPg) return null;
  return {
    lat: Number(lastPg.lat),
    lng: Number(lastPg.lng),
    serverSpeedKmh: lastPg.speedKmh !== null ? Number(lastPg.speedKmh) : null,
    clientSpeedKmh: lastPg.clientReportedSpeedKmh !== null ? Number(lastPg.clientReportedSpeedKmh) : 0,
    accuracyM: lastPg.accuracyM !== null ? Number(lastPg.accuracyM) : 0,
    recordedAt: lastPg.recordedAt.toISOString(),
    receivedAt: lastPg.receivedAt.toISOString(),
    timestampDrift: false,

export async function ingestGpsSample(
  userId: string,
  tripId: string,
  body: RawGpsSample,
  deviceHashes: { sessionDeviceHash: string | null; activeSessionDeviceHash: string | null },
): Promise<GpsIngestResult> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (trip.status !== 'IN_PROGRESS') throw tripErrors.notInProgress();

  const previous = await readPreviousSample(tripId);
  const receivedAt = new Date();

  let outcome;
  try {
    outcome = runProvenance({
      current: body,
      previous,
      receivedAt,
      sessionDeviceHash: deviceHashes.sessionDeviceHash,
      activeSessionDeviceHash: deviceHashes.activeSessionDeviceHash,
    });
  } catch (err) {
    const errCode = (err as { code?: string }).code;
    let failType: events.GpsFailureType | null = null;
    if (errCode === 'TELEPORT_DETECTED') failType = 'TELEPORT_DETECTED';
    else if (errCode === 'INVALID_COORDINATES' || errCode === 'INVALID_ACCURACY') failType = 'COORDINATE_INVALID';
    if (failType && trip.driver.id) void recordGpsFailure(tripId, trip.driver.id, failType).catch(() => {});
    throw err;
  }

  if (!outcome.accepted || !outcome.sample) {
    if (outcome.reason === 'speed_mismatch' && trip.driver.id) {
      void recordGpsFailure(tripId, trip.driver.id, 'SPEED_MISMATCH').catch(() => {});
    }
    return { accepted: false, serverSpeedKmh: null };
  }

  const redisKey = keys.gpsTrip(tripId);
  try {
    await redis.lpush(redisKey, JSON.stringify(outcome.sample));
    await redis.ltrim(redisKey, 0, GPS_LIST_MAX - 1);
    await redis.expire(redisKey, GPS_TTL_SEC);
  } catch { /* non-fatal: Redis down, Postgres below still persists */ }

  let counter = 0;
  try {
    const counterKey = `gps:count:${tripId}`;
    counter = await redis.incr(counterKey);
    await redis.expire(counterKey, GPS_TTL_SEC);
  } catch { counter = GPS_PG_EVERY_N; /* force write when Redis down */ }

  if (counter % GPS_PG_EVERY_N === 0) {
    await prisma.gpsSample.create({
      data: {
        tripId,
        lat: outcome.sample.lat,
        lng: outcome.sample.lng,
        speedKmh: outcome.sample.serverSpeedKmh,
        accuracyM: outcome.sample.accuracyM,
        clientReportedSpeedKmh: outcome.sample.clientSpeedKmh,
        receivedAt: new Date(outcome.sample.receivedAt),
        recordedAt: new Date(outcome.sample.recordedAt),
      },
    });
  }

  emitBusPositionThrottled(tripId, events.busPosition({
    tripId,
    scheduleId: trip.scheduleId,
    lat: outcome.sample.lat,
    lng: outcome.sample.lng,
    speedKmh: outcome.sample.serverSpeedKmh ?? 0,
    bearing: null,
    accuracyM: outcome.sample.accuracyM,
    recordedAt: outcome.sample.recordedAt,
    receivedAt: outcome.sample.receivedAt,
    nextStandId: null,
    etaMinutes: null,
  }));

  return { accepted: true, serverSpeedKmh: outcome.sample.serverSpeedKmh };
}

function isDomainError(
  err: unknown,
): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return typeof e.code === 'string' && typeof e.httpStatus === 'number' && typeof e.message === 'string';
}

function isZodError(err: unknown): err is { issues: unknown[] } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  if (!Array.isArray(e.issues)) return false;
  const ctor = (err as { constructor?: { name?: string } }).constructor;
  return (err as { name?: string }).name === 'ZodError' || ctor?.name === 'ZodError';
}

export async function tripRouter(server: FastifyInstance): Promise<void> {
  server.setErrorHandler((error, request, reply) => {
    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({
        error: {
          code: error.code,
          message: error.message,
          ...(error.details ? { details: error.details } : {}),
          requestId: request.id,
        },
      });
    }
    if (isZodError(error)) {
      return reply.status(422).send({
        error: {
          code: 'VALIDATION_ERROR',
          message: 'Validation failed',
          details: { issues: error.issues },
          requestId: request.id,
        },
      });
    }
    const anyErr = error as { statusCode?: number; code?: string; message?: string };
    const statusCode = anyErr.statusCode ?? 500;
    const isProd = process.env.NODE_ENV === 'production';
    return reply.status(statusCode).send({
      error: {
        code: anyErr.code ?? 'INTERNAL_ERROR',
        message: isProd ? 'Something went wrong' : (anyErr.message ?? 'Unknown error'),
        requestId: request.id,
      },
    });
  });

  const auth = { preHandler: requireRole('DRIVER') };

  server.get('/driver/config', auth, c.getConfig);
  server.get('/driver/me', auth, c.getMe);
  server.get('/driver/trips', auth, c.getMyTrips);
  server.get('/driver/trips/history', auth, c.getHistory);
  server.get('/driver/trips/:id', auth, c.getTrip);
  server.post('/driver/trips/:id/start', auth, c.postStartTrip);
  server.post('/driver/trips/:id/end', auth, c.postEndTrip);
  server.get('/driver/trips/:id/manifest', auth, c.getManifest);

  server.post('/driver/uploads', auth, c.postDriverUploads);

// ---------- Start pre-checks (buscon/02 §7.2 + §7.3) ----------
// Note: license expiry check skipped — driver_profiles has no licenseExpiresAt column yet.
// Wire when Phase 7 adds the column.

type StartContext = {
  trip: NonNullable<Awaited<ReturnType<typeof repo.findTripForStart>>>;
  driver: NonNullable<Awaited<ReturnType<typeof repo.findDriverContextForStart>>>;
  userId: string;
  now: Date;
};

async function runStartPreChecks(ctx: StartContext): Promise<void> {
  const { trip, driver, userId, now } = ctx;

  // 1. Driver not blocked/deactivated (checked first so TS narrowing works)
  if (driver.kycStatus === 'BLOCKED' || driver.kycStatus === 'DEACTIVATED') {
    throw tripErrors.driverBlocked();
  }
  // 2. Driver KYC must be VERIFIED or ACTIVE
  if (driver.kycStatus !== 'VERIFIED' && driver.kycStatus !== 'ACTIVE') {
    throw tripErrors.driverNotVerified();
  }
  // 3. (license expiry — skipped, no column)
  // 4. Bus approved
  if (trip.bus.approvalStatus !== 'APPROVED') {
    throw tripErrors.busNotApproved();
  }
  // 5. Bus has layout
  if (!trip.bus.layout) {
    throw tripErrors.busLayoutMissing();
  }
  // 6. Schedule is published (not cancelled)
  if (trip.schedule.status === 'CANCELLED') {
    throw tripErrors.scheduleCancelled();
  }
  if (trip.schedule.status !== 'PUBLISHED') {
    throw tripErrors.notActive();
  }
  // 7. Trip currently SCHEDULED (not already IN_PROGRESS/COMPLETED/CANCELLED)
  if (trip.status !== 'SCHEDULED') {
    if (trip.status === 'IN_PROGRESS') throw tripErrors.alreadyStarted();
    throw tripErrors.notActive();
  }
  // 8. Time window: within ±2h of scheduled departure
  const delta = Math.abs(now.getTime() - trip.schedule.departureTime.getTime());
  if (delta > TRIP_START_WINDOW_MS) {
    throw tripErrors.outsideWindow();
  }
  // 9. No other active trip for this driver
  const active = await repo.findActiveTripForDriver(userId);
  if (active && active.id !== trip.id) {
    throw tripErrors.activeTripExists(active.id);
  }
}
// ---------- Start trip (WF-D-02) ----------
// A5: SELECT ... FOR UPDATE on driver_profiles to prevent concurrent start race.
// Session metadata active_trip_id skipped — sessions has no metadata_json column.
// Double-start prevention comes from runStartPreChecks + FOR UPDATE, which is sufficient.

export async function startTrip(
  userId: string,
  tripId: string,
  actorIp?: string | undefined,
): Promise<StartTripResultDTO> {
  const trip = await repo.findTripForStart(tripId);
  if (!trip) throw tripErrors.notFound();

  const driver = await repo.findDriverContextForStart(userId);
  if (!driver) throw tripErrors.notAssigned();
  if (trip.driver.userId !== userId) throw tripErrors.notAssigned();

  const now = new Date();
  await runStartPreChecks({ trip, driver, userId, now });

  const startedAt = await prisma.$transaction(async (tx) => {
    // A5: row-lock driver_profiles so concurrent /start calls serialize
    const locked = await tx.$queryRaw<Array<{ id: string }>>`
      SELECT id FROM driver_profiles
      WHERE id = ${trip.driver.id}::uuid AND deleted_at IS NULL
      FOR UPDATE
    `;
    if (locked.length === 0) throw tripErrors.notAssigned();

    const updated = await tx.trip.updateMany({
      where: { id: tripId, status: 'SCHEDULED' },
      data: { status: 'IN_PROGRESS', startedAt: now },
    });
    if (updated.count === 0) throw tripErrors.alreadyStarted();

    await tx.driverProfile.update({
      where: { id: trip.driver.id },
      data: { onDuty: true },
    });

    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_started',
      ipAddress: actorIp ?? null,
      metadata: { scheduleId: trip.schedule.id, busId: trip.bus.id },
    });

    return now;
  });

  emitToTopic(`trip:live:${tripId}`, events.tripStarted({
    tripId,
    startedAt: startedAt.toISOString(),
  }));

  return {
    tripId,
    status: 'IN_PROGRESS',
    startedAt: startedAt.toISOString(),
  };
}
// ---------- End trip (WF-D-06) ----------
// Distance computed from all Postgres gps_samples (buscon/04 §18).
// Atomic: trips COMPLETED + driver.onDuty=false + audit_logs.

export async function endTrip(
  userId: string,
  tripId: string,
  actorIp?: string | undefined,
): Promise<EndTripResultDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (trip.status !== 'IN_PROGRESS') throw tripErrors.notInProgress();

  const gpsRows = await prisma.gpsSample.findMany({
    where: { tripId },
    select: { lat: true, lng: true, recordedAt: true },
    orderBy: { recordedAt: 'asc' },
  });
  const { computeTripDistanceKm } = await import('../../lib/eta.js');
  const samples = gpsRows.map((r) => ({
    lat: Number(r.lat),
    lng: Number(r.lng),
    recordedAt: r.recordedAt,
  }));
  const distanceKm = computeTripDistanceKm(samples);

  const counts = await repo
    .findTripsForDriver(userId, undefined)
    .then((all) => all.find((x) => x.id === tripId)?.counts ?? { boarded: 0, confirmed: 0 });

  const now = new Date();
  const startedAtMs = trip.startedAt?.getTime() ?? now.getTime();
  const durationMinutes = Math.round((now.getTime() - startedAtMs) / 60_000);

  await prisma.$transaction(async (tx) => {
    const updated = await tx.trip.updateMany({
      where: { id: tripId, status: 'IN_PROGRESS' },
      data: { status: 'COMPLETED', endedAt: now, distanceKm },
    });
    if (updated.count === 0) throw tripErrors.notInProgress();

    await tx.driverProfile.update({
      where: { id: trip.driverId },
      data: { onDuty: false },
    });

    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_ended',
      ipAddress: actorIp ?? null,
      metadata: { distanceKm, durationMinutes, boardedCount: counts.boarded },
    });
  });

  emitToTopic(`trip:live:${tripId}`, events.tripEnded({
    tripId,
    endedAt: now.toISOString(),
    distanceKm: Math.round(distanceKm * 100) / 100,
  }));
  clearThrottleState(tripId);

  return {
    tripId,
    status: 'COMPLETED',
    startedAt: trip.startedAt?.toISOString() ?? null,
    endedAt: now.toISOString(),
    durationMinutes,
    distanceKm: Math.round(distanceKm * 100) / 100,
    boardedCount: counts.boarded,
    confirmedCount: counts.confirmed,
  };
}
// ---------- Report delay (buscon/02 §12) ----------

export async function reportDelay(
  userId: string,
  tripId: string,
  body: { reason: DelayReason; delayMinutes: number; notes?: string | undefined },
  actorIp?: string | undefined,
): Promise<DelayReportDTO> {
  if (body.delayMinutes < DELAY_MIN_MINUTES || body.delayMinutes > DELAY_MAX_MINUTES) {
    throw tripErrors.notFound();
  }

  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();
  if (trip.status !== 'IN_PROGRESS') throw tripErrors.notInProgress();

  const now = new Date();
  await prisma.$transaction(async (tx) => {
    await tx.trip.update({
      where: { id: tripId },
      data: { delayReason: body.reason, delayMinutes: body.delayMinutes },
    });
    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_delayed',
      ipAddress: actorIp ?? null,
      metadata: {
        reason: body.reason,
        delayMinutes: body.delayMinutes,
        notes: body.notes ?? null,
      },
    });
  });

  emitToTopic(`trip:live:${tripId}`, events.tripDelayed({
    tripId,
    reason: body.reason as events.TripDelayReason,
    minutes: body.delayMinutes,
  }));

  return {
    tripId,
    delayReason: body.reason,
    delayMinutes: body.delayMinutes,
    reportedAt: now.toISOString(),
  };
}

export async function getManifest(userId: string, tripId: string): Promise<ManifestDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();

  const rows = await repo.findManifestRowsForSchedule(trip.scheduleId);
  const stands = await repo.findBoardingStandsForSchedule(trip.scheduleId);

  type Agg = {
    bookingId: string;
    bookingReference: string;
    seatLabels: string[];
    passengerNames: string[];
    boardingStatus: 'CONFIRMED' | 'BOARDED';
    boardedAt: string | null;
    boardingStandId: string;
  };

  const bookingsById = new Map<string, Agg>();
  for (const row of rows) {
    let agg = bookingsById.get(row.bookingId);
    if (!agg) {
      agg = {
        bookingId: row.bookingId,
        bookingReference: row.booking.bookingReference,
        seatLabels: [],
        passengerNames: [],
        boardingStatus: row.booking.status === 'CHECKED_IN' ? 'BOARDED' : 'CONFIRMED',
        boardedAt: row.booking.checkedInAt?.toISOString() ?? null,
        boardingStandId: row.booking.boardingStand.id,
      };
      bookingsById.set(row.bookingId, agg);
    }
    agg.seatLabels.push(row.seat.seatLabel);
    agg.passengerNames.push(row.name);
  }

  const byStandMap = new Map<string, ManifestStandGroupDTO>();
  for (const s of stands) {
    byStandMap.set(s.stand.id, {
      standId: s.stand.id,
      standName: s.stand.name,
      sequence: s.sequence,
      passengers: [],
    });
  }

  let totalPassengers = 0;
  let totalBoarded = 0;
  for (const agg of bookingsById.values()) {
    agg.seatLabels.sort();
    agg.passengerNames.sort();
    totalPassengers += agg.passengerNames.length;
    if (agg.boardingStatus === 'BOARDED') totalBoarded += agg.passengerNames.length;

    const passenger: ManifestPassengerDTO = {
      bookingId: agg.bookingId,
      bookingReference: agg.bookingReference,
      seatLabels: agg.seatLabels,
      passengerNames: agg.passengerNames,
      boardingStatus: agg.boardingStatus,
      boardedAt: agg.boardedAt,
    };

    const group = byStandMap.get(agg.boardingStandId);
    if (group) group.passengers.push(passenger);
    else
      byStandMap.set(agg.boardingStandId, {
        standId: agg.boardingStandId,
        standName: 'Unknown Stand',
        sequence: 999,
        passengers: [passenger],
      });
  }

  return {
    tripId,
    scheduleId: trip.scheduleId,
    totalPassengers,
    totalBoarded,
    lastUpdatedAt: new Date().toISOString(),
    byStand: Array.from(byStandMap.values()).sort((a, b) => a.sequence - b.sequence),
  };
}

export async function listHistory(
  userId: string,
  opts: { from?: string | undefined; to?: string | undefined; limit: number; cursor?: string | undefined },
): Promise<{ data: DriverTripSummaryDTO[]; nextCursor: string | null }> {
  let cursorCreatedAt: Date | undefined;
  if (opts.cursor) {
    try {
      const decoded = JSON.parse(Buffer.from(opts.cursor, 'base64').toString('utf8')) as { createdAt: string };
      cursorCreatedAt = new Date(decoded.createdAt);
    } catch {
      throw tripErrors.notFound();
    }
  }

  const rows = await repo.findHistoryForDriver(userId, {
    ...(opts.from !== undefined ? { from: opts.from } : {}),
    ...(opts.to !== undefined ? { to: opts.to } : {}),
    limit: opts.limit,
    ...(cursorCreatedAt !== undefined ? { cursorCreatedAt } : {}),
  });

  const hasMore = rows.length > opts.limit;
  const data = hasMore ? rows.slice(0, opts.limit) : rows;
  const nextCursor = hasMore
    ? Buffer.from(JSON.stringify({ createdAt: data[data.length - 1]!.createdAt.toISOString() })).toString('base64')
    : null;

  const scheduleIds = data.map((t) => t.schedule.id);
  type CountRow = { scheduleId: string; status: string; cnt: number };
  const countRows =
    scheduleIds.length === 0
      ? []
      : await prisma.$queryRaw<CountRow[]>`
          SELECT bp.schedule_id AS "scheduleId", b.status, COUNT(*)::int AS cnt
          FROM booking_passengers bp
          JOIN bookings b ON b.id = bp.booking_id
          WHERE bp.schedule_id = ANY(${scheduleIds}::uuid[])
            AND bp.deleted_at IS NULL
            AND b.deleted_at IS NULL
            AND b.status IN ('CONFIRMED', 'CHECKED_IN')
          GROUP BY bp.schedule_id, b.status
        `;
  const countsMap = new Map<string, { boarded: number; confirmed: number }>();
  for (const id of scheduleIds) countsMap.set(id, { boarded: 0, confirmed: 0 });
  for (const r of countRows) {
    const cur = countsMap.get(r.scheduleId) ?? { boarded: 0, confirmed: 0 };
    if (r.status === 'CHECKED_IN') cur.boarded += r.cnt;
    cur.confirmed += r.cnt;
    countsMap.set(r.scheduleId, cur);
  }

  return {
    data: data.map((t) => {
      const counts = countsMap.get(t.schedule.id) ?? { boarded: 0, confirmed: 0 };
      return {
        tripId: t.id,
        scheduleId: t.schedule.id,
        bus: t.bus,
        route: {
          originCity: t.schedule.route.originCity.name,
          destinationCity: t.schedule.route.destCity.name,
        },
        departureTime: t.schedule.departureTime.toISOString(),
        arrivalTime: t.schedule.arrivalTime.toISOString(),
        status: t.status as TripStatus,
        boardedCount: counts.boarded,
        confirmedCount: counts.confirmed,
      };
    }),
    nextCursor,
  };
}

// ---------- Cancel trip (Phase 12 — cascade via cancelBooking) ----------

export async function cancelTrip(
  userId: string,
  tripId: string,
  body: { reason: string },
  actorIp?: string | undefined,
): Promise<import('./trip.types.js').CancelTripResultDTO> {
  const trip = await repo.findTripForDriver(userId, tripId);
  if (!trip) throw tripErrors.notFound();

  if (trip.status === 'CANCELLED') {
    throw errors.invalidTransition('Trip is already cancelled');
  }
  if (trip.status === 'COMPLETED') {
    throw errors.invalidTransition('Cannot cancel a completed trip');
  }

  const bookings = await prisma.booking.findMany({
    where: {
      scheduleId: trip.scheduleId,
      deletedAt: null,
      status: { in: ['CONFIRMED', 'OPERATOR_CANCELLED', 'PENDING'] },
    },
    select: { id: true, userId: true, status: true },
  });

  const { cancelBooking, expirePendingBooking } = await import('../booking/booking.service.js');

  let cancelledBookings = 0;
  let refundedPaise = 0;

  for (const b of bookings) {
    try {
      if (b.status === 'CONFIRMED') {
        await prisma.booking.update({
          where: { id: b.id },
          data: { status: 'OPERATOR_CANCELLED' },
        });
        const result = await cancelBooking(b.userId, b.id);
        cancelledBookings++;
        refundedPaise += result.refund.amountPaise;
      } else if (b.status === 'OPERATOR_CANCELLED') {
        const result = await cancelBooking(b.userId, b.id);
        cancelledBookings++;
        refundedPaise += result.refund.amountPaise;
      } else if (b.status === 'PENDING') {
        await expirePendingBooking(b.id);
        cancelledBookings++;
      }
    } catch (err) {
      logger.warn({ err, bookingId: b.id, tripId }, 'cancelTrip: cascade failed for booking');
    }
  }

  const now = new Date();
  await prisma.$transaction(async (tx) => {
    await tx.trip.update({
      where: { id: tripId },
      data: { status: 'CANCELLED', cancelledAt: now },
    });
    await writeAuditEvent({
      tx,
      actorId: userId,
      sessionId: null,
      entityType: 'Trip',
      entityId: tripId,
      action: 'trip_cancelled',
      ipAddress: actorIp ?? null,
      metadata: {
        scheduleId: trip.scheduleId,
        reason: body.reason,
        cancelledBookings,
        refundedPaise,
      },
    });
  });

  emitToTopic('trip:live:' + tripId, events.tripCancelled({
    tripId,
    reason: body.reason,
  }));
  clearThrottleState(tripId);

  return {
    tripId,
    status: 'CANCELLED',
    cancelledBookings,
    refundedPaise,
  };
}

  const rawBody = (request as RawBodyRequest).rawBody;

import { postRazorpayWebhook } from './webhooks.controller.js';

function isDomainError(err: unknown): err is { code: string; httpStatus: number; message: string; details?: Record<string, unknown> } {
  if (!err || typeof err !== 'object') return false;
  const e = err as Record<string, unknown>;
  return typeof e.code === 'string' && typeof e.httpStatus === 'number' && typeof e.message === 'string';
}

export async function webhooksRouter(server: FastifyInstance): Promise<void> {
  server.setErrorHandler((error, request, reply) => {
    if (isDomainError(error)) {
      return reply.status(error.httpStatus).send({
        error: {
          code: error.code,
          message: error.message,
          ...(error.details ? { details: error.details } : {}),
          requestId: request.id,
        },
      });
    }
    server.log.error({ err: error }, 'webhook route error');
    return reply.status(500).send({
      error: { code: 'INTERNAL_ERROR', message: 'Something went wrong', requestId: request.id },
    });
  });

  server.post('/webhooks/razorpay', postRazorpayWebhook);
}

import * as events from '../../realtime/events.js';
import { emitToBooking, emitToTopic } from '../../realtime/emit.js';
import { createHash } from 'node:crypto';
import { logger } from '../../lib/logger.js';
import { razorpayAdapter } from '../../integrations/razorpay.js';
import { writeFallback } from '../../lib/webhook-fallback.js';
import { razorpayWebhookSchema } from './webhooks.schema.js';
import { webhookErrors } from './webhooks.errors.js';
import type { RazorpayWebhookPayload, WebhookProcessResult } from './webhooks.types.js';

export const WEBHOOK_TX_OPTS = { timeout: 30000, maxWait: 10000 } as const;

export async function handleRazorpayWebhook(
  rawBody: Buffer,
  signature: string,
  opts: { isReplay?: boolean } = {},
): Promise<WebhookProcessResult> {
  if (!razorpayAdapter.verifyWebhookSignature(rawBody, signature)) {
    throw webhookErrors.invalidSignature();
  }

  let body: RazorpayWebhookPayload;
  try {
    body = razorpayWebhookSchema.parse(JSON.parse(rawBody.toString('utf8'))) as RazorpayWebhookPayload;
  } catch (err) {
    if (opts.isReplay) throw err;
    logger.warn({ err }, 'webhook schema parse failed');
    throw webhookErrors.invalidPayload();
  }

  const payloadHash = createHash('sha256').update(rawBody).digest('hex');

  try {
    return await dispatch(body, rawBody, payloadHash);
  } catch (err) {
    // 09 §13.5: DB error → local file fallback, 200 to Razorpay, replay job retries
    logger.error({ err, event: body.event }, 'webhook DB error — writing fallback');
    await writeFallback(rawBody, { event: body.event, reason: (err as Error).message, signature });
    return { status: 'processed', eventType: body.event };
  }
}

async function dispatch(
  body: RazorpayWebhookPayload,
  rawBody: Buffer,
  payloadHash: string,
): Promise<WebhookProcessResult> {
  if (body.event === 'payment.captured') return handlePaymentCaptured(body, rawBody, payloadHash);
  if (body.event === 'payment.failed') return handlePaymentFailed(body, rawBody, payloadHash);
  if (body.event === 'refund.processed') return handleRefundProcessed(body, rawBody, payloadHash);
  return { status: 'ignored', eventType: body.event };
}
import { prisma } from '../../lib/prisma.js';
import { redis, keys } from '../../lib/redis.js';
import * as repo from './webhooks.repository.js';
import { performPaymentCapturedTx, performPaymentFailedTx, performRefundProcessedTx } from './webhooks.handlers.js';

async function handlePaymentCaptured(
  body: RazorpayWebhookPayload,
  rawBody: Buffer,
  payloadHash: string,
): Promise<WebhookProcessResult> {
  const ent = body.payload.payment!.entity;
  const payment = await repo.findPaymentByGatewayIds({
    razorpayOrderId: ent.order_id ?? null,
    razorpayPaymentId: ent.id,
  });
  if (!payment) return { status: 'ignored', eventType: body.event };

  const result = await prisma.$transaction(async (tx) => {
    try {
      await tx.gatewayWebhook.create({
        data: {
          paymentId: payment.id,
          eventType: 'payment.captured',
          payloadHash,
          payloadJson: body as never,
          signature: 'verified',
          isTest: payment.isTest,
        },
      });
    } catch (e) {
      if ((e as { code?: string }).code === 'P2002') return { kind: 'duplicate' as const };
      throw e;
    }
    if (payment.status === 'SUCCEEDED') return { kind: 'duplicate' as const };
    await tx.payment.update({ where: { id: payment.id }, data: { gatewayPaymentId: ent.id, method: ent.method ?? null } });
    const r = await performPaymentCapturedTx(tx, payment, null);
    return { kind: 'processed' as const, ticketId: r.ticketId };
  }, WEBHOOK_TX_OPTS);

  if (result.kind === 'processed') {
    for (const seatId of payment.booking.passengers.map((p) => p.seatId)) {
      await redis.del(keys.seatLock(payment.booking.scheduleId, seatId)).catch(() => {});
      emitToTopic(`seat:availability:${payment.booking.scheduleId}`, events.seatBooked({
        seatId,
        bookingId: payment.bookingId,
      }));
    }
    const __fb = payment.booking.fareBreakdownJson as { total_paise?: number };
    emitToBooking(payment.bookingId, events.paymentSucceeded({
      paymentId: payment.id,

  }

}

  rawBody: Buffer,

    try {
      await tx.gatewayWebhook.create({

          payloadHash,
          payloadJson: body as never,
          signature: 'verified',
          isTest: payment.isTest,
        },
      });

      throw e;
    }
    if (payment.status === 'FAILED') return { kind: 'duplicate' as const };
    await tx.payment.update({ where: { id: payment.id }, data: { gatewayPaymentId: ent.id } });
    await performPaymentFailedTx(tx, payment, null);
    return { kind: 'processed' as const };
  }, WEBHOOK_TX_OPTS);

  return { status: result.kind, eventType: body.event, paymentId: payment.id };
}

import { verifyAccessToken } from '../modules/identity/jwt.service.js';
import { prisma } from '../lib/prisma.js';
import { redis, keys } from '../lib/redis.js';

export interface WsClaims {
  userId: string;
  sessionId: string;
  role: string;
  exp: number;
}

export type WsAuthReason = 'INVALID_TOKEN' | 'SESSION_REVOKED' | 'SESSION_INACTIVE';

export class WsAuthError extends Error {
  constructor(public readonly reason: WsAuthReason) {
    super(reason);
    this.name = 'WsAuthError';
  }
}

export function isWsAuthError(err: unknown): err is { reason: WsAuthReason } {
  return !!err && typeof err === 'object' && typeof (err as { reason?: unknown }).reason === 'string';
}

export async function verifyHandshake(token: string): Promise<WsClaims> {
  let claims;
  try {
    claims = verifyAccessToken(token);
  } catch {
    throw new WsAuthError('INVALID_TOKEN');
  }

  const revoked = await redis.exists(keys.revokedSession(claims.sid));
  if (revoked === 1) throw new WsAuthError('SESSION_REVOKED');

  const session = await prisma.session.findUnique({ where: { id: claims.sid } });
  if (!session || session.status !== 'ACTIVE' || session.expiresAt < new Date()) {
    throw new WsAuthError('SESSION_INACTIVE');
  }

  return {
    userId: claims.sub,
    sessionId: claims.sid,
    role: claims.role,
    exp: claims.exp,
  };
}

const BUS_POSITION_FLOOR_MS = 3000;

let ioRef: SocketIOServer | null = null;
const lastBusPositionEmit = new Map<string, number>();

export function setIO(io: SocketIOServer | null): void {
  ioRef = io;
}

export function getIO(): SocketIOServer | null {
  return ioRef;
}

interface WireEnvelope {
  event: string;
  data: Record<string, unknown>;
  ts: string;
}

function toWire(e: EventEnvelope): WireEnvelope {
  return { event: e.event, data: e.data, ts: new Date().toISOString() };
}

function safeEmit(target: string, e: EventEnvelope): void {
  const io = ioRef;
  if (!io) {
    logger.warn({ event: e.event, target }, 'emit skipped - ws not attached');
    return;
  }
  try {
    io.to(target).emit('message', toWire(e));
  } catch (err) {
    logger.warn({ err, event: e.event, target }, 'emit failed');
  }
}

export function emitToTopic(topic: string, e: EventEnvelope): void {
  safeEmit(topic, e);
}

export function emitToBooking(bookingId: string, e: EventEnvelope): void {
  safeEmit(`booking:${bookingId}`, e);
}

export function emitBusPositionThrottled(tripId: string, e: EventEnvelope): void {
  const now = Date.now();
  const last = lastBusPositionEmit.get(tripId) ?? 0;
  if (now - last < BUS_POSITION_FLOOR_MS) {
    logger.debug({ tripId, deltaMs: now - last }, 'bus:position throttled');
    return;
  }
  lastBusPositionEmit.set(tripId, now);
  emitToTopic(`trip:live:${tripId}`, e);
}

export function clearThrottleState(tripId: string): void {
  lastBusPositionEmit.delete(tripId);
}

export function clearAllThrottleState(): void {
  lastBusPositionEmit.clear();
}

import type { Server as HttpServer } from 'node:http';
import { Server as SocketIOServer, type Socket } from 'socket.io';
import { env } from '../config/env.js';
import { logger } from '../lib/logger.js';
import { verifyAccessToken, rotateRefreshToken } from '../modules/identity/jwt.service.js';
import { verifyHandshake, isWsAuthError, type WsClaims } from './auth.js';
import { subscribeToTopic } from './topics.js';
import { setIO, clearAllThrottleState } from './emit.js';

const AUTH_TIMEOUT_MS = 5000;
const AUTH_EXPIRING_LEAD_MS = env.WS_AUTH_EXPIRING_LEAD_SEC * 1000;
const AUTH_REFRESH_WINDOW_MS = env.WS_AUTH_REFRESH_WINDOW_SEC * 1000;
const MAX_MESSAGE_BYTES = 64 * 1024;

let io: SocketIOServer | null = null;

interface SocketMeta {
  claims: WsClaims | null;
  authTimer: NodeJS.Timeout | null;
  expiringTimer: NodeJS.Timeout | null;
  refreshTimer: NodeJS.Timeout | null;
}

const metaMap = new WeakMap<Socket, SocketMeta>();

function getAllowedOrigins(): string[] {
  return env.CORS_ALLOWED_ORIGINS.split(',').map((s) => s.trim()).filter(Boolean);
}

function originAllowed(origin: string | undefined): boolean {
  if (!origin) return env.NODE_ENV === 'test';
  return getAllowedOrigins().includes(origin);
}

function closeWith(socket: Socket, code: number, message: string): void {
  try {
    socket.emit('error', { code, message });
    socket.disconnect(true);
  } catch (err) {
    logger.warn({ err, code }, 'socket close failed');
  }
}

function clearTimers(m: SocketMeta): void {
  if (m.authTimer) clearTimeout(m.authTimer);
  if (m.expiringTimer) clearTimeout(m.expiringTimer);
  if (m.refreshTimer) clearTimeout(m.refreshTimer);
  m.authTimer = null; m.expiringTimer = null; m.refreshTimer = null;
}

function scheduleReAuth(socket: Socket, m: SocketMeta, claims: WsClaims): void {
  const nowSec = Math.floor(Date.now() / 1000);
  const expiringInMs = Math.max(0, (claims.exp - 60 - nowSec) * 1000);
  m.expiringTimer = setTimeout(() => {
    if (!socket.connected) return;
    socket.emit('auth:expiring', { exp: claims.exp });
    m.refreshTimer = setTimeout(() => {
      if (socket.connected) closeWith(socket, 4001, 'TOKEN_EXPIRED');
    }, AUTH_REFRESH_WINDOW_MS).unref();
  }, expiringInMs).unref();
}

async function handleAuth(socket: Socket, m: SocketMeta, raw: { token?: unknown }): Promise<void> {
  if (m.claims) { socket.emit('error', { code: 400, message: 'ALREADY_AUTHENTICATED' }); return; }
  if (typeof raw.token !== 'string') { closeWith(socket, 4001, 'INVALID_AUTH_MESSAGE'); return; }
  try {
    const claims = await verifyHandshake(raw.token);
    m.claims = claims;
    if (m.authTimer) { clearTimeout(m.authTimer); m.authTimer = null; }
    socket.join(`user:${claims.userId}`);
    socket.emit('auth:ok', { role: claims.role });
    scheduleReAuth(socket, m, claims);
  } catch (err) {
    const reason = isWsAuthError(err) ? err.reason : 'INVALID_TOKEN';
    closeWith(socket, 4001, reason);
  }
}

async function handleSubscribe(socket: Socket, m: SocketMeta, raw: { topics?: unknown }): Promise<void> {
  if (!m.claims) { closeWith(socket, 4001, 'NOT_AUTHENTICATED'); return; }
  const topics = Array.isArray(raw.topics) ? raw.topics : [];
  if (topics.length === 0) { socket.emit('error', { code: 400, message: 'NO_TOPICS' }); return; }
  for (const t of topics) {
    if (typeof t !== 'string') continue;
    const result = await subscribeToTopic(socket, t, m.claims);
    if (!result.allowed) { closeWith(socket, 1008, result.reason ?? 'FORBIDDEN'); return; }
  }
  socket.emit('subscribed', { topics });
}

async function handleAuthRefresh(socket: Socket, m: SocketMeta, raw: { refreshToken?: unknown }): Promise<void> {
  if (!m.claims) { closeWith(socket, 4001, 'NOT_AUTHENTICATED'); return; }
  if (typeof raw.refreshToken !== 'string') { closeWith(socket, 4001, 'INVALID_REFRESH'); return; }
  try {
    const tokens = await rotateRefreshToken(raw.refreshToken);
    const decoded = verifyAccessToken(tokens.accessToken);
    m.claims = { userId: decoded.sub, sessionId: decoded.sid, role: decoded.role, exp: decoded.exp };
    if (m.refreshTimer) { clearTimeout(m.refreshTimer); m.refreshTimer = null; }
    if (m.expiringTimer) { clearTimeout(m.expiringTimer); m.expiringTimer = null; }
    socket.emit('auth:refreshed', {
      accessToken: tokens.accessToken,
      refreshToken: tokens.refreshToken,
      expiresAt: tokens.expiresAt,
    });
    scheduleReAuth(socket, m, m.claims);
  } catch (err) {
    logger.warn({ err, userId: m.claims.userId }, 'auth:refresh failed');
    closeWith(socket, 4001, 'REFRESH_FAILED');
  }
}

async function handleMessage(socket: Socket, m: SocketMeta, raw: unknown): Promise<void> {
  if (!raw || typeof raw !== 'object') {
    socket.emit('error', { code: 400, message: 'INVALID_MESSAGE' });
    return;
  }
  const msg = raw as { action?: unknown; token?: unknown; refreshToken?: unknown; topics?: unknown };
  if (msg.action === 'auth') { await handleAuth(socket, m, msg); return; }
  if (msg.action === 'subscribe') { await handleSubscribe(socket, m, msg); return; }
  if (msg.action === 'auth:refresh') { await handleAuthRefresh(socket, m, msg); return; }
  socket.emit('error', { code: 400, message: 'UNKNOWN_ACTION' });
}

export function attachRealtime(httpServer: HttpServer): SocketIOServer {
  if (io) return io;
  io = new SocketIOServer(httpServer, {
    path: '/v1/ws',
    cors: { origin: getAllowedOrigins(), credentials: true },
    maxHttpBufferSize: MAX_MESSAGE_BYTES,
    transports: ['websocket', 'polling'],
  });

  io.use((socket, next) => {
    if (!originAllowed(socket.handshake.headers.origin)) return next(new Error('ORIGIN_NOT_ALLOWED'));
    next();
  });

  io.on('connection', (socket) => {
    logger.debug({ id: socket.id }, 'ws connection opened');
    const m: SocketMeta = { claims: null, authTimer: null, expiringTimer: null, refreshTimer: null };
    metaMap.set(socket, m);
    m.authTimer = setTimeout(() => {
      if (!m.claims) closeWith(socket, 4003, 'AUTH_TIMEOUT');
    }, AUTH_TIMEOUT_MS).unref();
    socket.on('message', (raw: unknown) => { void handleMessage(socket, m, raw); });
    socket.on('disconnect', (reason) => {
      logger.debug({ id: socket.id, reason }, 'ws connection closed');
      clearTimers(m);
    });
  });

  setIO(io);
  logger.info({ path: '/v1/ws' }, 'ws server attached');
  return io;
}

export async function closeRealtime(): Promise<void> {
  if (!io) return;
  clearAllThrottleState();
  const server = io;
  io = null;
  setIO(null);
  await new Promise<void>((resolve) => { server.close(() => resolve()); });
  logger.info('ws server closed');
}

import type { Socket } from 'socket.io';
import { prisma } from '../lib/prisma.js';
import { logger } from '../lib/logger.js';
import type { WsClaims } from './auth.js';

export interface SubscribeResult {
  allowed: boolean;
  reason?: string;
}

const UUID = '[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}';
const SEAT_AVAILABILITY_RE = new RegExp(`^seat:availability:(${UUID})$`);
const TRIP_LIVE_RE = new RegExp(`^trip:live:(${UUID})$`);
const BOOKING_RE = new RegExp(`^booking:(${UUID})$`);
const ADMIN_NOTIFICATIONS = 'admin:notifications';

async function canSeatAvailability(scheduleId: string): Promise<boolean> {
  const row = await prisma.schedule.findFirst({
    where: { id: scheduleId, status: 'PUBLISHED', deletedAt: null },
    select: { id: true },
  });
  return !!row;
}

async function canTripLive(tripId: string, claims: WsClaims): Promise<boolean> {
  const trip = await prisma.trip.findUnique({
    where: { id: tripId },
    select: { scheduleId: true },
  });
  if (!trip) return false;
  if (claims.role === 'SUPER_ADMIN') return true;
  const booking = await prisma.booking.findFirst({
    where: {
      userId: claims.userId,
      scheduleId: trip.scheduleId,
      status: { in: ['CONFIRMED', 'CHECKED_IN'] },
      deletedAt: null,
    },
    select: { id: true },
  });
  return !!booking;
}

async function canBooking(bookingId: string, claims: WsClaims): Promise<boolean> {
  if (claims.role === 'SUPER_ADMIN') return true;
  const b = await prisma.booking.findFirst({
    where: { id: bookingId, userId: claims.userId, deletedAt: null },
    select: { id: true },
  });
  return !!b;
}

export async function subscribeToTopic(
  socket: Socket,
  topic: string,
  claims: WsClaims,
): Promise<SubscribeResult> {
  if (topic === ADMIN_NOTIFICATIONS) {
    if (claims.role !== 'SUPER_ADMIN') return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  let m = topic.match(SEAT_AVAILABILITY_RE);
  if (m) {
    if (!(await canSeatAvailability(m[1]!))) return { allowed: false, reason: 'SCHEDULE_NOT_FOUND' };
    socket.join(topic);
    return { allowed: true };
  }

  m = topic.match(TRIP_LIVE_RE);
  if (m) {
    if (!(await canTripLive(m[1]!, claims))) return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  m = topic.match(BOOKING_RE);
  if (m) {
    if (!(await canBooking(m[1]!, claims))) return { allowed: false, reason: 'FORBIDDEN' };
    socket.join(topic);
    return { allowed: true };
  }

  logger.warn({ topic, userId: claims.userId }, 'unknown topic pattern');
  return { allowed: false, reason: 'UNKNOWN_TOPIC' };
}
