function resolveApiBase(): string {
  // ---- Web ----
  if (Platform.OS === 'web') {
    if (typeof window !== 'undefined' && window.location) {
      const proto = window.location.protocol === 'https:' ? 'https' : 'http';
      const host = window.location.hostname || 'localhost';
      return `${proto}://${host}:4310/v1`;
    }
    return 'http://localhost:4310/v1';
  }

  // ---- Native ----
  const hostUri =
    (Constants.expoConfig as any)?.hostUri ??
    (Constants.expoGoConfig as any)?.debuggerHost ??
    null;

  let host: string | null = null;
  if (hostUri && typeof hostUri === 'string') {
    host = hostUri.split(':')[0] ?? null;
  }

  // Tunnel domains cannot serve :4310 â€” skip them
  if (host && (host.endsWith('.exp.direct') || host.endsWith('.exp.host') || host.endsWith('.ngrok.io'))) {
    host = null;
  }

  // Android emulator loopback -> swap to the special host alias
  if (Platform.OS === 'android' && (host === 'localhost' || host === '127.0.0.1' || host === '10.0.2.2')) {
    host = '10.0.2.2';
  }

  if (host) return `http://${host}:4310/v1`;

  // Final fallbacks
  if (Platform.OS === 'android') return 'http://10.0.2.2:4310/v1';
  if (Platform.OS === 'ios')     return 'http://localhost:4310/v1';
  return 'http://localhost:4310/v1';
}

class ApiClient {
  private accessToken: string | null = null;
  private refreshHandler: RefreshHandler | null = null;
  get baseUrl(): string { return API_BASE; }
  setAccessToken(t: string | null): void { this.accessToken = t; }
  setRefreshHandler(f: RefreshHandler | null): void { this.refreshHandler = f; }

  async request<T>(path: string, opts: RequestOptions = {}): Promise<ApiResponse<T>> {
    const url = API_BASE + (path.startsWith('/') ? path : '/' + path);
    const headers: Record<string, string> = { 'Content-Type': 'application/json', ...opts.headers };
    if (!opts.skipAuth && this.accessToken) headers.Authorization = 'Bearer ' + this.accessToken;
    if (opts.idempotencyKey) headers['Idempotency-Key'] = opts.idempotencyKey;

    const init: RequestInit = { method: opts.method ?? 'GET', headers };
    if (opts.body !== undefined && init.method !== 'GET') init.body = JSON.stringify(opts.body);

    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), REQUEST_TIMEOUT_MS);
    init.signal = controller.signal;

    if (__DEV__) console.log('[api] ->', init.method, url);

    let response: Response;
    try {
      response = await fetch(url, init);
    } catch (e: any) {
      clearTimeout(timer);
      if (e?.name === 'AbortError') {
        if (__DEV__) console.warn('[api] TIMEOUT', url);
        throw new ApiError(0, { code: 'NETWORK_TIMEOUT', message: `Timeout after ${REQUEST_TIMEOUT_MS / 1000}s (${API_BASE})` });
      }
      if (__DEV__) console.warn('[api] NETWORK ERROR', url, e?.message);
      throw new ApiError(0, { code: 'NETWORK_ERROR', message: `Cannot reach ${API_BASE} - ${e?.message ?? 'network error'}` });
    }
    clearTimeout(timer);

    if (__DEV__) console.log('[api] <-', response.status, url);

    if (response.status === 401 && !opts.skipAuth && this.refreshHandler) {
      const fresh = await this.refreshHandler();
      if (fresh) {
        headers.Authorization = 'Bearer ' + fresh;
        response = await fetch(url, { ...init, headers });
      }
    }
    return this.parse<T>(response);
  }

  private async parse<T>(response: Response): Promise<ApiResponse<T>> {
    const isJson = (response.headers.get('content-type') ?? '').includes('application/json');
    if (!response.ok) {
      if (isJson) {
        const body = (await response.json()) as ApiErrorResponse;
        if (__DEV__) console.warn('[api] ERROR', response.status, body);
        if (body?.error) throw new ApiError(response.status, body.error);
      }
      throw new ApiError(response.status, { code: 'HTTP_ERROR', message: response.statusText || `HTTP ${response.status}` });
    }
    if (response.status === 204) return { data: {} as T };
    return (await response.json()) as ApiResponse<T>;
  }

  get<T>(p: string) { return this.request<T>(p, { method: 'GET' }); }
  post<T>(p: string, b?: unknown, e: Partial<RequestOptions> = {}) { return this.request<T>(p, { method: 'POST', body: b, ...e }); }
  patch<T>(p: string, b?: unknown) { return this.request<T>(p, { method: 'PATCH', body: b }); }

    await authApi.requestOtp(phone);

    const fingerprint = await getDeviceFingerprint();

export function MapScreen({ route, navigation }: Props) {

  const t = useDriverTheme();
  const [fix, setFix] = useState<Fix | null>(null);
  const [err, setErr] = useState<string | null>(null);

  // 1. Get Trip Data
  const tripQ = useQuery({ queryKey: ['trip', tripId], queryFn: () => getTrip(tripId) });
  const trip = tripQ.data;

  // 2. Get Stops with coordinates from public schedule API
  const routeQ = useQuery({
    queryKey: ['public-schedule', trip?.scheduleId],
    queryFn: async () => {
      const res = await api.get<any>(`/public/schedules/${trip?.scheduleId}`);
      return res.data;
    },
    enabled: !!trip?.scheduleId,
  });

  // Extract coordinates safely (public API returns flat stands or nested)
  const rawStops = routeQ.data?.route?.stands ?? routeQ.data?.route?.routeStands ?? [];
  const stops = rawStops.map((rs: any) => ({
    lat: Number(rs.stand?.lat ?? rs.lat),
    lng: Number(rs.stand?.lng ?? rs.lng),
    label: rs.stand?.name ?? rs.standName ?? 'Stop'
  })).filter((s: any) => !isNaN(s.lat) && !isNaN(s.lng));

  // 3. Fetch actual road route from OSRM
  const osrmQ = useQuery({
    queryKey: ['osrm-route', stops.map((s: any) => `${s.lng},${s.lat}`).join(';')],
    queryFn: async () => {
      if (stops.length < 2) return null;
      const coords = stops.map((s: any) => `${s.lng},${s.lat}`).join(';');
      const res = await fetch(`https://router.project-osrm.org/route/v1/driving/${coords}?overview=full&geometries=geojson`);
      if (!res.ok) return null;
      const json = await res.json();
      return json.routes?.[0]?.geometry;
    },
    enabled: stops.length >= 2,
    staleTime: Infinity,
  });

  // 4. Live GPS tracking
  useEffect(() => {

      try {
        const { status } = await Location.requestForegroundPermissionsAsync();
        if (status !== 'granted') { setErr('Location permission denied'); return; }
        const current = await Location.getCurrentPositionAsync({ accuracy: Location.Accuracy.High });
        if (!cancelled) setFix(toFix(current));
        sub = await Location.watchPositionAsync(
          { accuracy: Location.Accuracy.High, distanceInterval: 5, timeInterval: 3000 },
          (loc) => { if (!cancelled) setFix(toFix(loc)); },
        );
      } catch (e: any) {
        setErr(e?.message ?? 'Could not read GPS');
      }
    })();

    return () => { cancelled = true; sub?.remove(); };
  }, []);

  const centerLat = fix?.lat ?? (stops[0]?.lat ?? 28.6139);
  const centerLng = fix?.lng ?? (stops[0]?.lng ?? 77.2090);
  const markers = stops.map((s: any) => ({ lat: s.lat, lng: s.lng, label: s.label, color: '#7FA890' }));
  const routeGeometry = osrmQ.data;

  return (

        <TripMap lat={centerLat} lng={centerLng} zoom={11} height={360} markers={markers} routeGeometry={routeGeometry} />

        {err && (

            <Text style={{ color: '#F87171', fontSize: 13 }}>{err}</Text>

import Link from 'next/link';
import { useCreateSchedule } from '../hooks/useSchedules';
import { useBuses } from '@/features/buses/hooks/useBuses';

interface FieldErrors {
  busId?: string;
  routeId?: string;
  departureDate?: string;
  departureTime?: string;
  arrivalDate?: string;
  arrivalTime?: string;
  fare?: string;
  general?: string;
}

export function ScheduleForm() {
  const router = useRouter();
  const { data: buses, isLoading: loadingBuses } = useBuses();
  const { data: routes, isLoading: loadingRoutes } = useRoutes();
  const createMutation = useCreateSchedule();
  const [errors, setErrors] = useState<FieldErrors>({});

  const [formData, setFormData] = useState({
    busId: '',
    routeId: '',
    departureDate: '',
    departureTime: '',
    arrivalDate: '',
    arrivalTime: '',
    fareRupees: 500,
  });

  const availableBuses = useMemo(() => {
    return (buses || []).filter((b) => b.approvalStatus === 'APPROVED');
  }, [buses]);

  const availableRoutes = useMemo(() => {
    return (routes || []).filter((r) => (r.approvalStatus as string) !== 'REJECTED');
  }, [routes]);

  // Build a valid ISO string from date + time inputs.
  // Returns null if invalid so we can show a clear error.
  const toISO = (dateStr: string, timeStr: string): string | null => {
    if (!dateStr || !timeStr) return null;
    // date input gives "2026-11-05", time gives "10:00"
    const combined = `${dateStr}T${timeStr}:00.000`;
    const d = new Date(combined);
    if (isNaN(d.getTime())) return null;
    return d.toISOString();
  };

  const validate = (): FieldErrors => {
    const errs: FieldErrors = {};

    if (!formData.busId) errs.busId = 'Select a bus';
    if (!formData.routeId) errs.routeId = 'Select a route';
    if (!formData.departureDate) errs.departureDate = 'Required';
    if (!formData.departureTime) errs.departureTime = 'Required';
    if (!formData.arrivalDate) errs.arrivalDate = 'Required';
    if (!formData.arrivalTime) errs.arrivalTime = 'Required';
    if (!formData.fareRupees || formData.fareRupees <= 0) errs.fare = 'Fare must be > 0';

    // If date+time both present, verify they parse correctly
    if (formData.departureDate && formData.departureTime) {
      const dep = toISO(formData.departureDate, formData.departureTime);
      if (!dep) errs.departureDate = 'Invalid departure datetime';
    }
    if (formData.arrivalDate && formData.arrivalTime) {
      const arr = toISO(formData.arrivalDate, formData.arrivalTime);
      if (!arr) errs.arrivalDate = 'Invalid arrival datetime';
    }

    // Verify arrival > departure
    const dep = toISO(formData.departureDate, formData.departureTime);
    const arr = toISO(formData.arrivalDate, formData.arrivalTime);
    if (dep && arr && new Date(arr) <= new Date(dep)) {
      errs.arrivalDate = 'Arrival must be after departure';
    }

    return errs;
  };

    e.preventDefault();
    setErrors({});

    const errs = validate();

      return;
    }

        routeId: formData.routeId,

          message: err.message,
          details: (err as any).details,
        });
      } else {
        setErrors({ general: 'Failed to create schedule. Check console for details.' });
      }
    }
  };

  const hasNoBuses = !loadingBuses && availableBuses.length === 0;
  const hasNoRoutes = !loadingRoutes && availableRoutes.length === 0;

      {hasNoBuses && (
        <div className="p-4 rounded-lg bg-amber-500/10 border border-amber-500/30 text-amber-700 dark:text-amber-500 text-sm">
          <div className="flex items-start gap-3">
            <AlertCircle size={20} className="shrink-0 mt-0.5" />
            <div className="flex-1">
              <p className="font-semibold">No approved buses available</p>
              <p className="text-xs mt-1 text-amber-600/80">
                Schedules need an admin-approved bus. Approve one from Admin → Approvals → Buses, or add a new bus.
              </p>
              <Link href="/operator/buses/new" className="inline-flex items-center gap-1.5 mt-3 text-xs font-bold text-amber-700 dark:text-amber-500 hover:underline">
                <PlusCircle size={12} /> Add a new bus
              </Link>
            </div>
          </div>
        </div>
      )}

      {hasNoRoutes && (
        <div className="p-4 rounded-lg bg-amber-500/10 border border-amber-500/30 text-amber-700 dark:text-amber-500 text-sm">
          <div className="flex items-start gap-3">
            <AlertCircle size={20} className="shrink-0 mt-0.5" />
            <div className="flex-1">
              <p className="font-semibold">No routes available</p>
              <p className="text-xs mt-1 text-amber-600/80">
                Create a route first, then come back to add schedules.

    } finally {
      clearTimeout(timer);
    }
  }

  private async refreshTokens(): Promise<string> {
    const refreshToken = this.getRefresh();
    if (!refreshToken || !this.surface) {
      this.clearAuth();
      throw new ApiError(401, { code: 'SESSION_REVOKED', message: 'Session expired.' });
    }
    const res = await fetch(`${this.baseUrl}/${this.surface}/auth/refresh`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ refreshToken }),
    });
    if (!res.ok) {
      this.clearAuth();
      throw new ApiError(res.status, { code: 'SESSION_REVOKED', message: 'Session expired.' });
    }
    const payload = (await res.json()) as ApiResponse<{ accessToken: string; refreshToken: string }>;
    this.saveTokens(payload.data.accessToken, payload.data.refreshToken);
    return payload.data.accessToken;
  }

  public async request<T>(endpoint: string, opts: Options = {}): Promise<ApiResponse<T>> {
    const { method = 'GET', body, headers = {}, skipAuth = false, timeoutMs = DEFAULT_TIMEOUT_MS } = opts;
    const url = `${this.baseUrl}${endpoint.startsWith('/') ? endpoint : `/${endpoint}`}`;
    const reqHeaders: Record<string, string> = { 'Content-Type': 'application/json', ...headers };

    if (!skipAuth) {
      const token = this.getAccess();
      if (token) reqHeaders.Authorization = `Bearer ${token}`;
    }

    const init: RequestInit = { method, headers: reqHeaders };
    if (body !== undefined && method !== 'GET') init.body = JSON.stringify(body);

    let response: Response;
    try {
      response = await this.fetchWithTimeout(url, init, timeoutMs);
    } catch (err: any) {
      if (err?.name === 'AbortError') {
        throw new ApiError(408, { code: 'REQUEST_TIMEOUT', message: `Request timed out after ${timeoutMs / 1000}s.` });
      }
      throw new ApiError(0, { code: 'NETWORK_ERROR', message: err?.message || 'Network request failed.' });
    }

    if (response.status === 401 && !skipAuth) {
      if (!isRefreshing) {
        isRefreshing = true;
        try {
          const fresh = await this.refreshTokens();
          subscribers.forEach((cb) => cb(fresh));
          subscribers = [];
        } catch (err) {
          subscribers = [];
          throw err;
        } finally {
          isRefreshing = false;
        }
      }
      return new Promise((resolve, reject) => {
        subscribers.push(async (token) => {
          try {
            const retryHeaders = { ...reqHeaders, Authorization: `Bearer ${token}` };
            const replayed = await this.fetchWithTimeout(url, { ...init, headers: retryHeaders }, timeoutMs);
            resolve(await this.parse<T>(replayed));
          } catch (err) { reject(err); }
        });
      });
    }

    return this.parse<T>(response);
  }

  private async parse<T>(response: Response): Promise<ApiResponse<T>> {
    const ct = response.headers.get('content-type') || '';
    const isJson = ct.includes('application/json');

    if (!response.ok) {
      let rawText = '';
      try { rawText = await response.text(); } catch { /* ignore */ }

      let parsed: any = null;
      if (rawText) {
        try { parsed = JSON.parse(rawText); } catch { /* not JSON */ }
      }

      let errDetail: any = null;
      if (parsed) {
        if (parsed.error && typeof parsed.error === 'object' && (parsed.error.code || parsed.error.message)) {
          errDetail = parsed.error;
        } else if (parsed.code || parsed.message) {
          errDetail = parsed;
        }
      }

      if (!errDetail) {
        errDetail = {
          code: `HTTP_${response.status}`,
          message:
            response.status === 401 ? 'Not authorized â€” please sign in again.' :
            response.status === 403 ? 'You do not have access to this resource.' :
            response.status === 404 ? 'Not found.' :
            response.status === 429 ? 'Too many requests â€” please slow down.' :
            response.status >= 500 ? `Server error (HTTP ${response.status}).` :
            response.statusText || `Request failed (HTTP ${response.status})`,
        };
      }

      // QUIET LOGGING: 4xx with structured error = expected business rule (debug only)
      // 5xx or unparseable = real problem (error to console)
      const hasStructuredError = !!(errDetail?.code && errDetail?.message);
      const isExpectedBusinessError = hasStructuredError && response.status >= 400 && response.status < 500;

      if (isExpectedBusinessError) {
        console.debug(`[HTTP] ${response.status} ${errDetail.code} â†’ ${method} ${endpointOf(response.url)} â€” handled`);
      } else {
        console.error(
          `[HTTP] ${response.status} ${errDetail.code} â†’ ${method} ${endpointOf(response.url)}\n` +
          `  message: ${errDetail.message}\n` +
          `  body: ${rawText.slice(0, 500) || '(empty)'}`
        );
      }

      throw new ApiError(response.status, errDetail);
    }

    if (response.status === 204) return { data: {} as T };

    if (!isJson) {
      const text = await response.text();
      throw new ApiError(response.status, {
        code: 'INVALID_RESPONSE',
        message: `Expected JSON but got ${ct || 'unknown'}: ${text.slice(0, 200)}`,
      });
    }

    return (await response.json()) as ApiResponse<T>;
  }

  public get<T>(endpoint: string) { return this.request<T>(endpoint, { method: 'GET' }); }
  public post<T>(endpoint: string, body?: unknown, opts: Omit<Options, 'method' | 'body'> = {}) {
    return this.request<T>(endpoint, { method: 'POST', body, ...opts });
  }
  public put<T>(endpoint: string, body?: unknown) {
    return this.request<T>(endpoint, { method: 'PUT', body });
  }
  public patch<T>(endpoint: string, body?: unknown) {
    return this.request<T>(endpoint, { method: 'PATCH', body });
  }
  public delete<T>(endpoint: string) { return this.request<T>(endpoint, { method: 'DELETE' }); }
}

function endpointOf(url: string): string {
  try {
    const u = new URL(url);
    return u.pathname;
  } catch {
    return url;
  }
}

        await assignMutation.mutateAsync({
          scheduleId: schedule.id,
          driverId: selectedDriverId,
        });
      }
      onSuccess();
    } catch (err) {
      if (isApiError(err)) {
        setError(`(${err.code}) ${err.message}`);
      } else {
        setError('Failed to assign driver');
      }
    }
  };

      <div className="bg-surface-card w-full max-w-lg rounded-xl shadow-modal border border-line-hairline overflow-hidden animate-in zoom-in-95">
        <div className="flex items-center justify-between p-5 border-b border-line-hairline">

              <UserPlus size={20} />
            </div>
            <div>
              <h2 className="text-lg font-bold text-ink-bright">
                {isReassign ? 'Reassign Driver' : 'Assign Driver'}
              </h2>
              <p className="text-xs text-ink-secondary">
                {schedule.route?.originCity?.name} â†’ {schedule.route?.destinationCity?.name}
              </p>
            </div>
          </div>
          <button onClick={onClose} className="p-1 rounded-md text-ink-muted hover:text-ink-primary hover:bg-surface-hover">
            <X size={20} />
          </button>
        </div>

        <div className="p-5 space-y-4">
          {currentTrip && (
            <div className="p-3 rounded-lg bg-surface-sunken border border-line-hairline text-xs">
              <p className="text-ink-muted">Currently assigned:</p>
              <p className="font-medium text-ink-primary mt-1">
                {currentTrip.driverName} Â· {currentTrip.driverPhone}
              </p>
            </div>
          )}

          <div>
            <label className="block text-xs font-medium text-ink-secondary mb-2">
              Select a driver {loadingDrivers && '(loading...)'}
            </label>

            {!loadingDrivers && availableDrivers.length === 0 && (
              <Card className="bg-amber-500/10 border-amber-500/30 p-4">
                <div className="flex items-start gap-2 text-amber-700 dark:text-amber-500 text-xs">
                  <AlertCircle size={14} className="shrink-0 mt-0.5" />
                  <span>No verified drivers yet. Invite and verify a driver from the Drivers page first.</span>
                </div>
              </Card>
            )}

            <div className="space-y-2 max-h-64 overflow-y-auto">
              {availableDrivers.map((driver) => {
                const isSelected = selectedDriverId === driver.id;
                return (
                  <button
                    key={driver.id}
                    type="button"
                    onClick={() => setSelectedDriverId(driver.id)}
                    className={`w-full text-left p-3 rounded-lg border transition-all ${
                      isSelected
                        ? 'border-brand bg-brand/5 shadow-sm'
                        : 'border-line-hairline hover:border-ink-muted hover:bg-surface-hover'
                    }`}
                  >
                    <div className="flex items-center justify-between">
                      <div>
                        <p className="font-medium text-ink-bright">{driver.name}</p>
                        <p className="text-xs text-ink-secondary mt-0.5">{driver.phone}</p>
                        <p className="text-[10px] text-ink-muted mt-0.5 font-mono">License: {driver.licenseNumber}</p>
                      </div>
                      <div className="flex items-center gap-2">
                        <Badge tone={driver.onDuty ? 'amber' : 'emerald'}>
                          {driver.onDuty ? 'On Trip' : 'Available'}
                        </Badge>
                        {isSelected && <CheckCircle2 size={18} className="text-brand-text" />}
                      </div>
                    </div>
                  </button>
                );
              })}
            </div>
          </div>

          {error && (
            <div className="p-3 rounded-lg bg-danger-crimson/10 text-danger-crimson text-sm border border-danger-crimson/20 font-mono text-xs break-words">
              {error}
            </div>
          )}
        </div>

        <div className="flex items-center justify-end gap-3 p-5 border-t border-line-hairline bg-surface-sunken">
          <Button variant="ghost" onClick={onClose}>Cancel</Button>
          <Button
            onClick={handleAssign}
            loading={isPending}
            disabled={!selectedDriverId || availableDrivers.length === 0}
          >
            {isReassign ? 'Reassign' : 'Assign'} Driver
          </Button>
        </div>
      </div>
    </div>
  );
}
